    70→function StatChip({

   103→function JsonView({ plan }: { plan: QueryPlan }) {
   104→  const [copied, setCopied] = useState(false);
   105→  const json = JSON.stringify({ ...plan, tree: undefined }, null, 2);
   106→
   107→  const copy = async () => {
   108→    try {
   109→      await navigator.clipboard.writeText(json);
   110→      setCopied(true);
   111→      setTimeout(() => setCopied(false), 1600);
   112→    } catch {
   113→      // clipboard unavailable — ignore
   114→    }
   115→  };
   116→
   117→  return (
   118→    <div className="space-y-3">
   119→      <div className="flex items-center justify-between">
   120→        <p className="text-xs text-zinc-400">
   121→          The structured plan the visualizer derived from your query.
   122→        </p>
   123→        <Button
   124→          variant="outline"
   125→          size="sm"
   126→          onClick={copy}
   127→          className="h-7 gap-1.5 border-zinc-700 text-xs text-zinc-300 hover:bg-zinc-800"
   128→        >
   129→          {copied ? <Check className="h-3.5 w-3.5 text-emerald-400" /> : <Copy className="h-3.5 w-3.5" />}
   130→          {copied ? "Copied" : "Copy JSON"}
   131→        </Button>
   132→      </div>
   133→      <pre className="thin-scrollbar max-h-[520px] overflow-auto rounded-xl border border-zinc-800 bg-zinc-900/70 p-4 font-mono text-xs leading-relaxed text-zinc-300">
   134→        {json}
   135→      </pre>
   136→    </div>
   137→  );
   138→}
