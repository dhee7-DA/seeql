    45→const MOD = COURSE_MODULES[0];
    46→
    47→function lessonDone(
    48→  progress: ReturnType<typeof useProgress.getState>,
    49→  lessonId: string,
    50→  mcqIds: string[],
    51→  problemIds: string[],
    52→): boolean {
    53→  if (manuallyCompleted(progress, lessonId)) return true;
    54→  const allMcq = mcqIds.length > 0 && mcqIds.every((id) => mcqPassed(progress, id));
    55→  const allProblems =
    56→    problemIds.length === 0 || problemIds.every((id) => problemSolved(progress, id));
    57→  return allMcq && allProblems;
    58→}
