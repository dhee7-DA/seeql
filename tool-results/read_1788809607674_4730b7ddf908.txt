   135→   135→function PlaygroundView({
   136→   136→  go,
   137→   137→  sharedSql,
   138→   138→  sharedDialect,
   139→   139→}: {
   140→   140→  go: (m: Mode) => void;
   141→   141→  /** SQL coming from a #playground?q=… share link — boots the editor with it. */
   142→   142→  sharedSql?: string | null;
   143→   143→  sharedDialect?: DialectName;
   144→   144→}) {
   145→   145→  const [sql, setSql] = useState(sharedSql ?? DEFAULT_SAMPLE.sql);
   146→   146→  const [dialect, setDialect] = useState<DialectName>(sharedDialect ?? DEFAULT_SAMPLE.dialect);
   147→   147→  const [sampleId, setSampleId] = useState<string>(sharedSql ? "" : DEFAULT_SAMPLE.id);
   148→   148→  const [copiedLink, setCopiedLink] = useState(false);
   149→   149→  const [plan, setPlan] = useState<ParseResult | null>(null);
   150→   150→  const [exec, setExec] = useState<ExecResponse | null>(null);
   151→   151→  const [running, setRunning] = useState(false);
   152→   152→  const [history, setHistory] = useState<string[]>([]);
   153→   153→  const [resetting, setResetting] = useState(false);
   154→   154→
   155→   155→  const loadHistory = useCallback(() => {
   156→   156→    try {
   157→   157→      const raw = window.localStorage.getItem(HISTORY_KEY);
   158→   158→      if (raw) setHistory(JSON.parse(raw) as string[]);
   159→   159→    } catch {
   160→   160→      // ignore malformed history
   161→   161→    }
   162→   162→  }, []);
   163→   163→
   164→   164→  const pushHistory = useCallback((text: string) => {
   165→   165→    setHistory((prev) => {
   166→   166→      const next = [text, ...prev.filter((h) => h !== text)].slice(0, HISTORY_MAX);
   167→   167→      try {
   168→   168→        window.localStorage.setItem(HISTORY_KEY, JSON.stringify(next));
   169→   169→      } catch {
   170→   170→        // storage unavailable
   171→   171→      }
   172→   172→      return next;
   173→   173→    });
   174→   174→  }, []);
   175→   175→
   176→   176→  /** Parse + execute in parallel; both panels update independently. */
   177→   177→  const runAll = useCallback(
   178→   178→    async (sqlText: string, dia: DialectName) => {
   179→   179→      const trimmed = sqlText.trim();
   180→   180→      if (!trimmed) return;
   181→   181→      setRunning(true);
   182→   182→      try {
   183→   183→        const [parseRes, execRes] = await Promise.all([
   184→   184→          fetch("/api/parse", {
   185→   185→            method: "POST",
   186→   186→            headers: { "Content-Type": "application/json" },
   187→   187→            body: JSON.stringify({ sql: trimmed, dialect: dia }),
   188→   188→          }).then((r) => r.json() as Promise<ParseResult>),
   189→   189→          fetch("/api/query", {
   190→   190→            method: "POST",
   191→   191→            headers: { "Content-Type": "application/json" },
   192→   192→            body: JSON.stringify({ sql: trimmed }),
   193→   193→          }).then((r) => r.json() as Promise<ExecResponse>),
   194→   194→        ]);
   195→   195→        setPlan(parseRes);
   196→   196→        setExec(execRes);
   197→   197→        pushHistory(trimmed);
   198→   198→      } catch {
   199→   199→        setPlan((p) => p ?? { ok: false, error: "Could not reach the server. Please try again." });
   200→   200→        setExec({ ok: false, error: "Could not reach the server. Please try again." });
   201→   201→      } finally {
   202→   202→        setRunning(false);
   203→   203→      }
   204→   204→    },
   205→   205→    [pushHistory],
   206→   206→  );
   207→   207→
   208→   208→  const parseOnly = useCallback(async (sqlText: string, dia: DialectName) => {
   209→   209→    try {
   210→   210→      const res = await fetch("/api/parse", {
   211→   211→        method: "POST",
   212→   212→        headers: { "Content-Type": "application/json" },
   213→   213→        body: JSON.stringify({ sql: sqlText, dialect: dia }),
   214→   214→      });
   215→   215→      setPlan((await res.json()) as ParseResult);
   216→   216→    } catch {
   217→   217→      setPlan({ ok: false, error: "Could not reach the parser. Please try again." });
   218→   218→    }
   219→   219→  }, []);
   220→   220→
   221→   221→  // run the initial query once on load — a #playground?q=… share link boots
   222→   222→  // its own SQL, otherwise the default sample runs so the page shows live
   223→   223→  // diagrams + data immediately
   224→   224→  const bootSql = sharedSql ?? DEFAULT_SAMPLE.sql;
   225→   225→  const bootDialect = sharedDialect ?? DEFAULT_SAMPLE.dialect;
   226→   226→  useEffect(() => {
   227→   227→    loadHistory();
   228→   228→    runAll(bootSql, bootDialect);
   229→   229→  }, [loadHistory, runAll, bootSql, bootDialect]);
   230→   230→
   231→   231→  /** Copy a link that reopens the playground with the current query loaded. */
   232→   232→  const shareLink = async () => {
   233→   233→    const params = new URLSearchParams({ q: sql });
   234→   234→    if (dialect !== DEFAULT_SAMPLE.dialect) params.set("d", dialect);
   235→   235→    const url = `${window.location.origin}${window.location.pathname}#playground?${params.toString()}`;
   236→   236→    let ok = false;
   237→   237→    try {
   238→   238→      await navigator.clipboard.writeText(url);
   239→   239→      ok = true;
   240→   240→    } catch {
   241→   241→      // clipboard API unavailable/denied — legacy fallback for restricted contexts
   242→   242→      try {
   243→   243→        const ta = document.createElement("textarea");
   244→   244→        ta.value = url;
   245→   245→        ta.style.position = "fixed";
   246→   246→        ta.style.opacity = "0";
   247→   247→        document.body.appendChild(ta);
   248→   248→        ta.select();
   249→   249→        ok = document.execCommand("copy");
   250→   250→        document.body.removeChild(ta);
   251→   251→      } catch {
   252→   252→        ok = false;
   253→   253→      }
   254→   254→    }
   255→   255→    if (ok) {
   256→   256→      setCopiedLink(true);
   257→   257→      setTimeout(() => setCopiedLink(false), 1600);
   258→   258→    }
   259→   259→  };
   260→   260→
   261→   261→  const onSampleChange = (id: string) => {
   262→   262→    const sample = SAMPLE_QUERIES.find((s) => s.id === id);
   263→   263→    if (!sample) return;
   264→   264→    setSampleId(id);
   265→   265→    setSql(sample.sql);
   266→   266→    setDialect(sample.dialect);
   267→   267→    runAll(sample.sql, sample.dialect);
   268→   268→  };
   269→   269→
   270→   270→  const onDialectChange = (d: string) => {
   271→   271→    const next = d as DialectName;
   272→   272→    setDialect(next);
   273→   273→    parseOnly(sql, next);
   274→   274→  };
   275→   275→
   276→   276→  const onHistoryPick = (text: string) => {
   277→   277→    setSampleId("");
   278→   278→    setSql(text);
   279→   279→    runAll(text, dialect);
   280→   280→  };
   281→   281→
   282→   282→  const onSchemaPickTable = (table: string) => {
   283→   283→    const text = `SELECT *\nFROM ${table}\nLIMIT 5;`;
   284→   284→    setSampleId("");
   285→   285→    setSql(text);
   286→   286→    runAll(text, dialect);
   287→   287→  };
   288→   288→
   289→   289→  const resetDatabase = async () => {
   290→   290→    if (!window.confirm("Restore the sample database to its original data?")) return;
   291→   291→    setResetting(true);
   292→   292→    try {
   293→   293→      await fetch("/api/reset", { method: "POST" });
   294→   294→      // re-run current query so the user sees fresh results
   295→   295→      await runAll(sql, dialect);
   296→   296→    } catch {
   297→   297→      // keep silent — the re-run will surface errors if any
   298→   298→    } finally {
   299→   299→      setResetting(false);
   300→   300→    }
   301→   301→  };
   302→   302→
   303→   303→  const good = plan !== null && plan.ok;
   304→   304→  const execOk = exec !== null && exec.ok;
   305→   305→  const planSteps = execOk ? (exec as ExecResult).plan?.length ?? 0 : 0;
   306→   306→  const isWrite = execOk && (exec as ExecResult).changes !== null;
   307→   307→
   308→   308→  return (
   309→   309→    <div className="flex min-h-screen flex-col bg-zinc-950 text-zinc-100">
   310→   310→      {/* header */}
   311→   311→      <header className="sticky top-0 z-40 border-b border-zinc-800/80 bg-zinc-950/85 backdrop-blur">
   312→   312→        <div className="mx-auto flex w-full max-w-7xl flex-wrap items-center gap-x-4 gap-y-3 px-4 py-3">
   313→   313→          <div className="flex items-center gap-3">
   314→   314→            <button onClick={() => go("landing")} aria-label="SeeQL — back to home" className="transition-opacity hover:opacity-80">
   315→   315→              <LogoFull />
   316→   316→            </button>
   317→   317→            <div className="hidden border-l border-zinc-800 pl-3 sm:block">
   318→   318→              <p className="text-xs text-zinc-500">
   319→   319→                Write a query — see it parsed, executed and explained
   320→   320→              </p>
   321→   321→            </div>
   322→   322→          </div>
   323→   323→
   324→   324→          <div className="ml-auto flex flex-wrap items-center gap-2">
   325→   325→            <ThemeSwitcher />
   326→   326→            <button
   327→   327→              onClick={() => go("learn")}
   328→   328→              className="hidden items-center gap-1.5 rounded-md border border-zinc-800 px-3.5 py-2 text-sm text-zinc-300 transition-colors hover:bg-zinc-900 sm:flex"
   329→   329→            >
   330→   330→              <GraduationCap className="h-4 w-4 text-emerald-400" />
   331→   331→              Learn
   332→   332→            </button>
   333→   333→            <Select value={sampleId} onValueChange={onSampleChange}>
   334→   334→              <SelectTrigger
   335→   335→                size="sm"
   336→   336→                className="w-[190px] border-zinc-700 bg-zinc-900 text-xs sm:w-[220px]"
   337→   337→                aria-label="Load a sample query"
   338→   338→              >
   339→   339→                <SelectValue placeholder="Sample queries" />
   340→   340→              </SelectTrigger>
   341→   341→              <SelectContent>
   342→   342→                {SAMPLE_QUERIES.map((s) => (
   343→   343→                  <SelectItem key={s.id} value={s.id} className="text-xs">
   344→   344→                    <span className="font-medium">{s.name}</span>
   345→   345→                    <span className="ml-2 text-zinc-500">{s.description}</span>

   352→   352→              <SelectTrigger

   368→   368→            <Button
   369→   369→              variant="ghost"
   370→   370→              size="sm"
   371→   371→              onClick={shareLink}
   372→   372→              className="h-8 gap-1.5 px-2.5 text-xs text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200"
   373→   373→              aria-label="Copy a link to this query"
   374→   374→            >
   375→   375→              {copiedLink ? (
   376→   376→                <Check className="h-4 w-4 text-emerald-400" />
   377→   377→              ) : (
   378→   378→                <Link2 className="h-4 w-4" />
   379→   379→              )}
   380→   380→              {copiedLink ? "Copied" : "Share"}
   381→   381→            </Button>
   382→   382→
   383→   383→            <Button
   384→   384→              onClick={() => runAll(sql, dialect)}
   385→   385→              disabled={running}

   387→   387→            >

   389→   389→                <Loader2 className="h-4 w-4 animate-spin" />

   396→   396→        </div>
   397→   397→      </header>
   398→   398→
   399→   399→      {/* main */}
   400→   400→      <main className="mx-auto w-full max-w-7xl flex-1 px-4 py-5">
   401→   401→        <div className="grid items-start gap-5 lg:grid-cols-[minmax(360px,2fr)_minmax(0,3fr)]">
   402→   402→          {/* editor panel */}
   403→   403→          <section aria-label="SQL editor" className="flex flex-col gap-2">

   408→   408→              <div className="flex items-center gap-1">
   409→   409→                {history.length > 0 && (

   411→   411→                    <SelectTrigger

   719→   719→/**
   720→   720→ * App shell — single exposed route, three views:
   721→   721→ *   /            → landing
   722→   722→ *   /#playground → SQL playground (the original visualizer)
   723→   723→ *   /#learn      → the learning app
   724→   724→ * Hash syncs with browser history so back/forward work.
   725→   725→ */
   726→   726→function subscribeHash(cb: () => void) {
   727→   727→  window.addEventListener("hashchange", cb);
   728→   728→  return () => window.removeEventListener("hashchange", cb);
   729→   729→}
   730→   730→
   731→   731→export default function Home() {
   732→   732→  // The hash IS the router — subscribe to it as an external system.
   733→   733→  // useSyncExternalStore renders the server snapshot during hydration
   734→   734→  // (landing) and re-renders with the real hash right after — no
   735→   735→  // hydration mismatch, no setState-in-effect.
   736→   736→  const hash = useSyncExternalStore(
   737→   737→    subscribeHash,
   738→   738→    () => window.location.hash,
   739→   739→    () => "",
   740→   740→  );
   741→   741→  // The route is the part before "?" — a shared query rides along as
   742→   742→  // #playground?q=<encoded sql>&d=<dialect> and never breaks deep links.
   743→   743→  const qMark = hash.indexOf("?");
   744→   744→  const route = qMark === -1 ? hash : hash.slice(0, qMark);
   745→   745→  const mode: Mode =
   746→   746→    route === "#playground" ? "playground" : route === "#learn" ? "learn" : "landing";
   747→   747→
   748→   748→  let sharedSql: string | null = null;
