    64→function StatChip({
    65→  icon: Icon,
    66→  label,
    67→  value,
    68→  tone = "zinc",
    69→}: {
    70→  icon: React.ComponentType<{ className?: string }>;
    71→  label: string;
    72→  value: string | number;
    73→  tone?: "zinc" | "emerald" | "amber" | "rose" | "violet" | "teal";
    74→}) {
    75→  const tones: Record<string, string> = {
    76→    zinc: "border-zinc-700/70 bg-zinc-800/50 text-zinc-300",
    77→    emerald: "border-emerald-400/25 bg-emerald-400/10 text-emerald-200",

    79→    rose: "border-rose-400/25 bg-rose-400/10 text-rose-200",
    80→    violet: "border-violet-400/25 bg-violet-400/10 text-violet-200",
    81→    teal: "border-teal-400/25 bg-teal-400/10 text-teal-200",
    82→  };
    83→  return (
    84→    <span
    85→      className={cn(
    86→        "inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-xs font-medium",

    94→  );

    97→function JsonView({ plan }: { plan: QueryPlan }) {
    98→  const [copied, setCopied] = useState(false);
    99→  const json = JSON.stringify({ ...plan, tree: undefined }, null, 2);
   100→
   101→  const copy = async () => {
   102→    try {
   103→      await navigator.clipboard.writeText(json);
   104→      setCopied(true);
   105→      setTimeout(() => setCopied(false), 1600);
   106→    } catch {
   107→      // clipboard unavailable — ignore
   108→    }
   109→  };
   110→
   111→  return (
   112→    <div className="space-y-3">
   113→      <div className="flex items-center justify-between">
   114→        <p className="text-xs text-zinc-400">
   115→          The structured plan the visualizer derived from your query.
   116→        </p>
   117→        <Button
   118→          variant="outline"
   119→          size="sm"
   120→          onClick={copy}
   121→          className="h-7 gap-1.5 border-zinc-700 text-xs text-zinc-300 hover:bg-zinc-800"
   122→        >
   123→          {copied ? <Check className="h-3.5 w-3.5 text-emerald-400" /> : <Copy className="h-3.5 w-3.5" />}
   124→          {copied ? "Copied" : "Copy JSON"}
   125→        </Button>
   126→      </div>
   127→      <pre className="thin-scrollbar max-h-[520px] overflow-auto rounded-xl border border-zinc-800 bg-zinc-900/70 p-4 font-mono text-xs leading-relaxed text-zinc-300">
   128→        {json}
   129→      </pre>
   130→    </div>
   131→  );
   132→}

   158→        window.localStorage.setItem(HISTORY_KEY, JSON.stringify(next));

   177→            body: JSON.stringify({ sql: trimmed, dialect: dia }),

   180→            method: "POST",
