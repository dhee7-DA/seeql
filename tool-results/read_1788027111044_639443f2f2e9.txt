   100→function JsonView({ plan }: { plan: QueryPlan }) {
   101→  const [copied, setCopied] = useState(false);
   102→  const json = JSON.stringify({ ...plan, tree: undefined }, null, 2);
   103→
   104→  const copy = async () => {
   105→    try {
   106→      await navigator.clipboard.writeText(json);
   107→      setCopied(true);
   108→      setTimeout(() => setCopied(false), 1600);
   109→    } catch {
   110→      // clipboard unavailable — ignore
   111→    }
   112→  };
   113→
   114→  return (
   115→    <div className="space-y-3">
   116→      <div className="flex items-center justify-between">
   117→        <p className="text-xs text-zinc-400">
   118→          The structured plan the visualizer derived from your query.
   119→        </p>
   120→        <Button
   121→          variant="outline"
   122→          size="sm"
   123→          onClick={copy}
   124→          className="h-7 gap-1.5 border-zinc-700 text-xs text-zinc-300 hover:bg-zinc-800"
   125→        >
   126→          {copied ? <Check className="h-3.5 w-3.5 text-emerald-400" /> : <Copy className="h-3.5 w-3.5" />}
   127→          {copied ? "Copied" : "Copy JSON"}
   128→        </Button>
   129→      </div>
   130→      <pre className="thin-scrollbar max-h-[520px] overflow-auto rounded-xl border border-zinc-800 bg-zinc-900/70 p-4 font-mono text-xs leading-relaxed text-zinc-300">
   131→        {json}
   132→      </pre>
   133→    </div>
   134→  );
   135→}
