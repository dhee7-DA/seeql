    73→    73→function useBadgeWatcher() {
    74→    74→  const xpEvents = useProgress((s) => s.xpEvents);
    75→    75→  const mcqAttempts = useProgress((s) => s.mcqAttempts);
    76→    76→  const badges = useProgress((s) => s.badges);
    77→    77→  const xpByDay = useProgress((s) => s.xpByDay);
    78→    78→  const dailyGoal = useProgress((s) => s.dailyGoal);
    79→    79→  const prevLevel = useRef(-1);
    80→    80→  const ready = useRef(false);
    81→    81→
    82→    82→  const xp = useMemo(() => totalXp({ xpEvents }), [xpEvents]);
    83→    83→  const level = levelFor(xp);
    84→    84→
    85→    85→  const ctx = useMemo(() => {
    86→    86→    const problemsSolved = Object.keys(xpEvents).filter((k) => k.startsWith("solve:")).length;
    87→    87→    const mcqsCorrect = Object.values(mcqAttempts).filter((a) => a.correct).length;
    88→    88→    const lessonsDone = Object.keys(xpEvents).filter((k) => k.startsWith("lesson:")).length;
    89→    89→    const perfectQuizzes = LESSONS.filter((l) =>
    90→    90→      l.mcqs.every((m) => mcqAttempts[m.id]?.firstTry),
    91→    91→    ).length;
    92→    92→    return { lessonsDone, problemsSolved, mcqsCorrect, perfectQuizzes };
    93→    93→  }, [xpEvents, mcqAttempts]);
    94→    94→
    95→    95→  const streak = useMemo(() => currentStreak({ xpByDay, dailyGoal }), [xpByDay, dailyGoal]);
    96→    96→
    97→    97→  useEffect(() => {
    98→    98→    if (!ready.current) {
    99→    99→      ready.current = true;
   100→   100→      prevLevel.current = level.index;
   101→   101→      return;
   102→   102→    }
   103→   103→    const store = useProgress.getState();
   104→   104→    for (const b of BADGES) {
   105→   105→      if (store.badges[b.id] === undefined && b.test(ctx, xp, streak)) {
   106→   106→        if (store.unlockBadge(b.id)) {
   107→   107→          celebrate({ kind: "badge", title: `Badge · ${b.title}`, sub: b.desc });
   108→   108→        }
   109→   109→      }
   110→   110→    }
   111→   111→    if (level.index > prevLevel.current) {
   112→   112→      celebrate({ kind: "level", title: `Level up — ${level.title}`, sub: `Rank ${level.index + 1} of 6` });
   113→   113→      prevLevel.current = level.index;
   114→   114→    }
   115→   115→  }, [ctx, xp, streak, level]);
   116→   116→
   117→   117→  return { xp, level, streak, ctx };
   118→   118→}
