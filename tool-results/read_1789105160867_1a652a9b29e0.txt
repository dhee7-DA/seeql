   131→async function runInline(sql: string, sandbox: boolean): Promise<SingleResult> {
   132→  const r = await fetch("/api/query", {
   133→    method: "POST",
   134→    headers: { "Content-Type": "application/json" },
   135→    body: JSON.stringify({ sql, sandbox }),
   136→  });
   137→  return (await r.json()) as SingleResult;
   138→}
   139→
   140→async function runBatch(sqls: string[]): Promise<{ ok: boolean; results?: BatchItemResult[]; error?: string }> {
   141→  const r = await fetch("/api/query", {
   142→    method: "POST",
   143→    headers: { "Content-Type": "application/json" },
   144→    body: JSON.stringify({ sqls }),
   145→  });
   146→  return (await r.json()) as { ok: boolean; results?: BatchItemResult[]; error?: string };
   147→}

   149→function CodeBlock({ sql, caption, runnable = true }: { sql: string; caption?: string; runnable?: boolean }) {
   150→  const [result, setResult] = useState<SingleResult | null>(null);
   151→  const [batch, setBatch] = useState<BatchItemResult[] | null>(null);
   152→  const [batchError, setBatchError] = useState<string | null>(null);
   153→  const [busy, setBusy] = useState(false);
   154→  // Multi-statement blocks run as one sandboxed sequence — the statements see
   155→  // each other's effects and nothing persists.
   156→  const statements = useMemo(() => splitStatements(sql), [sql]);
   157→  const isMulti = statements.length > 1;
   158→  const isRead = looksLikeRead(sql);
   159→  const canRun = runnable && sql.trim().length > 0;
   160→
   161→  async function run() {
   162→    if (busy) return;
   163→    setBusy(true);
   164→    setBatchError(null);
   165→    try {
   166→      if (isMulti) {
   167→        setResult(null);
   168→        setBatch(null);
   169→        const r = await runBatch(statements);
   170→        if (r.ok && r.results) setBatch(r.results);
   171→        else setBatchError(r.error ?? "The sequence could not be run.");
   172→      } else {
   173→        setBatch(null);
   174→        // Writes and DDL run sandboxed: the statement really executes,
   175→        // then the whole thing is rolled back — the store stays untouched.
   176→        setResult(await runInline(sql, !isRead));
   177→      }
   178→    } finally {
   179→      setBusy(false);
   180→    }
   181→  }
   182→
   183→  const affected = result?.ok && (result.rows ?? []).length === 0 && result.changes != null;
   184→  const label = isMulti
   185→    ? `Run all · ${statements.length} steps`
   186→    : isRead ? "Run here" : "Run sandboxed";
   187→
   188→  return (
   189→    <div className="my-4 overflow-hidden rounded-lg border border-zinc-800 bg-zinc-950">
   190→      {(caption || canRun) && (
   191→        <div className="flex items-center justify-between border-b border-zinc-800 px-3.5 py-2">
   192→          <span className="text-[11px] font-medium uppercase tracking-wider text-zinc-500">{caption ?? "SQL"}</span>
   193→          {canRun && (
   194→            <button
   195→              onClick={run}
   196→              disabled={busy}
   197→              className="rounded border border-emerald-400/40 px-2 py-0.5 font-mono text-[11px] text-emerald-300 hover:bg-emerald-400/10 disabled:opacity-50"
   198→            >
   199→              {busy ? "running…" : label}
   200→            </button>
   201→          )}
   202→        </div>
   203→      )}
   204→      <pre className="overflow-x-auto px-3.5 py-3 font-mono text-xs leading-relaxed text-zinc-300 thin-scrollbar"><SqlNodes text={sql} /></pre>
   205→      {result && (
   206→        <div className="border-t border-zinc-800 p-3">
   207→          {result.ok ? (
   208→            affected ? (
   209→              <p className="font-mono text-xs text-zinc-400">
   210→                <span className="text-emerald-300">{result.changes}</span> row(s) affected — sandboxed run, rolled back.
   211→              </p>
   212→            ) : (
   213→              <ResultGrid data={{ columns: result.columns ?? [], rows: (result.rows ?? []) as never[] }} maxHeight="max-h-56" />
   214→            )
   215→          ) : (
   216→            <p className="font-mono text-xs text-rose-300">{result.error}</p>
   217→          )}
   218→        </div>
   219→      )}
   220→      {batch && (
   221→        <div className="divide-y divide-zinc-800/70 border-t border-zinc-800">
   222→          {batch.map((item, i) => (
   223→            <div key={i} className="p-3">
   224→              <div className="mb-2 flex items-center gap-2">
   225→                <span className="flex h-4.5 w-4.5 items-center justify-center rounded border border-zinc-700 bg-zinc-900 font-mono text-[10px] text-zinc-400">
   226→                  {i + 1}
   227→                </span>
   228→                <span className="font-mono text-[10px] uppercase tracking-wider text-zinc-500">
   229→                  {item.statementType ?? "SQL"}
   230→                </span>
   231→                {item.ok ? (
   232→                  <span className="font-mono text-[10px] text-emerald-300/90">
   233→                    {item.rows && item.rows.length > 0
   234→                      ? `${item.rows.length} row${item.rows.length === 1 ? "" : "s"}`
   235→                      : item.changes != null ? `${item.changes} affected` : "ok"}
   236→                  </span>
   237→                ) : (
   238→                  <span className="font-mono text-[10px] text-rose-300">error</span>
   239→                )}
   240→              </div>
   241→              {item.ok ? (
   242→                item.rows && item.rows.length > 0 ? (
   243→                  <ResultGrid data={{ columns: item.columns ?? [], rows: item.rows as never[] }} maxHeight="max-h-44" />
   244→                ) : (
   245→                  <p className="font-mono text-xs text-zinc-400">
   246→                    <span className="text-emerald-300">{item.changes}</span> row(s) affected — sandboxed run, rolled back.
   247→                  </p>
   248→                )
   249→              ) : (
   250→                <p className="font-mono text-xs text-rose-300">{item.error}</p>
   251→              )}
   252→            </div>
   253→          ))}
   254→        </div>
   255→      )}
   256→      {batchError && (
   257→        <div className="border-t border-zinc-800 p-3">
   258→          <p className="font-mono text-xs text-rose-300">{batchError}</p>
   259→        </div>
   260→      )}
   261→    </div>
   262→  );
   263→}

   430→export function LessonView({ lesson, onBonusChecked }: { lesson: Lesson; onBonusChecked?: () => void }) {
   431→  const [tab, setTab] = useState<Tab>("learn");
   432→  const store = useProgress();
   433→  const bonusRef = useRef(false);
   434→
   435→  const quizStats = useMemo(() => {
   436→    const first = lesson.mcqs.every((m) => store.mcqAttempts[m.id]?.firstTry);
   437→    const allChecked = lesson.mcqs.every((m) => store.mcqAttempts[m.id] !== undefined);
   438→    return { allChecked, first, allCorrect: lesson.mcqs.every((m) => store.mcqAttempts[m.id]?.correct) };
   439→  }, [lesson, store.mcqAttempts]);
   440→
   441→  const solvedCount = lesson.problems.filter((p) => store.xpEvents[`solve:${p.id}`] !== undefined).length;
   442→  const complete = quizStats.allCorrect && solvedCount === lesson.problems.length;
   443→
   444→  // lesson completion bonus — checked after quiz/practice events
   445→  function checkBonus() {
   446→    if (bonusRef.current) return;
   447→    const s = useProgress.getState();
   448→    const allCorrect = lesson.mcqs.every((m) => s.mcqAttempts[m.id]?.correct);
   449→    const allSolved = lesson.problems.every((p) => s.xpEvents[`solve:${p.id}`] !== undefined);
   450→    if (allCorrect && allSolved && s.awardXp(`lesson:${lesson.id}`, 50)) {
   451→      celebrate({ kind: "level", title: `Lesson ${lesson.num} complete · +50 XP`, sub: lesson.title });
   452→      s.markLessonDone(lesson.id);
   453→      bonusRef.current = true;
   454→      onBonusChecked?.();
   455→    }
   456→  }
   457→
   458→  const tabs: Array<{ id: Tab; label: string; icon: typeof BookOpen; note: string }> = [
   459→    { id: "learn", label: "Learn", icon: BookOpen, note: `${lesson.minutes} min` },
   460→    ...(lesson.deep?.length
   461→      ? [{ id: "deep" as Tab, label: "Deep dive", icon: Microscope, note: "go further" }]
   462→      : []),
   463→    { id: "quiz", label: "Quiz", icon: CircleHelp, note: `${lesson.mcqs.length} questions` },
   464→    {
   465→      id: "practice",
   466→      label: "Practice",
   467→      icon: TerminalSquare,
   468→      note: `${solvedCount}/${lesson.problems.length} solved`,
   469→    },
   470→  ];
   471→
   472→  return (
   473→    <div>
