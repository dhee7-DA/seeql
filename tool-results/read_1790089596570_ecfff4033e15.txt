    43→function Reveal({ children, delay = 0, className }: { children: React.ReactNode; delay?: number; className?: string }) {
    44→  return (
    45→    <motion.div
    46→      initial={{ opacity: 0, y: 14 }}
    47→      whileInView={{ opacity: 1, y: 0 }}
    48→      viewport={{ once: true, margin: "-60px" }}
    49→      transition={{ duration: 0.5, delay }}
    50→      className={className}
    51→    >
    52→      {children}
    53→    </motion.div>
    54→  );

    57→/** Count-up numeral — springs from 0 to n the first time it scrolls into view. */
    58→function CountUp({ value }: { value: number }) {
    59→  const ref = useRef<HTMLSpanElement>(null);
    60→  const inView = useInView(ref, { once: true, margin: "-40px" });
    61→  const mv = useMotionValue(0);
    62→  const spring = useSpring(mv, { stiffness: 60, damping: 18 });
    63→  const [display, setDisplay] = useState(0);
    64→
    65→  useEffect(() => {
    66→    if (inView) mv.set(value);
    67→  }, [inView, mv, value]);
    68→  useEffect(() => spring.on("change", (v) => setDisplay(Math.round(v))), [spring]);
    69→
    70→  return (
    71→    <span ref={ref} className="tnum">
    72→      {display.toLocaleString()}
    73→    </span>
    74→  );
    75→}

    98→/** The live landing mascot: eyes follow the cursor, poke or a move chip
    99→    makes him perform, then he settles back to idle. Same 1:1 bloub engine
   100→    the corner buddy runs — no mouth, two capsule eyes, that's the character. */
   101→function SeeqMascot() {
   102→  const [command, setCommand] = useState<BloubCommand>({ state: "idle", nonce: 0 });
   103→  const nonce = useRef(0);
   104→  const revert = useRef<number | null>(null);
   105→
   106→  useEffect(
   107→    () => () => {
   108→      if (revert.current !== null) window.clearTimeout(revert.current);
   109→    },
   110→    [],
   111→  );
   112→
   113→  const play = (state: StateId, hold: number) => {
   114→    if (revert.current !== null) window.clearTimeout(revert.current);
   115→    nonce.current += 1;
   116→    setCommand({ state, nonce: nonce.current });
   117→    revert.current = window.setTimeout(() => {
   118→      nonce.current += 1;
   119→      setCommand({ state: "idle", nonce: nonce.current });
   120→    }, hold * 1000);
   121→  };
   122→
   123→  return (
   124→    <div data-seeq-stage="1" className="flex flex-col items-center gap-5">
   125→      <button
   126→        type="button"
   127→        onClick={() => play("wide", 1.6)}
   128→        data-seeq-hero="1"
   129→        aria-label="Poke SeeQ — he performs a move"
   130→        className="block cursor-pointer rounded-full outline-none transition-transform duration-300 hover:scale-105 focus-visible:ring-2 focus-visible:ring-emerald-400/60"
   131→      >
   132→        <BloubBot
   133→          size={176}
   134→          paper="#f9f9f9"
   135→          contour="rgba(255,255,255,0.26)"
   136→          follow
   137→          command={command}
   138→          className="h-40 w-40 drop-shadow-[0_18px_40px_rgba(0,0,0,0.55)] sm:h-44 sm:w-44"
   139→          ariaLabel="SeeQ — his eyes follow your cursor"
   140→        />
   141→      </button>
   142→      <div className="flex max-w-[280px] flex-wrap justify-center gap-1.5">
   143→        {SEEQ_MOVES.map((m) => (
   144→          <button
   145→            key={m.id}
   146→            type="button"
   147→            onClick={() => play(m.id, m.hold)}
   148→            data-seeq-chip={m.id}
   149→            className="cursor-pointer rounded-full border border-zinc-800 bg-zinc-900/50 px-2.5 py-1 text-[11px] text-zinc-400 transition-colors hover:border-zinc-600 hover:text-zinc-200"
   150→          >
   151→            {m.label}
   152→          </button>
   153→        ))}
   154→      </div>
   155→    </div>
   156→  );
   157→}
