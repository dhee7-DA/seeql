    76→function loadOpenModules(): number[] | null {
    77→  try {
    78→    const raw = localStorage.getItem(MODULES_OPEN_KEY);
    79→    if (!raw) return null;
    80→    const arr = JSON.parse(raw);
    81→    return Array.isArray(arr) ? arr.filter((n) => typeof n === "number") : null;
    82→  } catch {
    83→    return null;
    84→  }
    85→}

   111→function useBadgeWatcher() {
   112→  const xpEvents = useProgress((s) => s.xpEvents);
   113→  const mcqAttempts = useProgress((s) => s.mcqAttempts);
   114→  const badges = useProgress((s) => s.badges);
   115→  const xpByDay = useProgress((s) => s.xpByDay);
   116→  const dailyGoal = useProgress((s) => s.dailyGoal);
   117→  const prevLevel = useRef(-1);
   118→  const ready = useRef(false);
   119→
   120→  const xp = useMemo(() => totalXp({ xpEvents }), [xpEvents]);
   121→  const level = levelFor(xp);
   122→
   123→  const ctx = useMemo(() => {
   124→    const problemsSolved = Object.keys(xpEvents).filter((k) => k.startsWith("solve:")).length;
   125→    const mcqsCorrect = Object.values(mcqAttempts).filter((a) => a.correct).length;
   126→    const lessonsDone = Object.keys(xpEvents).filter((k) => k.startsWith("lesson:")).length;
   127→    const perfectQuizzes = LESSONS.filter((l) =>
   128→      l.mcqs.every((m) => mcqAttempts[m.id]?.firstTry),
   129→    ).length;
   130→    return { lessonsDone, problemsSolved, mcqsCorrect, perfectQuizzes };
   131→  }, [xpEvents, mcqAttempts]);
   132→
   133→  const streak = useMemo(() => currentStreak({ xpByDay, dailyGoal }), [xpByDay, dailyGoal]);
   134→
   135→  useEffect(() => {
   136→    if (!ready.current) {
   137→      ready.current = true;
   138→      prevLevel.current = level.index;
   139→      return;
   140→    }
   141→    const store = useProgress.getState();
   142→    for (const b of BADGES) {
   143→      if (store.badges[b.id] === undefined && b.test(ctx, xp, streak)) {
   144→        if (store.unlockBadge(b.id)) {
   145→          celebrate({ kind: "badge", title: `Badge · ${b.title}`, sub: b.desc });
   146→        }
   147→      }
   148→    }
   149→    if (level.index > prevLevel.current) {
   150→      celebrate({ kind: "level", title: `Level up — ${level.title}`, sub: `Rank ${level.index + 1} of 6` });
   151→      prevLevel.current = level.index;
   152→    }
   153→  }, [ctx, xp, streak, level]);
   154→
   155→  return { xp, level, streak, ctx };
   156→}

   177→      </div>
   178→    </div>

   240→              const done = xpEvents[`lesson:${l.id}`] !== undefined;
   241→              return (
   242→                <button
   243→                  key={l.id}
   244→                  onClick={() => onOpenLesson(l.id)}
   245→                  className="group/lesson flex items-center gap-2.5 rounded-md px-2 py-1.5 text-left transition-colors hover:bg-zinc-900"
   246→                >

   270→            </button>

   273→      </Collapse>
   274→    </div>
   275→  );
   276→}
