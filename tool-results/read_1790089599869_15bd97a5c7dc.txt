    57→    57→/** Count-up numeral — springs from 0 to n the first time it scrolls into view. */
    58→    58→function CountUp({ value }: { value: number }) {
    59→    59→  const ref = useRef<HTMLSpanElement>(null);
    60→    60→  const inView = useInView(ref, { once: true, margin: "-40px" });
    61→    61→  const mv = useMotionValue(0);
    62→    62→  const spring = useSpring(mv, { stiffness: 60, damping: 18 });
    63→    63→  const [display, setDisplay] = useState(0);
    64→    64→
    65→    65→  useEffect(() => {
    66→    66→    if (inView) mv.set(value);
    67→    67→  }, [inView, mv, value]);
    68→    68→  useEffect(() => spring.on("change", (v) => setDisplay(Math.round(v))), [spring]);
    69→    69→
    70→    70→  return (
    71→    71→    <span ref={ref} className="tnum">
    72→    72→      {display.toLocaleString()}
    73→    73→    </span>
