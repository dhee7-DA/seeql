   110→   110→function useBadgeWatcher() {
   111→   111→  const xpEvents = useProgress((s) => s.xpEvents);
   112→   112→  const mcqAttempts = useProgress((s) => s.mcqAttempts);
   113→   113→  const badges = useProgress((s) => s.badges);
   114→   114→  const xpByDay = useProgress((s) => s.xpByDay);
   115→   115→  const dailyGoal = useProgress((s) => s.dailyGoal);
   116→   116→  const prevLevel = useRef(-1);
   117→   117→  const ready = useRef(false);
   118→   118→
   119→   119→  const xp = useMemo(() => totalXp({ xpEvents }), [xpEvents]);
   120→   120→  const level = levelFor(xp);
   121→   121→
   122→   122→  const ctx = useMemo(() => {
   123→   123→    const problemsSolved = Object.keys(xpEvents).filter((k) => k.startsWith("solve:")).length;
   124→   124→    const mcqsCorrect = Object.values(mcqAttempts).filter((a) => a.correct).length;
   125→   125→    const lessonsDone = Object.keys(xpEvents).filter((k) => k.startsWith("lesson:")).length;
   126→   126→    const perfectQuizzes = LESSONS.filter((l) =>
   127→   127→      l.mcqs.every((m) => mcqAttempts[m.id]?.firstTry),
   128→   128→    ).length;
   129→   129→    return { lessonsDone, problemsSolved, mcqsCorrect, perfectQuizzes };
   130→   130→  }, [xpEvents, mcqAttempts]);
   131→   131→
   132→   132→  const streak = useMemo(() => currentStreak({ xpByDay, dailyGoal }), [xpByDay, dailyGoal]);
   133→   133→
   134→   134→  useEffect(() => {
   135→   135→    if (!ready.current) {
   136→   136→      ready.current = true;
   137→   137→      prevLevel.current = level.index;
   138→   138→      return;
   139→   139→    }
   140→   140→    const store = useProgress.getState();
   141→   141→    for (const b of BADGES) {
   142→   142→      if (store.badges[b.id] === undefined && b.test(ctx, xp, streak)) {
   143→   143→        if (store.unlockBadge(b.id)) {
   144→   144→          celebrate({ kind: "badge", title: `Badge · ${b.title}`, sub: b.desc });
   145→   145→        }
   146→   146→      }
   147→   147→    }
   148→   148→    if (level.index > prevLevel.current) {
   149→   149→      celebrate({ kind: "level", title: `Level up — ${level.title}`, sub: `Rank ${level.index + 1} of 6` });
   150→   150→      prevLevel.current = level.index;
   151→   151→    }
   152→   152→  }, [ctx, xp, streak, level]);
   153→   153→
   154→   154→  return { xp, level, streak, ctx };
   155→   155→}
