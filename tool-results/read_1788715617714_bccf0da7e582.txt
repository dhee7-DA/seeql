    61→/** Every clause keyword with its position, sorted by where it appears —
    62→ *  the bridge between caret, editor keyword glow, playback frames and the
    63→ *  schema map. A clause OWNS the whole region from its keyword to the next
    64→ *  clause's keyword, so "caret inside WHERE …" lights up WHERE, not just
    65→ *  the exact letters W-H-E-R-E. */
    66→const CLAUSE_RES: Array<[ClauseId, RegExp]> = [
    67→  ["base", /\b(?:from|join|inner|left|right|full|outer|cross|on|using)\b/gi],
    68→  ["where", /\bwhere\b/gi],
    69→  ["group", /\bgroup\s+by\b/gi],
    70→  ["having", /\bhaving\b/gi],
    71→  ["result", /\b(?:select|distinct|order\s+by|limit|offset)\b/gi],
    72→];
    73→
    74→function clauseAt(sql: string, pos: number): ClauseId | null {
    75→  const marks: Array<{ index: number; id: ClauseId }> = [];
    76→  for (const [id, re] of CLAUSE_RES) {
    77→    re.lastIndex = 0;
    78→    let m: RegExpExecArray | null;
    79→    while ((m = re.exec(sql)) !== null) marks.push({ index: m.index, id });
    80→  }
    81→  marks.sort((a, b) => a.index - b.index);
    82→  // The clause whose keyword appears last at-or-before the caret owns it —
    83→  // SELECT's region starts at its keyword, so a caret in the projection
    84→  // list lands on "result" (SELECT shapes the output, it filters nothing).
    85→  let owner: ClauseId | null = null;
    86→  for (const mk of marks) {
    87→    if (mk.index <= pos) owner = mk.id;
    88→    else break;
    89→  }
    90→  return owner;
    91→}

   288→export function LiveSandbox() {
   289→  const [sql, setSql] = useState(QUESTIONS[0].sql);
   290→  const [res, setRes] = useState<SandboxResult | null>(null);
   291→  const [pending, setPending] = useState(false);
   292→  const [runId, setRunId] = useState(0);
   293→  const [knownTables, setKnownTables] = useState<SandboxTable[]>([]);
   294→  // last successful run's sources — drives the "tables your query touches"
   295→  // glow in the schema map. Kept across failed runs (that's when you need
   296→  // the map most) just like knownTables.
   297→  const [lastSources, setLastSources] = useState<SandboxSource[]>([]);
   298→  // clause focus — hover (playback frames) wins over caret position
   299→  const [hoverClause, setHoverClause] = useState<ClauseId | null>(null);
   300→  const [caretPos, setCaretPos] = useState<number | null>(null);
   301→  // A/B compare — the pinned "A" snapshot
   302→  const [pinned, setPinned] = useState<CompareSnapshot | null>(null);
   303→  // the on-screen guide — open for first-timers, collapsible forever after
   304→  const [guideOpen, setGuideOpen] = useState(true);
   305→  // fix-the-query puzzles — public list, active one, progressive hints, solved set
   306→  const [puzzles, setPuzzles] = useState<PublicPuzzle[]>([]);
   307→  const [activePuzzleId, setActivePuzzleId] = useState<string | null>(null);
   308→  const [revealedHints, setRevealedHints] = useState<Record<string, number>>({});
   309→  const [solvedIds, setSolvedIds] = useState<string[]>([]);
   310→  const activePuzzleRef = useRef<string | null>(null);
   311→  const editorTaRef = useRef<HTMLTextAreaElement | null>(null);
   312→  const seqRef = useRef(0);
   313→  const abortRef = useRef<AbortController | null>(null);
   314→
   315→  // puzzles + past solves — answers stay server-side; badges persist locally
   316→  useEffect(() => {
   317→    fetch("/api/landing-puzzles")
   318→      .then((r) => r.json())
   319→      .then((d: { puzzles?: PublicPuzzle[] }) => {
   320→        if (Array.isArray(d.puzzles)) setPuzzles(d.puzzles);
   321→      })
   322→      .catch(() => {});
   323→    try {
   324→      const raw = localStorage.getItem("seeql-puzzles-solved");
   325→      if (raw) setSolvedIds(JSON.parse(raw) as string[]);
   326→    } catch {
   327→      /* fresh start */
   328→    }
   329→  }, []);
   330→
   331→  useEffect(() => {
   332→    activePuzzleRef.current = activePuzzleId;
   333→  }, [activePuzzleId]);
   334→
   335→  const markSolved = useCallback((id: string) => {
   336→    setSolvedIds((prev) => {
   337→      if (prev.includes(id)) return prev;
   338→      const next = [...prev, id];
   339→      try {
   340→        localStorage.setItem("seeql-puzzles-solved", JSON.stringify(next));
   341→      } catch {
   342→        /* private mode — badge just won't persist */
   343→      }
   344→      return next;
   345→    });
   346→  }, []);
   347→
   348→  const run = useCallback(async (q: string) => {
   349→    const seq = ++seqRef.current;
   350→    abortRef.current?.abort();
   351→    const ac = new AbortController();
   352→    abortRef.current = ac;
   353→    setPending(true);
   354→    try {
   355→      const r = await fetch("/api/landing-sandbox", {
   356→        method: "POST",
   357→        headers: { "content-type": "application/json" },
   358→        body: JSON.stringify({
   359→          sql: q,
   360→          ...(activePuzzleRef.current ? { puzzleId: activePuzzleRef.current } : {}),
   361→        }),
   362→        signal: ac.signal,
   363→      });
   364→      const j = (await r.json()) as SandboxResult;
   365→      if (seq === seqRef.current) {
   366→        setRes(j);
   367→        // Cache the schema — error responses don't carry tables, and the
   368→        // schema panel must survive (it's exactly what you need then).
   369→        if (j.tables && j.tables.length > 0) setKnownTables(j.tables);
   370→        if (j.ok) {
   371→          setRunId((n) => n + 1);
   372→          // A successful run OWNS the glow — even when it touches no known
   373→          // tables (VALUES, scalar only): stale glow would read as "the map
   374→          // doesn't follow my query". Only error responses keep the old map.
   375→          setLastSources(j.sources ?? []);
   376→        }
   377→      }
   378→    } catch (e) {
   379→      if ((e as Error | undefined)?.name !== "AbortError" && seq === seqRef.current) {
   380→        setRes({ ok: false, error: "Could not reach the sandbox." });
   381→      }
   382→    } finally {
   383→      if (seq === seqRef.current) setPending(false);
   384→    }
   385→  }, []);
   386→
   387→  // debounced auto-run — the sandbox is live, no button needed
   388→  useEffect(() => {
   389→    const t = window.setTimeout(() => void run(sql), 500);
   390→    return () => window.clearTimeout(t);
   391→  }, [sql, run]);
   392→
   393→  // a solved verdict earns its badge (once)
   394→  const verdict = res?.ok && res.puzzle && res.puzzle.id === activePuzzleId ? res.puzzle : null;
   395→  useEffect(() => {
   396→    if (verdict?.solved) markSolved(verdict.id);
   397→  }, [verdict?.solved, verdict?.id, markSolved]);
   398→
   399→  const stages = res?.ok ? (res.stages ?? []) : [];
   400→  const tables = knownTables;
   401→
   402→  const caretClause = useMemo(
   403→    () => (caretPos === null ? null : clauseAt(sql, caretPos)),
   404→    [sql, caretPos],
   405→  );
   406→  const focusClause = hoverClause ?? caretClause;
   407→
   408→  const currentSnapshot: CompareSnapshot | null =
   409→    res?.ok && res.columns
   410→      ? { sql, columns: res.columns, rows: res.rows ?? [], rowCount: res.rowCount ?? 0 }
   411→      : null;
   412→
   413→  const hasLeftJoin = /\bleft\s+join\b/i.test(sql);
   414→  const hasJoin = /\bjoin\b/i.test(sql);
   415→  const flipLabel = hasLeftJoin
   416→    ? "⚖️ flip LEFT JOIN → INNER"
   417→    : hasJoin
   418→      ? "⚖️ flip JOIN → LEFT JOIN"
   419→      : null;
   420→  const flipJoin = () => {
   421→    if (currentSnapshot) setPinned(currentSnapshot);
   422→    setSql(
   423→      hasLeftJoin
   424→        ? sql.replace(/\bleft\s+join\b/i, "INNER JOIN")
   425→        : sql.replace(/\bjoin\b/i, "LEFT JOIN"),
   426→    );
   427→  };
   428→  const pinCurrent = () => {
   429→    if (currentSnapshot) setPinned(currentSnapshot);
   430→  };
   431→
   432→  // fix-the-query — load the broken query, checker verifies every run
   433→  const loadPuzzle = (p: PublicPuzzle) => {
   434→    setActivePuzzleId(p.id);
   435→    setSql(p.broken);
   436→  };
   437→  const leavePuzzle = () => setActivePuzzleId(null);
   438→  const revealHint = (id: string, n: number) =>
   439→    setRevealedHints((prev) => ({ ...prev, [id]: Math.max(prev[id] ?? 0, n) }));
   440→
   441→  /** Drop a token (table/column name) at the caret — from the schema strip. */
   442→  const insertToken = useCallback(
   443→    (token: string) => {
   444→      const ta = editorTaRef.current;
   445→      if (!ta) {
   446→        setSql((prev) => `${prev}${/[\s(]$/.test(prev) || prev === "" ? "" : " "}${token}`);
   447→        return;
   448→      }
   449→      setSql((prev) => {
   450→        const start = ta.selectionStart ?? prev.length;
   451→        const end = ta.selectionEnd ?? start;
   452→        const before = prev.slice(0, start);
   453→        const sep = before === "" || /[\s(,.\n]$/.test(before) ? "" : " ";
   454→        const next = `${before}${sep}${token}${prev.slice(end)}`;
   455→        const caret = before.length + sep.length + token.length;
   456→        requestAnimationFrame(() => {
   457→          ta.focus();
   458→          ta.selectionStart = ta.selectionEnd = caret;
   459→        });
   460→        return next;
   461→      });
   462→    },
   463→    [],
   464→  );
   465→
   466→  return (
   467→    <div className="mx-auto w-full max-w-6xl px-4 py-20 sm:px-6">
   468→      <div className="flex flex-wrap items-end justify-between gap-3">
   469→        <div>
   470→          <p className="inline-flex items-center gap-2 rounded-full border border-emerald-400/20 bg-emerald-400/10 px-3 py-1 text-xs font-medium text-emerald-300">
   471→            <span className="pulse-soft h-1.5 w-1.5 rounded-full bg-emerald-400" />
   472→            Live sandbox
   473→          </p>
   474→          <h2 className="font-display mt-4 text-3xl font-semibold tracking-[-0.02em] text-zinc-50">
   475→            Don&apos;t take our word for it. Query it.
   476→          </h2>
   477→          <p className="mt-3 max-w-2xl text-sm leading-6 text-zinc-400">
   478→            This is the real storefront database — the same one every lesson grades against. Pick a
   479→            question below or write your own SELECT: it runs on live SQLite as you type, and the
   480→            playback replays how each clause moves your rows — tables glow on the map, arrows carry
   481→            the counts. Click any table or column to insert it at your cursor. Read-only, nothing to
   482→            break.
   483→          </p>
   484→        </div>
   485→        <span className="sheen hidden rounded-full border border-zinc-800 bg-zinc-900/50 px-3 py-1 font-mono text-[11px] text-zinc-500 sm:block">
   486→          ⌘/Ctrl + Enter to run · auto-runs as you type
   487→        </span>
   488→      </div>
   489→
   490→      {/* the guide — every interactive element, explained in plain words */}
   491→      <SandboxGuide open={guideOpen} onToggle={() => setGuideOpen((o) => !o)} />
