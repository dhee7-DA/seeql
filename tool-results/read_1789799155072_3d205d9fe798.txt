    41→type Flight = {

    44→  from: THREE.Vector3;

    49→// Farther customers get a higher lane, so the four fans layer in depth
    50→// instead of crossing through each other.
    51→const LANE_LIFT = [0.95, 1.2, 1.45, 1.7] as const;
    52→const DUR = 1.5;
    53→const LAUNCH_STEP = 0.085;
    54→const TUBE_SEGS = 44;
    55→const TRAIL_R = 0.021;
    56→
    57→// Round-robin launch (1st order of every customer, then 2nd, …) so the four
    58→// colors stream as parallel lanes instead of one monochrome swarm.
    59→const FLIGHTS: Flight[] = (() => {
    60→  const out: Flight[] = [];
    61→  const maxN = Math.max(...CUSTOMERS.map((c) => c.n));
    62→  for (let i = 0; i < maxN; i++) {
    63→    CUSTOMERS.forEach((c, ci) => {
    64→      if (i >= c.n) return;
    65→      const from = new THREE.Vector3(STACK_X[ci] - 0.5, STACK_BASE + i * BAR_PITCH + 0.06, 0);
    66→      // Land fanned along the customer row's right edge — heads slot in
    67→      // side by side instead of piling onto one point.
    68→      const k = i % 5;
    69→      const to = new THREE.Vector3(CUST_X + 1.38, rowY(ci) + (k - 2) * 0.062, 0.3);
    70→      const ctrl = new THREE.Vector3(
    71→        (from.x + to.x) / 2,
    72→        (from.y + to.y) / 2 + LANE_LIFT[ci],
    73→        0.16,
    74→      );
    75→      out.push({ ci, delay: out.length * LAUNCH_STEP, from, ctrl, to });
    76→    });
    77→  }
    78→  return out;
    79→})();

    81→function JoinScene({ replayRef }: { replayRef?: { current: (() => void) | null } }) {

    86→  const startRef = useRef(0);
    87→  const clockRef = useRef(0);

    94→  const flights = useMemo(() => FLIGHTS, []);
    95→  const curves = useMemo(
    96→    () => flights.map((f) => new THREE.QuadraticBezierCurve3(f.from, f.ctrl, f.to)),
    97→    [flights],
    98→  );
    99→  const tubeGeoms = useMemo(
   100→    () => curves.map((c) => new THREE.TubeGeometry(c, TUBE_SEGS, TRAIL_R, 6, false)),
   101→    [curves],
   102→  );
   103→  useEffect(() => () => tubeGeoms.forEach((g) => g.dispose()), [tubeGeoms]);
   104→
   105→  useFrame((state, dt) => {
   106→    clockRef.current = state.clock.elapsedTime;
   107→    const e = clockRef.current - startRef.current;
   108→
   109→    // idle drift — the whole model breathes
   110→    if (groupRef.current) {
   111→      groupRef.current.rotation.y = Math.sin(e * 0.18) * 0.05;

   115→    // draw the projection ribbons

   121→      const t = (e - f.delay) / DUR;
   122→
   123→      if (t < 0) {
   124→        trail.visible = false;
   125→        head.visible = false;
   126→        return;
   127→      }
   128→
   129→      if (t >= 1) {
   130→        if (!arrivedRef.current[i]) {
   131→          arrivedRef.current[i] = true;
   132→          arrivalAtRef.current[i] = e;
   133→          flashRef.current[f.ci] = 1;
   134→          count += 1;
   135→        }
   136→        head.visible = false;
   137→        // ribbon lingers briefly after arrival, then dissolves — the rest
   138→        // state of the scene stays clean
   139→        const age = e - arrivalAtRef.current[i];
   140→        const mat = trail.material as THREE.MeshBasicMaterial;
   141→        const op = 0.5 * Math.exp(-age * 2.4);
   142→        if (op < 0.02) {
   143→          trail.visible = false;
   144→        } else {
   145→          trail.visible = true;
   146→          mat.opacity = op;
   147→        }
   148→        return;
   149→      }
   150→
   151→      // smoothstep easing — slow in, glide, slow out (no abrupt launches)
   152→      const p = t * t * (3 - 2 * t);
   153→      const mat = trail.material as THREE.MeshBasicMaterial;
   154→      trail.visible = true;
   155→      mat.opacity = Math.min(1, 0.25 + p * 10) * 0.5;
   156→      trail.geometry.setDrawRange(0, Math.max(1, Math.floor(p * TUBE_SEGS)) * 36);
   157→      head.visible = true;
   158→      curves[i].getPointAt(p, head.position);
   159→      head.scale.setScalar(1 + 0.22 * Math.sin(p * Math.PI));
   160→    });
   161→    if (count !== countRef.current) {
   162→      countRef.current = count;
   163→      setMatched(count);
   164→    }
   165→
   166→    // arrival glow decay
   167→    flashRef.current = flashRef.current.map((f) => f * Math.exp(-dt * 2.4));

   170→    });
   171→  });

   174→    startRef.current = clockRef.current;
   175→    arrivedRef.current = new Array(FLIGHTS.length).fill(false);
   176→    arrivalAtRef.current = new Array(FLIGHTS.length).fill(-1);
   177→    countRef.current = 0;
   178→    flashRef.current = [0, 0, 0, 0];
   179→    setMatched(0);
   180→  };

   182→  // publish replay so the host can auto-replay when the figure scrolls into view
   183→  useEffect(() => {
   184→    if (replayRef) replayRef.current = replay;
   185→  });

   357→function J2JoinSpace() {
   358→  const wrapRef = useRef<HTMLDivElement>(null);
   359→  const replayRef = useRef<(() => void) | null>(null);
   360→  const lastAutoRef = useRef(0);
   361→
   362→  // Auto-replay whenever the figure scrolls into view — the scene never sits
   363→  // frozen at a finished state. Cooldown keeps fast scrolling from strobing.
   364→  useEffect(() => {
   365→    const el = wrapRef.current;
   366→    if (!el || typeof IntersectionObserver === "undefined") return;
   367→    const io = new IntersectionObserver(
   368→      (entries) => {
   369→        for (const en of entries) {
   370→          if (en.isIntersecting && en.intersectionRatio >= 0.55) {
   371→            const now = performance.now();
   372→            if (now - lastAutoRef.current > 7000) {
   373→              lastAutoRef.current = now;
   374→              replayRef.current?.();
   375→            }
   376→          }
   377→        }
   378→      },
   379→      { threshold: [0.55] },
   380→    );
   381→    io.observe(el);
   382→    return () => io.disconnect();
   383→  }, []);
   384→
   385→  return (
   386→    <div
   387→      ref={wrapRef}
   388→      className="relative h-[400px] w-full"
   389→      role="img"
   390→      aria-label="Interactive 3D scene: a customers table and an orders table float in space. Each of 33 drawn order rows draws a glowing ribbon to its customer row; the full INNER JOIN result is 103 rows. Alice Chen's stack of 24 towers over Bruno Silva's 2 — the table shapes themselves show who buys most. Drag to orbit."
   391→    >
   392→      <Canvas
   393→        dpr={[1, 1.75]}
   394→        gl={{ antialias: true, alpha: true }}
   395→        camera={{ position: [0, 2.4, 12.8], fov: 40 }}
   396→      >
   397→        <SuspenseBoundary>
   398→          <JoinScene replayRef={replayRef} />
   399→        </SuspenseBoundary>
   400→      </Canvas>
   401→    </div>
   402→  );
   403→}

   496→function LeftJoinScene({ replayRef }: { replayRef?: { current: (() => void) | null } }) {
   497→  const [matched, setMatched] = useState(0);
   498→  const [gaps, setGaps] = useState(0);
   499→  const countRef = useRef(0);
   500→  const gapsRef = useRef(0);
   501→  const arrivedRef = useRef<boolean[]>(new Array(L12_FLIGHTS.length).fill(false));
   502→  const arrivalAtRef = useRef<number[]>(new Array(L12_FLIGHTS.length).fill(-1));
   503→  const gapsStartRef = useRef(-1);
   504→  const poppedRef = useRef<boolean[]>(new Array(L12_CUSTOMERS.length).fill(false));
   505→  const startRef = useRef(0);
   506→  const clockRef = useRef(0);
   507→  const flashRef = useRef<number[]>(new Array(L12_CUSTOMERS.length).fill(0));
   508→  const groupRef = useRef<THREE.Group>(null);
   509→  const trailRefs = useRef<Array<THREE.Mesh | null>>([]);
   510→  const headRefs = useRef<Array<THREE.Mesh | null>>([]);
   511→  const plaqueRefs = useRef<Array<THREE.Group | null>>([]);
   512→  const rowMats = useRef<Array<THREE.MeshStandardMaterial | null>>([]);
   513→
   514→  const flights = useMemo(() => L12_FLIGHTS, []);
   515→  const curves = useMemo(
   516→    () => flights.map((f) => new THREE.QuadraticBezierCurve3(f.from, f.ctrl, f.to)),
   517→    [flights],
   518→  );
   519→  const tubeGeoms = useMemo(
   520→    () => curves.map((c) => new THREE.TubeGeometry(c, L12_TUBE_SEGS, L12_TRAIL_R, 6, false)),
   521→    [curves],
   522→  );
   523→  useEffect(() => () => tubeGeoms.forEach((g) => g.dispose()), [tubeGeoms]);
   524→
   525→  useFrame((state, dt) => {
   526→    clockRef.current = state.clock.elapsedTime;
   527→    const e = clockRef.current - startRef.current;
   528→
   529→    if (groupRef.current) {
   530→      groupRef.current.rotation.y = Math.sin(e * 0.16) * 0.045;
   531→      groupRef.current.position.y = Math.sin(e * 0.55) * 0.035;
   532→    }
   533→
   534→    // phase A — projection ribbons for the 30 matched pairs
   535→    let count = countRef.current;
   536→    flights.forEach((f, i) => {
   537→      const trail = trailRefs.current[i];
   538→      const head = headRefs.current[i];
   539→      if (!trail || !head) return;
   540→      const t = (e - f.delay) / L12_DUR;
   541→
   542→      if (t < 0) {
   543→        trail.visible = false;
   544→        head.visible = false;
   545→        return;
   546→      }
   547→
   548→      if (t >= 1) {
   549→        if (!arrivedRef.current[i]) {
   550→          arrivedRef.current[i] = true;
   551→          arrivalAtRef.current[i] = e;
   552→          flashRef.current[f.ci] = 1;
   553→          count += 1;
   554→        }
   555→        head.visible = false;
   556→        const age = e - arrivalAtRef.current[i];
   557→        const mat = trail.material as THREE.MeshBasicMaterial;
   558→        const op = 0.5 * Math.exp(-age * 2.4);
   559→        if (op < 0.02) {
   560→          trail.visible = false;
   561→        } else {
   562→          trail.visible = true;
   563→          mat.opacity = op;
   564→        }
   565→        return;
   566→      }
   567→
   568→      const p = t * t * (3 - 2 * t);
   569→      const mat = trail.material as THREE.MeshBasicMaterial;
   570→      trail.visible = true;
   571→      mat.opacity = Math.min(1, 0.25 + p * 10) * 0.5;
   572→      trail.geometry.setDrawRange(0, Math.max(1, Math.floor(p * L12_TUBE_SEGS)) * 36);
   573→      head.visible = true;
   574→      curves[i].getPointAt(p, head.position);
   575→      head.scale.setScalar(1 + 0.22 * Math.sin(p * Math.PI));
   576→    });
   577→    if (count !== countRef.current) {
   578→      countRef.current = count;
   579→      setMatched(count);
   580→    }
   581→
   582→    // phase B — after every order landed, the 6 gap customers get their
   583→    // NULL plaques: the rows a WHERE clause would have deleted
   584→    if (countRef.current >= flights.length) {
   585→      if (gapsStartRef.current < 0) gapsStartRef.current = e + 0.45;
   586→      const ge = e - gapsStartRef.current;
   587→      if (ge >= 0) {
   588→        L12_CUSTOMERS.forEach((c, ci) => {
   589→          if (c.shipped > 0) return;
   590→          const gi = L12_GAP_NAMES.indexOf(c.name);
   591→          const age = ge - 0.35 - gi * 0.22;
   592→          const plaque = plaqueRefs.current[ci];
   593→          if (!plaque) return;
   594→          if (age <= 0) {
   595→            plaque.visible = false;
   596→            return;
   597→          }
   598→          if (!poppedRef.current[ci]) {
   599→            poppedRef.current[ci] = true;
   600→            flashRef.current[ci] = 1;
   601→            gapsRef.current += 1;
   602→            setGaps(gapsRef.current);
   603→          }
   604→          plaque.visible = true;
   605→          const p = age / 0.28;
   606→          const s = l12Pop(p);
   607→          plaque.scale.setScalar(s);
   608→          plaque.traverse((o) => {
   609→            const m = (o as THREE.Mesh).material as THREE.MeshBasicMaterial | undefined;
   610→            if (m && "opacity" in m) m.opacity = Math.min(1, p * 1.6);
   611→          });
   612→        });
   613→      }
   614→    }
   615→
   616→    // arrival / pop glow decay
   617→    flashRef.current = flashRef.current.map((f) => f * Math.exp(-dt * 2.2));
   618→    rowMats.current.forEach((mat, i) => {
   619→      if (mat) mat.emissiveIntensity = 0.07 + (flashRef.current[i] ?? 0) * 1.6;
   620→    });
   621→  });
   622→
   623→  const replay = () => {
   624→    startRef.current = clockRef.current;
   625→    arrivedRef.current = new Array(L12_FLIGHTS.length).fill(false);
   626→    arrivalAtRef.current = new Array(L12_FLIGHTS.length).fill(-1);
   627→    gapsStartRef.current = -1;
   628→    poppedRef.current = new Array(L12_CUSTOMERS.length).fill(false);
   629→    countRef.current = 0;
   630→    gapsRef.current = 0;
   631→    flashRef.current = new Array(L12_CUSTOMERS.length).fill(0);
   632→    plaqueRefs.current.forEach((p) => {
   633→      if (p) p.visible = false;
   634→    });
   635→    setMatched(0);
   636→    setGaps(0);
   637→  };
   638→
   639→  useEffect(() => {
   640→    if (replayRef) replayRef.current = replay;
   641→  });
   642→
   643→  const done = gaps >= L12_GAP_NAMES.length;
   644→  const allMatched = matched >= flights.length;
   645→
   646→  return (
   647→    <>
   648→      <ambientLight intensity={0.55} />
   649→      <directionalLight position={[4, 8, 6]} intensity={1.15} />
   650→      <fog attach="fog" args={["#09090b", 15, 32]} />
   651→
   652→      <group ref={groupRef}>
   653→        {/* ---- customers table (left, every row survives) ---- */}
   654→        <Text font={FONT} fontSize={0.22} color="#a1a1aa" position={[L12_CUST_X, 3.0, 0]} anchorX="center">
   655→          customers
   656→        </Text>
   657→        <Text font={FONT} fontSize={0.13} color="#52525b" position={[L12_CUST_X, 2.78, 0]} anchorX="center">
   658→          16 rows · all preserved
   659→        </Text>
   660→        {L12_CUSTOMERS.map((c, ci) => (
   661→          <group key={c.id} position={[L12_CUST_X, l12RowY(ci), 0]}>
   662→            <RoundedBox args={[2.9, 0.36, 0.3]} radius={0.05} smoothness={3}>
   663→              <meshStandardMaterial
   664→                ref={(m) => {
   665→                  rowMats.current[ci] = m as THREE.MeshStandardMaterial | null;
   666→                }}
   667→                color="#27272a"
   668→                emissive={c.shipped === 0 ? L12_AMBER : L12_TEAL}
   669→                emissiveIntensity={0.07}
   670→                roughness={0.55}
   671→              />
   672→            </RoundedBox>
   673→            <RoundedBox args={[0.3, 0.28, 0.28]} radius={0.04} smoothness={2} position={[-1.16, 0, 0.05]}>
   674→              <meshStandardMaterial color="#3f3f46" roughness={0.5} />
   675→            </RoundedBox>
   676→            <Text font={FONT} fontSize={0.13} color="#d4d4d8" position={[-1.16, 0, 0.22]} anchorX="center" anchorY="middle">
   677→              {String(c.id)}
   678→            </Text>
   679→            <Text font={FONT} fontSize={0.145} color="#e4e4e7" position={[-0.88, 0, 0.2]} anchorX="left" anchorY="middle">
   680→              {c.name}
   681→            </Text>
   682→            <Text
   683→              font={FONT}
   684→              fontSize={0.125}
   685→              color={c.shipped === 0 ? "#fbbf24" : "#5eead4"}
   686→              position={[1.32, 0, 0.2]}
   687→              anchorX="right"
   688→              anchorY="middle"
   689→            >
   690→              {`${c.shipped} shipped`}
   691→            </Text>
   692→            {/* NULL plaque — pops in phase B for the gap customers */}
   693→            <group
   694→              ref={(g) => {
   695→                plaqueRefs.current[ci] = g;
   696→              }}
   697→              visible={false}
   698→              position={[2.32, 0, 0.12]}
   699→            >
   700→              <RoundedBox args={[0.74, 0.28, 0.06]} radius={0.04} smoothness={2}>
   701→                <meshBasicMaterial color={L12_AMBER} transparent opacity={0} />
   702→              </RoundedBox>
   703→              <Text font={FONT} fontSize={0.12} color="#78350f" position={[0, 0, 0.05]} anchorX="center" anchorY="middle">
   704→                NULL
   705→              </Text>
   706→            </group>
   707→          </group>

   709→

   711→        <Text font={FONT} fontSize={0.22} color="#a1a1aa" position={[L12_STACK_X, 3.9, 0]} anchorX="center">
   712→          orders · shipped

   714→        <Text font={FONT} fontSize={0.13} color="#52525b" position={[L12_STACK_X, 3.66, 0]} anchorX="center">

   719→            key={i}
   720→            args={[0.9, 0.11, 0.11]}
   721→            radius={0.03}
   722→            smoothness={2}
   723→            position={[L12_STACK_X, L12_STACK_BASE + i * L12_BAR_PITCH, 0]}
   724→          >
   725→            <meshStandardMaterial color={L12_TEAL} emissive={L12_TEAL} emissiveIntensity={0.55} roughness={0.45} />

   727→        ))}
   728→        <Text font={FONT} fontSize={0.125} color="#71717a" position={[L12_STACK_X, L12_STACK_BASE - 0.24, 0]} anchorX="center">
   729→          + 73 not shipped (52 delivered · 14 pending · 7 cancelled)

   731→
   732→        {/* ---- result slabs ---- */}
   733→        <RoundedBox args={[4.5, 0.5, 0.36]} radius={0.08} smoothness={3} position={[-2.5, -4.7, 0]}>

   838→function L12LeftJoinSpace() {
   839→  const wrapRef = useRef<HTMLDivElement>(null);
   840→  const replayRef = useRef<(() => void) | null>(null);
   841→  const lastAutoRef = useRef(0);
   842→
   843→  // Auto-replay on scroll-into-view, same contract as the INNER JOIN scene.
   844→  useEffect(() => {
   845→    const el = wrapRef.current;
   846→    if (!el || typeof IntersectionObserver === "undefined") return;
   847→    const io = new IntersectionObserver(
   848→      (entries) => {
   849→        for (const en of entries) {
   850→          if (en.isIntersecting && en.intersectionRatio >= 0.55) {
   851→            const now = performance.now();
   852→            if (now - lastAutoRef.current > 7000) {
   853→              lastAutoRef.current = now;
   854→              replayRef.current?.();
   855→            }
   856→          }
   857→        }
   858→      },
   859→      { threshold: [0.55] },
   860→    );
   861→    io.observe(el);
   862→    return () => io.disconnect();
   863→  }, []);
   864→
   865→  return (
   866→    <div
   867→      ref={wrapRef}
   868→      className="relative h-[400px] w-full"
   869→      role="img"
   870→      aria-label="Interactive 3D scene of a LEFT JOIN. All 16 customer rows stand on the left; 30 teal order bars stack on the right. Each shipped order flies a glowing ribbon to its customer, then six amber NULL plaques pop beside the customers nothing arrived for — Bruno Silva, Chloe Dubois, Dmitri Volkov, Farid Nasser, Keiko Sato and Olivia Brown — because the shipped-status condition sits in ON. The result keeps 36 rows: 30 matches plus 6 NULL-filled gaps; moving the condition into WHERE would collapse the result to 30. Drag to orbit."
   871→    >
   872→      <Canvas
   873→        dpr={[1, 1.75]}
   874→        gl={{ antialias: true, alpha: true }}
   875→        camera={{ position: [0, 0.4, 14.4], fov: 38 }}
   876→      >
   877→        <SuspenseBoundary>
   878→          <LeftJoinScene replayRef={replayRef} />
   879→        </SuspenseBoundary>
   880→      </Canvas>
   881→    </div>
   882→  );
   883→}
