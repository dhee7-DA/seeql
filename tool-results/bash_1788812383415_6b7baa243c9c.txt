    99→function JsonView({ plan }: { plan: QueryPlan }) {
   100→  const [copied, setCopied] = useState(false);
   101→  const json = JSON.stringify({ ...plan, tree: undefined }, null, 2);
   102→
   103→  const copy = async () => {
   104→    try {
   105→      await navigator.clipboard.writeText(json);
   106→      setCopied(true);
   107→      setTimeout(() => setCopied(false), 1600);
   108→    } catch {
   109→      // clipboard unavailable — ignore
   110→    }
   111→  };
   112→
   113→  return (
   114→    <div className="space-y-3">
   115→      <div className="flex items-center justify-between">
   116→        <p className="text-xs text-zinc-400">
   117→          The structured plan the visualizer derived from your query.
   118→        </p>
   119→        <Button
   120→          variant="outline"
   121→          size="sm"
   122→          onClick={copy}
   123→          className="h-7 gap-1.5 border-zinc-700 text-xs text-zinc-300 hover:bg-zinc-800"
   124→        >
   125→          {copied ? <Check className="h-3.5 w-3.5 text-emerald-400" /> : <Copy className="h-3.5 w-3.5" />}
   126→          {copied ? "Copied" : "Copy JSON"}
   127→        </Button>
   128→      </div>
   129→      <pre className="thin-scrollbar max-h-[520px] overflow-auto rounded-xl border border-zinc-800 bg-zinc-900/70 p-4 font-mono text-xs leading-relaxed text-zinc-300">
   130→        {json}
   131→      </pre>
   132→    </div>
   133→  );
   134→}

   138→function PlaygroundView({
   139→  go,
   140→  sharedSql,
   141→  sharedDialect,
   142→}: {
   143→  go: (m: Mode) => void;
   144→  /** SQL coming from a #playground?q=… share link — boots the editor with it. */
   145→  sharedSql?: string | null;
   146→  sharedDialect?: DialectName;
   147→}) {
   148→  const [sql, setSql] = useState(sharedSql ?? DEFAULT_SAMPLE.sql);
   149→  const [dialect, setDialect] = useState<DialectName>(sharedDialect ?? DEFAULT_SAMPLE.dialect);
   150→  const [sampleId, setSampleId] = useState<string>(sharedSql ? "" : DEFAULT_SAMPLE.id);
   151→  const [copiedLink, setCopiedLink] = useState(false);
   152→  const [plan, setPlan] = useState<ParseResult | null>(null);
   153→  const [exec, setExec] = useState<ExecResponse | null>(null);
   154→  const [running, setRunning] = useState(false);
   155→  const istory, setHistory] = useState<string[]>([]);
   156→  const [resetting, setResetting] = useState(false);
   157→  const [tableCounts, setTableCounts] = useState<Record<string, number> | null>(null);
   158→
   159→  // live table row counts — powers the FROM dot-field in the query flow strip
   160→  useEffect(() => {
   161→    fetch("/api/stats")
   162→      .then((r) => r.json())
   163→      .then((j) => {
   164→        if (j?.ok) setTableCounts(j.counts as Record<string, number>);
   165→      })
   166→      .catch(() => {
   167→        /* flow strip simply shows no dot-field */
   168→      });
   169→  }, []);
   170→
   171→  const loadHistory = useCallback(() => {
   172→    try {
   173→      const raw = window.localStorage.getItem(HISTORY_KEY);
   174→      if (raw) setHistory(JSON.parse(raw) as string[]);
   175→    } catch {
   176→      // ignore malformed history
   177→    }
   178→  }, []);
   179→
   180→  const pushHistory = useCallback((text: string) => {
   181→    setHistory((prev) => {
   182→      const next = [text, ...prev.filter((h) => h !== text)].slice(0, HISTORY_MAX);
   183→      try {
   184→        window.localStorage.setItem(HISTORY_KEY, JSON.stringify(next));
   185→      } catch {
   186→        // storage unavailable
   187→      }
   188→      return next;
   189→    });
   190→  }, []);
   191→
   192→  /** Parse + execute in parallel; both panels update independently. */
   193→  const runAll = useCallback(
   194→    async (sqlText: string, dia: DialectName) => {
   195→      const trimmed = sqlText.trim();
   196→      if (!trimmed) return;
   197→      setRunning(true);
   198→      try {
   199→        const [parseRes, execRes] = await Promise.all([
   200→          fetch("/api/parse", {
   201→            method: "POST",
   202→            headers: { "Content-Type": "application/json" },
   203→            body: JSON.stringify({ sql: trimmed, dialect: dia }),
   204→          }).then((r) => r.json() as Promise<ParseResult>),
   205→          fetch("/api/query", {
   206→            method: "POST",
   207→            headers: { "Content-Type": "application/json" },
   208→            body: JSON.stringify({ sql: trimmed }),
   209→          }).then((r) => r.json() as Promise<ExecResponse>),
   210→        ]);
   211→        setPlan(parseRes);
   212→        setExec(execRes);
   213→        pushHistory(trimmed);
   214→      } catch {
   215→        setPlan((p) => p ?? { ok: false, error: "Could not reach the server. Please try again." });
   216→        setExec({ ok: false, error: "Could not reach the server. Please try again." });
   217→      } finally {
   218→        setRunning(false);
   219→      }
   220→    },
   221→    [pushHistory],
   222→  );
   223→
   224→  const parseOnly = useCallback(async (sqlText: string, dia: DialectName) => {
   225→    try {
   226→      const res = await fetch("/api/parse", {
   227→        method: "POST",
   228→        headers: { "Content-Type": "application/json" },
   229→        body: JSON.stringify({ sql: sqlText, dialect: dia }),
   230→      });
   231→      setPlan((await res.json()) as ParseResult);
   232→    } catch {
   233→      setPlan({ ok: false, error: "Could not reach the parser. Please try again." });
   234→    }
   235→  }, []);
   236→
   237→  // run the initial query once on load — a #playground?q=… share link boots
   238→  // its own SQL, otherwise the default sample runs so the page shows live
   239→  // diagrams + data immediately
   240→  const bootSql = sharedSql ?? DEFAULT_SAMPLE.sql;
   241→  const bootDialect = sharedDialect ?? DEFAULT_SAMPLE.dialect;
   242→  useEffect(() => {
   243→    loadHistory();
   244→    runAll(bootSql, bootDialect);
   245→  }, [loadHistory, runAll, bootSql, bootDialect]);
   246→
   247→  /** Copy a link that reopens the playground with the current query loaded. */
   248→  const shareLink = async () => {
   249→    const params = new URLSearchParams({ q: sql });
   250→    if (dialect !== DEFAULT_SAMPLE.dialect) params.set("d", dialect);
   251→    const url = `${window.location.origin}${window.location.pathname}#playground?${params.toString()}`;
   252→    let ok = false;
   253→    try {
   254→      await navigator.clipboard.writeText(url);
   255→      ok = true;
   256→    } catch {
   257→      // clipboard API unavailable/denied — legacy fallback for restricted contexts
   258→      try {
   259→        const ta = document.createElement("textarea");
   260→        ta.value = url;
   261→        ta.style.position = "fixed";
   262→        ta.style.opacity = "0";
   263→        document.body.appendChild(ta);
   264→        ta.select();
   265→        ok = document.execCommand("copy");
   266→        document.body.removeChild(ta);
   267→      } catch {
   268→        ok = false;
   269→      }
   270→    }
   271→    if (ok) {
   272→      setCopiedLink(true);
   273→      setTimeout(() => setCopiedLink(false), 1600);
   274→    }
   275→  };
   276→
   277→  const onSampleChange = (id: string) => {
   278→    const sample = SAMPLE_QUERIES.find((s) => s.id === id);
   279→    if (!sample) return;
   280→    setSampleId(id);
   281→    setSql(sample.sql);
   282→    setDialect(sample.dialect);
   283→    runAll(sample.sql, sample.dialect);
   284→  };
   285→
   286→  const onDialectChange = (d: string) => {
   287→    const next = d as DialectName;
   288→    setDialect(next);
   289→    parseOnly(sql, next);
   290→  };
   291→
   292→  const onHistoryPick = (text: string) => {
   293→    setSampleId("");
   294→    setSql(text);
   295→    runAll(text, dialect);
   296→  };
   297→
   298→  const onSchemaPickTable = (table: string) => {
   299→    const text = `SELECT *\nFROM ${table}\nLIMIT 5;`;
   300→    setSampleId("");
   301→    setSql(text);
   302→    runAll(text, dialect);
   303→  };
   304→
   305→  const resetDatabase = async () => {
   306→    if (!window.confirm("Restore the sample database to its original data?")) return;
   307→    setResetting(true);
   308→    try {
   309→      await fetch("/api/reset", { method: "POST" });
   310→      // re-run current query so the user sees fresh results
   311→      await runAll(sql, dialect);
   312→    } catch {
   313→      // keep silent — the re-run will surface errors if any
   314→    } finally {
   315→      setResetting(false);
   316→    }
   317→  };
   318→
   319→  const good = plan !== null && plan.ok;
   320→  const execOk = exec !== null && exec.ok;
   321→  const planSteps = execOk ? (exec as ExecResult).plan?.length ?? 0 : 0;
   322→  const isWrite = execOk && (exec as ExecResult).changes !== null;
   323→
   324→  return (
   325→    <div className="flex min-h-screen flex-col bg-zinc-950 text-zinc-100">
   326→      {/* header */}
   327→      <header className="sticky top-0 z-40 border-b border-zinc-800/80 bg-zinc-950/85 backdrop-blur">
   328→        <div className="mx-auto flex w-full max-w-7xl flex-wrap items-center gap-x-4 gap-y-3 px-4 py-3">
   329→          <div className="flex items-center gap-3">
   330→            <button onClick={() => go("landing")} aria-label="SeeQL — back to home" className="transition-opacity hover:opacity-80">
   331→              <LogoFull />
   332→            </button>
   333→            <div className="hidden border-l border-zinc-800 pl-3 sm:block">
   334→              <p className="text-xs text-zinc-500">
   335→                Write a query — see it parsed, executed and explained
   336→              </p>
   337→            </div>
   338→          </div>
   339→
   340→          <div className="ml-auto flex flex-wrap items-center gap-2">
   341→            <ThemeSwitcher />
   342→            <button
   343→              onClick={() => go("learn")}
   344→              className="hidden items-center gap-1.5 rounded-md border border-zinc-800 px-3.5 py-2 text-sm text-zinc-300 transition-colors hover:bg-zinc-900 sm:flex"
   345→            >
   346→              <GraduationCap className="h-4 w-4 text-emerald-400" />
   347→              Learn
   348→            </button>
   349→            <Select value={sampleId} onValueChange={onSampleChange}>
   350→              <SelectTrigger
   351→                size="sm"
   352→                className="w-[190px] border-zinc-700 bg-zinc-900 text-xs sm:w-[220px]"
   353→                aria-label="Load a sample query"
   354→              >
   355→                <SelectValue placeholder="Sample queries" />
   356→              </SelectTrigger>
   357→              <SelectContent>
   358→                {SAMPLE_QUERIES.map((s) => (
   359→                  <SelectItem key={s.id} value={s.id} className="text-xs">
   360→                    <span className="font-medium">{s.name}</span>

   363→                ))}
   364→              </SelectContent>
   365→            </Select>
   366→
   367→            <Select value={dialect} onValueChange={onDialectChange}>
   368→              <SelectTrigger

   373→                <SelectValue />

   384→            <Button

   386→              size="sm"

   392→                <Check className="h-4 w-4 text-emerald-400" />

   394→                <Link2 className="h-4 w-4" />
   395→              )}
   396→              {copiedLink ? "Copied" : "Share"}
   397→            </Button>
   398→
   399→            <Button
   400→              onClick={() => runAll(sql, dialect)}
   401→              disabled={running}
   402→              className="h-8 gap-1.5 bg-emerald-500 font-medium text-zinc-950 hover:bg-emerald-400"
   403→            >

   406→              ) : (
   407→                <Play className="h-4 w-4" />
   408→              )}
   409→              Run
   410→            </Button>
   411→          </div>
   412→        </div>
   413→      </header>
   414→
   415→      {/* main */}
   416→      <main className="mx-auto w-full max-w-7xl flex-1 px-4 py-5">
   417→        <div className="grid items-start gap-5 lg:grid-cols-inmax(360px,2fr)_minmax(0,3fr)]">
   418→          {/* editor panel */}
   419→          <section aria-label="SQL editor" className="flex flex-col gap-2">
   420→            <div className="flex items-center justify-between px-1">
   421→              <h2 className="text-xs font-semibold tracking-widest text-zinc-400 uppercase">
   422→                Query
   423→              </h2>
   424→              <div className="flex items-center gap-1">
   425→                {history.length > 0 && (
   426→                  <Select onValueChange={onHistoryPick}>
   427→                    <SelectTrigger
   428→                      size="sm"
   429→                      className="h-7 gap-1 border-none bg-transparent px-2 text-xs text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200 focus:ring-0"
   430→                      aria-label="Query history"
   431→                    >
   432→                      <span className="flex items-center gap-1.5">
   433→                        <History className="h-3.5 w-3.5" />

   469→                Ctrl / ⌘

   471→              +{" "}

   532→                {plan.statementType !== "SELECT" && (

   756→/**
   757→ * App shell — single exposed route, three views:
   758→ *   /            → landing
   759→ *   /#playground → SQL playground (the original visualizer)
   760→ *   /#learn      → the learning app
   761→ * Hash syncs with browser history so back/forward work.
   762→ */
   763→function subscribeHash(cb: () => void) {
   764→  window.addEventListener("hashchange", cb);
   765→  return () => window.removeEventListener("hashchange", cb);
   766→}
   767→
   768→export default function Home() {
   769→  // The hash IS the router — subscribe to it as an external system.
   770→  // useSyncExternalStore renders the server snapshot during hydration
   771→  // (landing) and re-renders with the real hash right after — no
   772→  // hydration mismatch, no setState-in-effect.
   773→  const hash = useSyncExternalStore(
   774→    subscribeHash,
   775→    () => window.location.hash,
   776→    () => "",
   777→  );
   778→  // The route is the part before "?" — a shared query rides along as
   779→  // #playground?q=<encoded sql>&d=<dialect> and never breaks deep links.
   780→  const qMark = hash.indexOf("?");
   781→  const route = qMark === -1 ? hash : hash.slice(0, qMark);
   782→  const mode: Mode =
   783→    route === "#playground" ? "playground" : route === "#learn" ? "learn" : "landing";
   784→
   785→  let sharedSql: string | null = null;
   786→  let sharedDialect: DialectName | undefined;
   787→  if (mode === "playground" && qMark !== -1) {
   788→    const params = new URLSearchParams(hash.slice(qMark + 1));
   789→    const q = params.get("q");
   790→    if (q && q.trim()) sharedSql = q;
   791→    const d = params.get("d");
   792→    if (d && (DIALECTS as readonly string[]).includes(d)) sharedDialect = d as DialectName;
   793→  }
   794→
   795→  const go = useCallback((m: Mode) => {
   796→    window.location.hash = m === "landing" ? "" : `#${m}`;
   797→    window.scrollTo({ top: 0 });
   798→  }, []);
   799→
   800→  if (mode === "landing") {
   801→    return <Landing onStart={() => go("learn")} onPlayground={() => go("playground")} />;
   802→  }
   803→  if (mode === "learn") {
   804→    return (
   805→      <div className="flex min-h-screen flex-col bg-zinc-950 text-zinc-100">
   806→        <header className="sticky top-0 z-40 border-b border-zinc-800/80 bg-zinc-950/85 backdrop-blur">
   807→          <div className="mx-auto flex w-full max-w-7xl items-center justify-between gap-3 px-4 py-3 sm:px-6">
   808→            <button onClick={() => go("landing")} aria-label="SeeQL — back to home" className="transition-opacity hover:opacity-80">
   809→              <LogoFull />
   810→            </button>
   811→            <div className="flex items-center gap-2">
   812→              <ThemeSwitcher />
   813→              <button
   814→                onClick={() => go("playground")}
   815→                className="hidden rounded-md border border-zinc-800 px-3.5 py-2 text-sm text-zinc-300 transition-colors hover:bg-zinc-900 sm:block"
   816→              >
   817→                Playground
   818→              </button>
   819→            </div>
   820→          </div>
   821→        </header>
   822→        <LearnApp />
   823→        <footer className="mt-auto border-t border-zinc-800/80">
   824→          <div className="mx-auto flex w-full max-w-7xl items-center justify-between px-4 py-3 text-[11px] text-zinc-600 sm:px-6">
   825→            <span>SeeQL Learn · progress saved on this device</span>
   826→            <span className="font-mono">read · watch · prove</span>
   827→          </div>
   828→        </footer>
   829→      </div>
   830→    );
   831→  }
   832→  return <PlaygroundView key={sharedSql ?? ""} go={go} sharedSql={sharedSql} sharedDialect={sharedDialect} />;
   833→}
