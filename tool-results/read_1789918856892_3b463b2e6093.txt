   109→function JsonView({ plan }: { plan: QueryPlan }) {
   110→  const [copied, setCopied] = useState(false);
   111→  const json = JSON.stringify({ ...plan, tree: undefined }, null, 2);
   112→
   113→  const copy = async () => {
   114→    try {
   115→      await navigator.clipboard.writeText(json);
   116→      setCopied(true);
   117→      setTimeout(() => setCopied(false), 1600);
   118→    } catch {
   119→      // clipboard unavailable — ignore
   120→    }
   121→  };
   122→
   123→  return (
   124→    <div className="space-y-3">
   125→      <div className="flex items-center justify-between">
   126→        <p className="text-xs text-zinc-400">
   127→          The structured plan the visualizer derived from your query.
   128→        </p>
   129→        <Button
   130→          variant="outline"
   131→          size="sm"
   132→          onClick={copy}
   133→          className="h-7 gap-1.5 border-zinc-700 text-xs text-zinc-300 hover:bg-zinc-800"
   134→        >
   135→          {copied ? <Check className="h-3.5 w-3.5 text-emerald-400" /> : <Copy className="h-3.5 w-3.5" />}
   136→          {copied ? "Copied" : "Copy JSON"}
   137→        </Button>
   138→      </div>
   139→      <pre className="thin-scrollbar max-h-[520px] overflow-auto rounded-xl border border-zinc-800 bg-zinc-900/70 p-4 font-mono text-xs leading-relaxed text-zinc-300">
   140→        {json}
   141→      </pre>
   142→    </div>
   143→  );
   144→}

   148→function PlaygroundView({
   149→  go,
   150→  sharedSql,
   151→  sharedDialect,
   152→}: {
   153→  go: (m: Mode) => void;
   154→  /** SQL coming from a #playground?q=… share link — boots the editor with it. */
   155→  sharedSql?: string | null;
   156→  sharedDialect?: DialectName;
   157→}) {
   158→  const [sql, setSql] = useState(sharedSql ?? DEFAULT_SAMPLE.sql);
   159→  const [dialect, setDialect] = useState<DialectName>(sharedDialect ?? DEFAULT_SAMPLE.dialect);
   160→  const [sampleId, setSampleId] = useState<string>(sharedSql ? "" : DEFAULT_SAMPLE.id);
   161→  const [copiedLink, setCopiedLink] = useState(false);
   162→  const [plan, setPlan] = useState<ParseResult | null>(null);
   163→  const [exec, setExec] = useState<ExecResponse | null>(null);
   164→  const [running, setRunning] = useState(false);
   165→  const [history, setHistory] = useState<string[]>([]);
   166→  const [resetting, setResetting] = useState(false);
   167→  const [tableCounts, setTableCounts] = useState<Record<string, number> | null>(null);
   168→
   169→  // live layer — the picture follows the text while it is being written
   170→  const [livePlan, setLivePlan] = useState<ParseResult | null>(null);
   171→  const [liveSql, setLiveSql] = useState("");
   172→  const [liveError, setLiveError] = useState<ParseError | null>(null);
   173→  const [ranSql, setRanSql] = useState("");
   174→  const [caret, setCaret] = useState(0);
   175→  const [hoverKind, setHoverKind] = useState<ClauseKind | null>(null);
   176→  const [pinnedKind, setPinnedKind] = useState<ClauseKind | null>(null);
   177→  const [dataVersion, setDataVersion] = useState(0); // bumped on DB reset
   178→  // step inspector — which funnel stage's rows are open (null = closed)
   179→  const [inspectKind, setInspectKind] = useState<ClauseKind | "out" | null>(null);
   180→  // shared XP ledger — the Missions badge counts earned missions live
   181→  const missionXp = useProgress((s) => s.xpEvents);
   182→
   183→  // live table row counts — powers the FROM dot-field in the query flow strip
   184→  useEffect(() => {
   185→    fetch("/api/stats")
   186→      .then((r) => r.json())
   187→      .then((j) => {
   188→        if (j?.ok) setTableCounts(j.counts as Record<string, number>);
   189→      })
   190→      .catch(() => {
   191→        /* flow strip simply shows no dot-field */
   192→      });
   193→  }, []);
   194→
   195→  // live parse — debounced so the flow strip, pipeline and stats redraw while
   196→  // the query is being typed, without firing on every keystroke
   197→  useEffect(() => {
   198→    const timer = window.setTimeout(() => {
   199→      const trimmed = sql.trim();
   200→      if (!trimmed) {
   201→        setLivePlan(null);
   202→        setLiveSql("");
   203→        setLiveError(null);
   204→        return;
   205→      }
   206→      // raw text goes out so error offsets align with the editor content
   207→      fetch("/api/parse", {
   208→        method: "POST",
   209→        headers: { "Content-Type": "application/json" },
   210→        body: JSON.stringify({ sql, dialect }),
   211→      })
   212→        .then((r) => r.json() as Promise<ParseResult>)
   213→        .then((res) => {
   214→          setLivePlan(res);
   215→          setLiveSql(sql);
   216→          setLiveError(res.ok ? null : res);
   217→        })
   218→        .catch(() => {
   219→          /* transient network hiccup — keep the previous live state */
   220→        });
   221→    }, 400);
   222→    return () => window.clearTimeout(timer);
   223→  }, [sql, dialect]);
   224→
   225→  const loadHistory = useCallback(() => {
   226→    try {
   227→      const raw = window.localStorage.getItem(HISTORY_KEY);
   228→      if (raw) setHistory(JSON.parse(raw) as string[]);
   229→    } catch {
   230→      // ignore malformed history
   231→    }
   232→  }, []);
   233→
   234→  const pushHistory = useCallback((text: string) => {
   235→    setHistory((prev) => {
   236→      const next = [text, ...prev.filter((h) => h !== text)].slice(0, HISTORY_MAX);
   237→      try {
   238→        window.localStorage.setItem(HISTORY_KEY, JSON.stringify(next));
   239→      } catch {
   240→        // storage unavailable
   241→      }
   242→      return next;
   243→    });
   244→  }, []);
   245→
   246→  /** Parse + execute in parallel; both panels update independently. */
   247→  const runAll = useCallback(
   248→    async (sqlText: string, dia: DialectName) => {
   249→      const trimmed = sqlText.trim();
   250→      if (!trimmed) return;
   251→      setRunning(true);
   252→      try {
   253→        const [parseRes, execRes] = await Promise.all([
   254→          // raw text → parser shifts error offsets back into the editor text
   255→          fetch("/api/parse", {
   256→            method: "POST",
   257→            headers: { "Content-Type": "application/json" },
   258→            body: JSON.stringify({ sql: sqlText, dialect: dia }),
   259→          }).then((r) => r.json() as Promise<ParseResult>),
   260→          fetch("/api/query", {
   261→            method: "POST",
   262→            headers: { "Content-Type": "application/json" },
   263→            body: JSON.stringify({ sql: trimmed }),
   264→          }).then((r) => r.json() as Promise<ExecResponse>),
   265→        ]);
   266→        setPlan(parseRes);
   267→        setRanSql(sqlText);
   268→        setExec(execRes);
   269→        setLivePlan(parseRes);
   270→        setLiveSql(sqlText);
   271→        setLiveError(parseRes.ok ? null : parseRes);
   272→        pushHistory(trimmed);
   273→      } catch {
   274→        setPlan((p) => p ?? { ok: false, error: "Could not reach the server. Please try again." });
   275→        setExec({ ok: false, error: "Could not reach the server. Please try again." });
   276→      } finally {
   277→        setRunning(false);
   278→      }
   279→    },
   280→    [pushHistory],
   281→  );
   282→
   283→  const parseOnly = useCallback(async (sqlText: string, dia: DialectName) => {
   284→    try {
   285→      const res = await fetch("/api/parse", {
   286→        method: "POST",
   287→        headers: { "Content-Type": "application/json" },
   288→        body: JSON.stringify({ sql: sqlText, dialect: dia }),
   289→      });
   290→      const parsed = (await res.json()) as ParseResult;
   291→      setLivePlan(parsed);
   292→      setLiveSql(sqlText);
   293→      setLiveError(parsed.ok ? null : parsed);
   294→    } catch {
   295→      setPlan({ ok: false, error: "Could not reach the parser. Please try again." });
   296→    }
   297→  }, []);
   298→
   299→  // run the initial query once on load — a #playground?q=… share link boots
   300→  // its own SQL, otherwise the default sample runs so the page shows live
   301→  // diagrams + data immediately
   302→  const bootSql = sharedSql ?? DEFAULT_SAMPLE.sql;
   303→  const bootDialect = sharedDialect ?? DEFAULT_SAMPLE.dialect;
   304→  useEffect(() => {
   305→    loadHistory();
   306→    runAll(bootSql, bootDialect);
   307→  }, [loadHistory, runAll, bootSql, bootDialect]);
   308→
   309→  /** Copy a link that reopens the playground with the current query loaded. */
   310→  const shareLink = async () => {
   311→    const params = new URLSearchParams({ q: sql });
   312→    if (dialect !== DEFAULT_SAMPLE.dialect) params.set("d", dialect);
   313→    const url = `${window.location.origin}${window.location.pathname}#playground?${params.toString()}`;
   314→    let ok = false;
   315→    try {
   316→      await navigator.clipboard.writeText(url);
   317→      ok = true;
   318→    } catch {
   319→      // clipboard API unavailable/denied — legacy fallback for restricted contexts
   320→      try {
   321→        const ta = document.createElement("textarea");
   322→        ta.value = url;
   323→        ta.style.position = "fixed";
   324→        ta.style.opacity = "0";
   325→        document.body.appendChild(ta);
   326→        ta.select();
   327→        ok = document.execCommand("copy");
   328→        document.body.removeChild(ta);
   329→      } catch {
   330→        ok = false;
   331→      }
   332→    }
   333→    if (ok) {
   334→      setCopiedLink(true);
   335→      setTimeout(() => setCopiedLink(false), 1600);
   336→    }
   337→  };
   338→
   339→  const onSampleChange = (id: string) => {
   340→    const sample = SAMPLE_QUERIES.find((s) => s.id === id);
   341→    if (!sample) return;
   342→    setSampleId(id);
   343→    setSql(sample.sql);
   344→    setDialect(sample.dialect);
   345→    runAll(sample.sql, sample.dialect);
   346→  };
   347→
   348→  const onDialectChange = (d: string) => {
   349→    const next = d as DialectName;
   350→    setDialect(next);
   351→    parseOnly(sql, next);
   352→  };
   353→
   354→  const onHistoryPick = (text: string) => {
   355→    setSampleId("");
   356→    setSql(text);
   357→    runAll(text, dialect);
   358→  };
   359→
   360→  const onSchemaPickTable = (table: string) => {
   361→    const text = `SELECT *\nFROM ${table}\nLIMIT 5;`;
   362→    setSampleId("");
   363→    setSql(text);
   364→    runAll(text, dialect);
   365→  };
   366→
   367→  const resetDatabase = async () => {
   368→    if (!window.confirm("Restore the sample database to its original data?")) return;
   369→    setResetting(true);
   370→    try {
   371→      await fetch("/api/reset", { method: "POST" });
   372→      // the data under the same SQL changed — bump so the row funnel and the
   373→      // FROM dot-field re-count against the fresh database
   374→      setDataVersion((v) => v + 1);
   375→      setTableCounts(null);
   376→      fetch("/api/stats")
   377→        .then((r) => r.json())
   378→        .then((j) => {
   379→          if (j?.ok) setTableCounts(j.counts as Record<string, number>);
   380→        })
   381→        .catch(() => {
   382→          /* dot-field simply skips counts */
   383→        });
   384→      // re-run current query so the user sees fresh results
   385→      await runAll(sql, dialect);
   386→    } catch {
   387→      // keep silent — the re-run will surface errors if any
   388→    } finally {
   389→      setResetting(false);
   390→    }
   391→  };
   392→
   393→  const good = plan !== null && plan.ok;
   394→  const execOk = exec !== null && exec.ok;
   395→  const planSteps = execOk ? (exec as ExecResult).plan?.length ?? 0 : 0;
   396→  const isWrite = execOk && (exec as ExecResult).changes !== null;
   397→
   398→  // ---- text ↔ diagram linking ------------------------------------------
   399→  // The live (debounced) plan drives the picture; the last run's plan is the
   400→  // fallback while the current text is unparseable.

   402→  const viewPlanSql = livePlan?.ok ? liveSql : plan?.ok ? ranSql : "";
   403→  const clauseInfo = useMemo(
   404→    () => (viewPlanSql ? findClauses(viewPlanSql) : null),
   405→    [viewPlanSql],
   406→  );

   408→    kind && clauseInfo?.supported ? spanOf(clauseInfo.spans, kind) : null;
   409→  const hoverSpan = spanFor(hoverKind);
   410→  const pinnedSpan = spanFor(pinnedKind);
   411→  const caretKind = clauseInfo ? clauseAtOffset(clauseInfo.spans, caret) : null;
   412→  const activeKinds: ClauseKind[] | null = pinnedKind
   413→    ? [pinnedKind]
   414→    : caretKind
   415→      ? [caretKind]
   416→      : null;
   417→  const errorRange =
   418→    liveError && liveSql === sql ? liveError.location ?? null : null;
   419→  // editor marks only make sense while the editor text matches the diagram's
   420→  // text snapshot — otherwise offsets have drifted and marks would misplace
   421→  const editorInSync = viewPlanSql === sql;
   422→  const togglePinned = useCallback((kind: ClauseKind) => {
   423→    setPinnedKind((p) => (p === kind ? null : kind));
   424→    setHoverKind(null);
   425→  }, []);
   426→
   427→  return (
   428→    <div className="flex min-h-screen flex-col bg-zinc-950 text-zinc-100">
   429→      {/* header */}
   430→      <header className="sticky top-0 z-40 border-b border-zinc-800/80 bg-zinc-950/85 backdrop-blur">
   431→        <div className="mx-auto flex w-full max-w-7xl flex-wrap items-center gap-x-4 gap-y-3 px-4 py-3">
   432→          <div className="flex items-center gap-3">
   433→            <button onClick={() => go("landing")} aria-label="SeeQL — back to home" className="transition-opacity hover:opacity-80">
   434→              <LogoFull />
   435→            </button>
   436→            <div className="hidden border-l border-zinc-800 pl-3 sm:block">
   437→              <p className="text-xs text-zinc-500">
   438→                Write a query — see it parsed, executed and explained
   439→              </p>
   440→            </div>
   441→          </div>
   442→
   443→          <div className="ml-auto flex flex-wrap items-center gap-2">
   444→            <ThemeSwitcher />
   445→            <button
   446→              onClick={() => go("learn")}
   447→              className="hidden items-center gap-1.5 rounded-md border border-zinc-800 px-3.5 py-2 text-sm text-zinc-300 transition-colors hover:bg-zinc-900 sm:flex"
   448→            >
   449→              <GraduationCap className="h-4 w-4 text-emerald-400" />
   450→              Learn
   451→            </button>
   452→            <SampleGallery currentId={sampleId} onPick={onSampleChange} />
   453→
   454→            <Select value={dialect} onValueChange={onDialectChange}>
   455→              <SelectTrigger
   456→                size="sm"
   457→                className="w-[130px] border-zinc-700 bg-zinc-900 text-xs"
   458→                aria-label="SQL dialect"
   459→              >
   460→                <SelectValue />
   461→              </SelectTrigger>
   462→              <SelectContent>
   463→                {DIALECTS.map((d) => (
   464→                  <SelectItem key={d} value={d} className="font-mono text-xs">
   465→                    {d}
   466→                  </SelectItem>
   467→                ))}
   468→              </SelectContent>
   469→            </Select>
   470→
   471→            <Button
   472→              variant="ghost"
   473→              size="sm"
   474→              onClick={shareLink}
   475→              className="h-8 gap-1.5 px-2.5 text-xs text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200"
   476→              aria-label="Copy a link to this query"
   477→            >
   478→              {copiedLink ? (
   479→                <Check className="h-4 w-4 text-emerald-400" />
   480→              ) : (
   481→                <Link2 className="h-4 w-4" />
   482→              )}
   483→              {copiedLink ? "Copied" : "Share"}
   484→            </Button>
   485→
   486→            <Button
   487→              onClick={() => runAll(sql, dialect)}
   488→              disabled={running}
   489→              className="h-8 gap-1.5 bg-emerald-500 font-medium text-zinc-950 hover:bg-emerald-400"
   490→            >
   491→              {running ? (
   492→                <Loader2 className="h-4 w-4 animate-spin" />
   493→              ) : (
   494→                <Play className="h-4 w-4" />
   495→              )}
   496→              Run
   497→            </Button>
   498→          </div>
   499→        </div>
   500→      </header>
   501→
   502→      {/* main */}
   503→      <main className="mx-auto w-full max-w-7xl flex-1 px-4 py-5">
   504→        <div className="grid items-start gap-5 lg:grid-cols-[minmax(360px,2fr)_minmax(0,3fr)]">
   505→          {/* editor panel */}
   506→          <section aria-label="SQL editor" className="flex flex-col gap-2">
   507→            <div className="flex items-center justify-between px-1">
   508→              <h2 className="text-xs font-semibold tracking-widest text-zinc-400 uppercase">
   509→                Query
   510→              </h2>
   511→              <div className="flex items-center gap-1">
   512→                {history.length > 0 && (
   513→                  <Select onValueChange={onHistoryPick}>
   514→                    <SelectTrigger
   515→                      size="sm"
   516→                      className="h-7 gap-1 border-none bg-transparent px-2 text-xs text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200 focus:ring-0"
   517→                      aria-label="Query history"
   518→                    >
   519→                      <span className="flex items-center gap-1.5">
   520→                        <History className="h-3.5 w-3.5" />
   521→                        History

   523→                    </SelectTrigger>
   524→                    <SelectContent>
   525→                      {history.map((h, i) => (

   529→                      ))}
   530→                    </SelectContent>
   531→                  </Select>
   532→                )}
   533→                <Button
   534→                  variant="ghost"
   535→                  size="sm"
   536→                  onClick={() => setSql("")}
   537→                  className="h-7 gap-1 px-2 text-xs text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200"

   542→              </div>
