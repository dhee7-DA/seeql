    43→const MOD = COURSE_MODULES[0];
    44→
    45→function lessonDone(
    46→  progress: ReturnType<typeof useProgress.getState>,
    47→  lessonId: string,
    48→  mcqIds: string[],
    49→  problemIds: string[],
    50→): boolean {
    51→  if (manuallyCompleted(progress, lessonId)) return true;
    52→  const allMcq = mcqIds.length > 0 && mcqIds.every((id) => mcqPassed(progress, id));
    53→  const allProblems =
    54→    problemIds.length === 0 || problemIds.every((id) => problemSolved(progress, id));
    55→  return allMcq && allProblems;
    56→}

   522→/* ---------------- overview (module home) ---------------- */
   523→
   524→function Overview({
   525→  doneMap,
   526→  onOpen,
   527→}: {
   528→  doneMap: Record<string, boolean>;
   529→  onOpen: (ref: PracticeRef) => void;
   530→  onTryInPlayground: (sql: string) => void;
   531→}) {
