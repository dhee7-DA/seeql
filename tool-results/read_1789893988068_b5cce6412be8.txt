     3→// SeeQ — the SeeQL study buddy, motion engine v2.
     4→//
     5→// A soft emerald blob that lives in the corner: it breathes, blinks, its gaze
     6→// wanders (and follows the pointer), it reacts to mood, and every so often it
     7→// shares a short coaching tip. Poke it and it cheers you on.
     8→//
     9→// The motion techniques follow the measurements documented by the MIT project
    10→// "bloub" (https://github.com/jeremy-prt/bloub, © Jérémy Perret — technique
    11→// only, this character is our own):
    12→//
    13→// - The body is a PERFECT circle (radial deviation < 0.7% on the reference),
    14→//   sampled at 64 angles and drawn as one closed Catmull-Rom. Lumpy "organic"
    15→//   silhouettes read as deformed, not alive.
    16→// - At rest the body is still: breath is 0.5% of height, drift is sub-pixel.
    17→//   All the life comes from gaze drift and blinking.
    18→// - The eyes are capsules projected onto a rotating head sphere (yaw/pitch/
    19→//   roll). Position, compression and lean all fall out of the orthographic
    20→//   projection — eyes slide and squash near the sphere edge instead of
    21→//   translating flatly.
    22→// - Blinks come from a deterministic schedule (1.9–4.6 s apart, occasional
    23→//   double-blinks), close fast and reopen slower, and never fully close.
    24→//   Mood changes are masked by a forced blink, exactly like a cut.
    25→// - Every transition is a pure easeOutQuint — the body never overshoots.
    26→//   An interrupted morph freezes its composite and re-blends from it, so
    27→//   rapid mood changes can't jump.
    28→//
    29→// Honours prefers-reduced-motion (one static frame, no loop). Collapsible
    30→// (persisted); speech pauses while collapsed. React never re-renders per
    31→// frame: geometry is painted straight onto SVG attributes.

    38→// ---- small math --------------------------------------------------------
    39→
    40→const TAU = Math.PI * 2;
    41→
    42→const clamp = (v: number, lo = 0, hi = 1) => (v < lo ? lo : v > hi ? hi : v);
    43→const lerp = (a: number, b: number, t: number) => a + (b - a) * t;
    44→/** Short rounding: halves the weight of 60 fps path strings. */
    45→const r2 = (v: number) => Math.round(v * 100) / 100;
    46→
    47→/** Measured on the reference: transitions are pure ease-out, no overshoot. */
    48→const easeOutQuint = (t: number) => 1 - (1 - t) ** 5;
    49→
    50→/** 1D periodic noise, seamless over `period` seconds — harmonics are prime
    51→ * relative to each other, so the drift never visibly repeats. */
    52→function loopNoise(t: number, period: number, seed = 0): number {
    53→  const p = (t / period) * TAU;
    54→  return (
    55→    0.55 * Math.sin(p + seed) +
    56→    0.3 * Math.sin(2 * p + seed * 1.7 + 1.1) +
    57→    0.15 * Math.sin(3 * p + seed * 2.3 + 2.4)
    58→  );
    59→}
    60→
    61→/** Deterministic PRNG (mulberry32) — same blink calendar every visit. */
    62→function createRng(seed: number) {
    63→  let a = seed >>> 0;
    64→  return () => {
    65→    a = (a + 0x6d2b79f5) >>> 0;
    66→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    67→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    68→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    69→  };
    70→}

    72→// ---- blink -------------------------------------------------------------
    73→
    74→const BLINK_RNG = createRng(0x5eed);
    75→/** Pre-drawn blink calendar: 1.9–4.6 s apart, 18% chance of a double-blink. */
    76→const BLINKS: number[] = (() => {
    77→  const out: number[] = [];
    78→  let t = 1.4;
    79→  while (t < 900) {
    80→    out.push(t);
    81→    t += 1.9 + BLINK_RNG() * 2.7;
    82→    if (BLINK_RNG() < 0.18) {
    83→      out.push(t);
    84→      t += 0.24;
    85→    }
    86→  }
    87→  return out;
    88→})();

    90→const BLINK_DUR = 0.18;
    91→
    92→/** Lid over time: closes fast (45% of the dip), reopens a touch slower. */
    93→function blinkLid(t: number): number {
    94→  for (let i = 0; i < BLINKS.length; i++) {
    95→    const start = BLINKS[i]!;
    96→    if (t < start) break;
    97→    const k = (t - start) / BLINK_DUR;
    98→    if (k >= 0 && k <= 1) {
    99→      return k < 0.45 ? 1 - k / 0.45 : (k - 0.45) / 0.55;
   100→    }
   101→  }
   102→  return 1;
   103→}
   104→
   105→/** A blink is a screen-space vertical squash; the eye never fully shuts. */
   106→const blinkScale = (lid: number) => 0.06 + 0.94 * clamp(lid);

   110→/** Closed Catmull-Rom → cubics. 64 points: smooth to the pixel at any size. */
   111→function closedPath(pts: [number, number][], tension = 1 / 6): string {
   112→  const n = pts.length;
   113→  let d = `M${r2(pts[0]![0])} ${r2(pts[0]![1])}`;
   114→  for (let i = 0; i < n; i++) {
   115→    const p0 = pts[(i - 1 + n) % n]!;
   116→    const p1 = pts[i]!;
   117→    const p2 = pts[(i + 1) % n]!;
   118→    const p3 = pts[(i + 2) % n]!;
   119→    const c1x = p1[0] + (p2[0] - p0[0]) * tension;
   120→    const c1y = p1[1] + (p2[1] - p0[1]) * tension;
   121→    const c2x = p2[0] - (p3[0] - p1[0]) * tension;
   122→    const c2y = p2[1] - (p3[1] - p1[1]) * tension;
   123→    d += `C${r2(c1x)} ${r2(c1y)} ${r2(c2x)} ${r2(c2y)} ${r2(p2[0])} ${r2(p2[1])}`;
   124→  }
   125→  return `${d}Z`;
   126→}
   127→
   128→const N_PTS = 64;
   129→const BODY_CX = 60;
   130→const BODY_CY = 58;
   131→const BODY_R = 40;
   132→
   133→/** The body: a perfect circle (plus squash/breath applied by the caller). */
   134→function bodyPath(cx: number, cy: number, sx: number, sy: number): string {
   135→  const pts: [number, number][] = new Array(N_PTS);
   136→  for (let i = 0; i < N_PTS; i++) {
   137→    const a = (i / N_PTS) * TAU;
   138→    pts[i] = [cx + Math.cos(a) * BODY_R * sx, cy + Math.sin(a) * BODY_R * sy];
   139→  }
   140→  return closedPath(pts);
   141→}
   142→
   143→const INITIAL_D = bodyPath(BODY_CX, BODY_CY, 1, 1);

   145→/** Capsule (stadium) centred on the origin — the exact shape of the eyes. */
   146→function capsulePath(w: number, h: number): string {
   147→  const hw = Math.max(w, 0.01) / 2;
   148→  const hh = Math.max(h, 0.01) / 2;
   149→  const r = Math.min(hw, hh);
   150→  return (
   151→    `M${r2(-hw)} ${r2(-hh + r)}` +
   152→    `A${r2(r)} ${r2(r)} 0 0 1 ${r2(-hw + r)} ${r2(-hh)}` +
   153→    `L${r2(hw - r)} ${r2(-hh)}` +
   154→    `A${r2(r)} ${r2(r)} 0 0 1 ${r2(hw)} ${r2(-hh + r)}` +
   155→    `L${r2(hw)} ${r2(hh - r)}` +
   156→    `A${r2(r)} ${r2(r)} 0 0 1 ${r2(hw - r)} ${r2(hh)}` +
   157→    `L${r2(-hw + r)} ${r2(hh)}` +
   158→    `A${r2(r)} ${r2(r)} 0 0 1 ${r2(-hw)} ${r2(hh - r)}Z`
   159→  );
   160→}

   162→// ---- head sphere -------------------------------------------------------
   163→
   164→type Vec3 = [number, number, number];
   165→
   166→/** Rotate two orthonormal vectors within their common plane. */
   167→function spin(u: Vec3, v: Vec3, angle: number): [Vec3, Vec3] {
   168→  const c = Math.cos(angle);
   169→  const s = Math.sin(angle);
   170→  return [
   171→    [u[0] * c + v[0] * s, u[1] * c + v[1] * s, u[2] * c + v[2] * s],
   172→    [v[0] * c - u[0] * s, v[1] * c - u[1] * s, v[2] * c - u[2] * s],
   173→  ];
   174→}

   176→export interface HeadGaze {
   177→  yaw: number; // deg, positive = looks right
   178→  pitch: number; // deg, positive = looks up
   179→  roll: number; // deg, head tilt
   180→}

   182→interface EyePose {
   183→  /** centre on the unit sphere, screen space */
   184→  x: number;
   185→  y: number;
   186→  /** tangent frame columns, for SVG matrix(a,b,c,d,e,f) */
   187→  a: number;
   188→  b: number;
   189→  c: number;
   190→  d: number;
   191→  /** z of the eye normal: fades/reads the limb */
   192→  depth: number;
   193→}
   194→
   195→const deg = (d: number) => (d * Math.PI) / 180;
   196→
   197→/** Two eyes on a sphere of radius 1, separated by `split` degrees each side
   198→ * of the forward ray. The tangent projection gives position, lean and
   199→ * edge compression at once — no flat translation anywhere. */
   200→function eyePoses(gaze: HeadGaze, split: number): [EyePose, EyePose] {
   201→  let f: Vec3 = [0, 0, 1];
   202→  let right: Vec3 = [1, 0, 0];
   203→  let down: Vec3 = [0, 1, 0];
   204→
   205→  [f, right] = spin(f, right, deg(gaze.yaw));
   206→  [down, f] = spin(down, f, deg(gaze.pitch));
   207→  [right, down] = spin(right, down, deg(gaze.roll));
   208→
   209→  const build = (side: number): EyePose => {
   210→    const [ef, er] = spin(f, right, deg(split * side));
   211→    return {
   212→      x: ef[0],
   213→      y: ef[1],
   214→      a: er[0],
   215→      b: er[1],
   216→      c: down[0],
   217→      d: down[1],
   218→      depth: ef[2],
   219→    };
   220→  };
   221→
   222→  return [build(-1), build(1)];
   223→}

   375→function setAttr(el: Element | null, name: string, v: string) {
   376→  el?.setAttribute(name, v);

   379→function paintEye(
   380→  el: SVGPathElement | null,
   381→  pose: EyePose,
   382→  w: number,
   383→  h: number,
   384→  tilt: number,
   385→  lid: number,
   386→  driftX: number,
   387→  driftY: number,
   388→) {
   389→  if (!el) return;
   390→  const k = blinkScale(lid);
   391→  const phi = deg(tilt);
   392→  const cp = Math.cos(phi);
   393→  const sp = Math.sin(phi);
   394→  // per-eye lean: compose the tangent frame with an in-plane rotation
   395→  const ax = pose.a * cp + pose.c * sp;
   396→  const ay = pose.b * cp + pose.d * sp;
   397→  const bx = -pose.a * sp + pose.c * cp;
   398→  const by = -pose.b * sp + pose.d * cp;
   399→  const tx = BODY_CX + pose.x * BODY_R + driftX * BODY_R;
   400→  const ty = BODY_CY + pose.y * BODY_R + driftY * BODY_R;
   401→  el.setAttribute("d", capsulePath(w * BODY_R, h * BODY_R));
   402→  el.setAttribute(
   403→    "transform",
   404→    `matrix(${r2(ax)},${r2(ay * k)},${r2(bx)},${r2(by * k)},${r2(tx)},${r2(ty)})`,
   405→  );
   406→  el.setAttribute("opacity", r2(clamp(pose.depth / 0.12)).toFixed(2));
   407→}

   409→function paint(e: Engine, t: number, P: Painters) {
   410→  const p = e.pose;
   411→
   412→  // Body: still at rest. Breath is 0.5% of height; drift is sub-pixel.
   413→  const breath = 1 + Math.sin((t / 3.4) * TAU) * 0.005;
   414→  const driftX = loopNoise(t, 7.9, 1.9) * 0.006;
   415→  const driftY = loopNoise(t, 5.3, 0.3) * 0.007;
   416→  const sq = p.bounce * 0.055 * Math.abs(Math.sin(t * 5.2));
   417→  const sx = 1 + sq;
   418→  const sy = (1 - sq * 0.85) * breath;
   419→  setAttr(
   420→    P.body,
   421→    "d",
   422→    bodyPath(BODY_CX + driftX * BODY_R, BODY_CY + driftY * BODY_R, sx, sy),
   423→  );
   424→  setAttr(P.shadow, "rx", r2(23 * (2 - sy)).toFixed(2));
   425→
   426→  // Gaze: pose direction, replaced by the pointer look via mix, drift on top.
   427→  const wander = e.look.wander;
   428→  const yaw =
   429→    lerp(p.yaw, e.look.yaw, e.look.mix) +
   430→    (loopNoise(t, 11.3, 0.4) * 5.5 + loopNoise(t, 3.7, 2.1) * 1.6) * wander;
   431→  const pitch =
   432→    lerp(p.pitch, e.look.pitch, e.look.mix) +
   433→    (loopNoise(t, 9.1, 1.3) * 4.2 + loopNoise(t, 4.3, 0.7) * 1.3) * wander;
   434→  const roll = p.roll + loopNoise(t, 13.7, 3.2) * 2.2 * wander;
   435→
   436→  // Blink: scheduled, plus the forced cut that masks a mood morph.
   437→  const forced = clamp((t - e.blinkAt) / 0.2);
   438→  const forcedLid = forced < 1 ? Math.abs(forced * 2 - 1) : 1;
   439→  const lid = Math.min(blinkLid(t), forcedLid);
   440→
   441→  const [left, right] = eyePoses({ yaw, pitch, roll }, p.split);
   442→  paintEye(P.eyeL, left, p.wL, p.hL, p.tiltL, lid, driftX, driftY);
   443→  paintEye(P.eyeR, right, p.wR, p.hR, p.tiltR, lid, driftX, driftY);
   444→
   445→  // Mouth and cheeks ride the head roll so the face stays coherent.
   446→  setAttr(P.mouthG, "transform", `rotate(${r2(roll)} ${BODY_CX} ${BODY_CY})`);
   447→
   448→  const open = clamp(p.mouthOpen, 0, 1);
   449→  setAttr(P.mouthOpen, "ry", r2(1.2 + 3.4 * open).toFixed(2));
   450→  setAttr(P.mouthOpen, "opacity", open > 0.12 ? "1" : "0");
   451→  setAttr(
   452→    P.mouthCurve,
   453→    "d",
   454→    `M 54 76 Q 60 ${r2(76 - 5.5 * p.mouth)} 66 76`,
   455→  );
   456→  setAttr(
   457→    P.mouthCurve,
   458→    "opacity",
   459→    p.mouthShow > 0.5 && open <= 0.12 ? "1" : "0",
   460→  );
   461→  const cheek = (p.cheeks * 0.4).toFixed(3);
   462→  setAttr(P.cheekL, "opacity", cheek);
   463→  setAttr(P.cheekR, "opacity", cheek);
   464→}

   473→/** No-op store subscription for the client gate (nothing to subscribe to). */
   474→const emptySubscribe = () => () => {};
   475→
   476→export function SeeqBuddy() {
   477→  // SSR-safe client gate: no effect, no state — the server snapshot renders null.
   478→  const mounted = useSyncExternalStore(emptySubscribe, () => true, () => false);
   479→  const [collapsed, setCollapsed] = useState(() => {
   480→    try {
   481→      return localStorage.getItem(COLLAPSE_KEY) === "1";
   482→    } catch {
   483→      return false;
   484→    }
   485→  });
   486→  const [bubble, setBubble] = useState<{ text: string; mood: Mood } | null>(null);
   487→
   488→  const wrapRef = useRef<HTMLDivElement | null>(null);
   489→  const bodyRef = useRef<SVGPathElement | null>(null);
   490→  const shadowRef = useRef<SVGEllipseElement | null>(null);
   491→  const eyeLRef = useRef<SVGPathElement | null>(null);
   492→  const eyeRRef = useRef<SVGPathElement | null>(null);
   493→  const mouthGRef = useRef<SVGGElement | null>(null);
   494→  const mouthCurveRef = useRef<SVGPathElement | null>(null);
   495→  const mouthOpenRef = useRef<SVGEllipseElement | null>(null);
   496→  const cheekLRef = useRef<SVGEllipseElement | null>(null);
   497→  const cheekRRef = useRef<SVGEllipseElement | null>(null);
   498→
   499→  const engineRef = useRef<Engine | null>(null);
   500→  const paintersRef = useRef<Painters>({
   501→    body: null, shadow: null, eyeL: null, eyeR: null, mouthG: null,
   502→    mouthCurve: null, mouthOpen: null, cheekL: null, cheekR: null,
   503→  });
   504→  const moodRef = useRef<Mood>("idle");
   505→  const collapsedRef = useRef(collapsed);
   506→  const showTimer = useRef<number | null>(null);
   507→  const hideTimer = useRef<number | null>(null);
   508→  const bag = useRef<{ order: number[]; i: number }>({ order: [], i: 0 });
   509→  const praiseIdx = useRef(0);
   510→
   511→  const clearTimers = () => {
   512→    if (showTimer.current !== null) window.clearTimeout(showTimer.current);
   513→    if (hideTimer.current !== null) window.clearTimeout(hideTimer.current);
   514→    showTimer.current = null;
   515→    hideTimer.current = null;
   516→  };
   517→
   518→  const scheduleNextTip = (delayMs: number) => {
   519→    if (showTimer.current !== null) window.clearTimeout(showTimer.current);
   520→    showTimer.current = window.setTimeout(() => {
   521→      if (collapsedRef.current) return;
   522→      const b = bag.current;
   523→      if (b.i >= b.order.length) {
   524→        b.order = TIPS.map((_, i) => i);
   525→        for (let i = b.order.length - 1; i > 0; i--) {
   526→          const j = Math.floor(Math.random() * (i + 1));
   527→          [b.order[i], b.order[j]] = [b.order[j], b.order[i]];
   528→        }
   529→        b.i = 0;
   530→      }
   531→      const tip = TIPS[b.order[b.i++]];
   532→      setBubble({ text: tip.text, mood: tip.mood });
   533→      if (hideTimer.current !== null) window.clearTimeout(hideTimer.current);
   534→      hideTimer.current = window.setTimeout(() => {
   535→        setBubble(null);
   536→        scheduleNextTip(TIP_GAP_MS + Math.random() * TIP_GAP_JITTER_MS);
   537→      }, TIP_HOLD_MS);
   538→    }, delayMs);
   539→  };
   540→
   541→  // Paint a single static frame (reduced motion / mood sync).
   542→  const staticPaint = () => {
   543→    const e = engineRef.current;
   544→    if (!e) return;
   545→    e.pose = collapsedRef.current
   546→      ? { ...MOODS[moodRef.current], mouthShow: 0 }
   547→      : { ...MOODS[moodRef.current] };
   548→    paint(e, 1.3, paintersRef.current);
   549→  };
   550→
   551→  // Bubble drives the mood; also repaints the static frame under reduced motion.
   552→  useEffect(() => {
   553→    moodRef.current = bubble?.mood ?? "idle";
   554→    if (engineRef.current?.reduced) staticPaint();
   555→  }, [bubble]);
   556→
   557→  // Collapse changes: persist + keep the engine's view in sync. Speech is
   558→  // paused/resumed by the event handlers that flip `collapsed`.
   559→  useEffect(() => {
   560→    collapsedRef.current = collapsed;
   561→    try {
   562→      localStorage.setItem(COLLAPSE_KEY, collapsed ? "1" : "0");
   563→    } catch {
   564→      /* private mode — ignore */
   565→    }
   566→  }, [collapsed]);
   567→
   568→  // Mount: start the engine and begin the tip cycle.
   569→  useEffect(() => {
   570→    paintersRef.current = {
   571→      body: bodyRef.current,
   572→      shadow: shadowRef.current,
   573→      eyeL: eyeLRef.current,
   574→      eyeR: eyeRRef.current,
   575→      mouthG: mouthGRef.current,
   576→      mouthCurve: mouthCurveRef.current,
   577→      mouthOpen: mouthOpenRef.current,
   578→      cheekL: cheekLRef.current,
   579→      cheekR: cheekRRef.current,
   580→    };
   581→
   582→    const e: Engine = {
   583→      mood: "idle",
   584→      frozen: null,
   585→      tMood: 0,
   586→      blinkAt: -10,
   587→      pose: { ...IDLE },
   588→      look: { yaw: IDLE.yaw, pitch: IDLE.pitch, mix: 0, wander: 1 },
   589→      lookT: { yaw: IDLE.yaw, pitch: IDLE.pitch, mix: 0, wander: 1 },
   590→      lookPrev: { yaw: IDLE.yaw, pitch: IDLE.pitch, mix: 0, wander: 1 },
   591→      lookAt: -10,
   592→      pointer: null,
   593→      raf: 0,
   594→      last: 0,
   595→      running: false,
   596→      reduced: window.matchMedia("(prefers-reduced-motion: reduce)").matches,
   597→    };
   598→    engineRef.current = e;
   599→
   600→    /** Mood cut: freeze the composite, re-target, and mask with a blink. */
   601→    const setMood = (id: Mood, t: number) => {
   602→      e.frozen = { ...e.pose };
   603→      e.mood = id;
   604→      e.tMood = t;
   605→      e.blinkAt = t;
   606→    };
   607→
   608→    /** Re-pose the look from the CURRENT interpolated value, or the gaze
   609→     * would step backwards one catch-up per pointer move and tremble. The
   610→     * effective look for this frame is the value caught up to `t` — each
   611→     * frame advances easeOutQuint(dt/morph) of the remaining gap. */
   612→    const setLook = (look: Look, t: number) => {
   613→      const k = easeOutQuint(clamp((t - e.lookAt) / LOOK_MORPH));
   614→      const cur: Look = {
   615→        yaw: lerp(e.lookPrev.yaw, e.lookT.yaw, k),
   616→        pitch: lerp(e.lookPrev.pitch, e.lookT.pitch, k),
   617→        mix: lerp(e.lookPrev.mix, e.lookT.mix, k),
   618→        wander: lerp(e.lookPrev.wander, e.lookT.wander, k),
   619→      };
   620→      e.lookPrev = cur;
   621→      e.lookT = look;
   622→      e.lookAt = t;
   623→      e.look = cur;
   624→    };
   625→
   626→    const aim = (t: number) => {
   627→      const el = wrapRef.current;
   628→      if (!el) return;
   629→      const box = el.getBoundingClientRect();
   630→      // A zero-surface box (hidden tab) would make the normalisation 0/0 —
   631→      // a NaN here would poison the engine for good.
   632→      if (box.width === 0 || box.height === 0) return;
   633→      if (e.pointer) {
   634→        const nx = clamp((e.pointer.x - (box.left + box.width / 2)) / Math.max(1, window.innerWidth / 2), -1, 1);
   635→        const ny = clamp((e.pointer.y - (box.top + box.height / 2)) / Math.max(1, window.innerHeight / 2), -1, 1);
   636→        setLook({ yaw: nx * YAW_MAX, pitch: FOLLOW_PITCH - ny * PITCH_MAX, mix: 1, wander: 0 }, t);
   637→      } else {
   638→        setLook({ yaw: IDLE.yaw, pitch: IDLE.pitch, mix: 0, wander: 1 }, t);
   639→      }
   640→    };
   641→
   642→    const tick = (now: number) => {
   643→      const t = now / 1000;
   644→      e.last = now;
   645→
   646→      if (moodRef.current !== e.mood) setMood(moodRef.current, t);
   647→
   648→      // Mood blend: pure easeOutQuint from the frozen composite — no overshoot,
   649→      // and an interrupted morph re-blends from where it actually was.
   650→      const target = collapsedRef.current
   651→        ? { ...MOODS[e.mood], mouthShow: 0 }
   652→        : MOODS[e.mood];
   653→      const since = t - e.tMood;
   654→      if (e.frozen && since < MOOD_MORPH) {
   655→        e.pose = blendMood(e.frozen, target, easeOutQuint(since / MOOD_MORPH));
   656→      } else {
   657→        e.frozen = null;
   658→        e.pose = target;
   659→      }
   660→
   661→      aim(t);
   662→
   663→      paint(e, t, paintersRef.current);
   664→    };
   665→
   666→    const startLoop = () => {
   667→      if (e.reduced) {
   668→        staticPaint();
   669→        return;
   670→      }
   671→      if (e.running) return;
   672→      e.running = true;
   673→      e.last = 0;
   674→      const step = (now: number) => {
   675→        tick(now);
   676→        e.raf = requestAnimationFrame(step);
   677→      };
   678→      e.raf = requestAnimationFrame(step);
   679→    };
   680→    const stopLoop = () => {
   681→      e.running = false;
   682→      cancelAnimationFrame(e.raf);
   683→    };
   684→
   685→    // Touch pointers are ignored: a lifted finger would freeze the gaze on
   686→    // its last spot, which reads as a bug.
   687→    const onMove = (ev: PointerEvent) => {
   688→      if (ev.pointerType === "touch") return;
   689→      e.pointer = { x: ev.clientX, y: ev.clientY };
   690→    };
   691→    const onLeave = () => {
   692→      e.pointer = null;
   693→    };
   694→    const onVis = () => {
   695→      if (document.hidden) stopLoop();
   696→      else startLoop();
   697→    };
   698→
   699→    window.addEventListener("pointermove", onMove, { passive: true });
   700→    document.documentElement.addEventListener("pointerleave", onLeave);
   701→    document.addEventListener("visibilitychange", onVis);
   702→    startLoop();
   703→    if (!collapsedRef.current) scheduleNextTip(2600);
   704→
   705→    return () => {
   706→      window.removeEventListener("pointermove", onMove);
   707→      document.documentElement.removeEventListener("pointerleave", onLeave);
   708→      document.removeEventListener("visibilitychange", onVis);
   709→      stopLoop();
   710→      clearTimers();
   711→      engineRef.current = null;
   712→    };
   713→  }, []);
   714→
   715→  const poke = () => {
   716→    if (collapsedRef.current) {
   717→      expand();
   718→      return;
   719→    }
   720→    clearTimers();
   721→    const text = PRAISE[praiseIdx.current++ % PRAISE.length];
   722→    setBubble({ text, mood: "excited" });
   723→    hideTimer.current = window.setTimeout(() => {
   724→      setBubble(null);
   725→      scheduleNextTip(TIP_GAP_MS + Math.random() * TIP_GAP_JITTER_MS);
   726→    }, 3000);
   727→  };
   728→
   729→  const expand = () => {
   730→    setCollapsed(false);
   731→    scheduleNextTip(2600);
   732→  };
   733→
   734→  const collapseBuddy = () => {
   735→    clearTimers();
   736→    setBubble(null);
   737→    setCollapsed(true);
   738→  };
   739→
   740→  const dismiss = () => {
   741→    clearTimers();
   742→    setBubble(null);
   743→    scheduleNextTip(15000);
   744→  };
   745→
   746→  if (!mounted) return null;
   747→
   748→  return createPortal(
   749→    // Positioning: mobile floats above the dev-tools badge at the corner; on
   750→    // lg+ the left rail (300px) owns the corner, so the buddy docks to the
   751→    // content column instead. The calc tracks the centered max-w-7xl
   752→    // container: (vw-1280)/2 + page padding 16 + sidebar 300 + gap 24.
   753→    <div
   754→      ref={wrapRef}
   755→      data-seeq="1"
   756→      className="fixed bottom-14 left-3 z-[55] lg:bottom-4 lg:left-[max(340px,calc((100vw-1280px)/2+340px))]"
   757→    >
   758→      <div className="flex items-end gap-2">
   759→        <div className="group relative shrink-0">
   760→          <button
   761→            type="button"
   762→            onClick={poke}
   763→            aria-expanded={!collapsed}
   764→            aria-label={collapsed ? "Open SeeQ study buddy" : "SeeQ study buddy"}
   765→            className={
   766→              "block cursor-pointer touch-manipulation select-none rounded-full outline-none transition-transform duration-300 hover:scale-105 focus-visible:ring-2 focus-visible:ring-emerald-400/60 " +
   767→              (collapsed ? "h-11 w-11" : "h-28 w-28 sm:h-32 sm:w-32")
   768→            }
   769→          >
   770→            <svg viewBox="0 0 120 120" className="h-full w-full overflow-visible" aria-hidden>
   771→              <defs>
   772→                <radialGradient id="seeq-body" cx="38%" cy="30%" r="80%">
   773→                  <stop offset="0%" stopColor="#4be3ae" />
   774→                  <stop offset="55%" stopColor="#17b57f" />
   775→                  <stop offset="100%" stopColor="#0a7d58" />
   776→                </radialGradient>
   777→              </defs>
   778→              <ellipse ref={shadowRef} cx={BODY_CX} cy="106" rx="23" ry="4.2" fill="#000" opacity="0.28" />
   779→              <path ref={bodyRef} d={INITIAL_D} fill="url(#seeq-body)" />
   780→              <path ref={eyeLRef} d="" fill="#f4fdfa" transform="" />
   781→              <path ref={eyeRRef} d="" fill="#f4fdfa" transform="" />
   782→              <g ref={mouthGRef}>
   783→                <path
   784→                  ref={mouthCurveRef}
   785→                  d=""
   786→                  fill="none"
   787→                  stroke="#053b2b"
   788→                  strokeWidth="2.6"
   789→                  strokeLinecap="round"
   790→                  opacity="0"
   791→                />
   792→                <ellipse ref={mouthOpenRef} cx={BODY_CX} cy="77.5" rx="3.4" ry="1.4" fill="#053b2b" opacity="0" />
   793→                <ellipse ref={cheekLRef} cx="41" cy="67" rx="4.8" ry="2.9" fill="#fb7185" opacity="0" />
   794→                <ellipse ref={cheekRRef} cx="79" cy="67" rx="4.8" ry="2.9" fill="#fb7185" opacity="0" />
   795→              </g>
   796→            </svg>
   797→          </button>
   798→          {!collapsed && (
   799→            <button
   800→              type="button"
   801→              onClick={collapseBuddy}
   802→              aria-label="Collapse SeeQ"
   803→              title="Collapse SeeQ"
   804→              className="absolute -right-0.5 -top-0.5 flex h-6 w-6 items-center justify-center rounded-full border border-zinc-700 bg-zinc-900 text-zinc-400 opacity-0 transition-all hover:border-zinc-600 hover:text-zinc-200 focus-visible:opacity-100 group-hover:opacity-100"
   805→            >
   806→              <Minus className="h-3 w-3" />
   807→            </button>
   808→          )}
   809→        </div>
   810→
   811→        <AnimatePresence>
   812→          {bubble && !collapsed && (
   813→            <motion.div
   814→              data-seeq-bubble="1"
   815→              initial={{ opacity: 0, y: 10, scale: 0.92 }}
   816→              animate={{ opacity: 1, y: 0, scale: 1 }}
   817→              exit={{ opacity: 0, y: 6, scale: 0.96 }}
   818→              transition={{ type: "spring", stiffness: 380, damping: 26 }}
   819→              className="pointer-events-auto relative mb-2 w-56 max-w-[62vw] rounded-2xl rounded-bl-md border border-zinc-800 bg-zinc-900/95 p-3 pr-7 shadow-xl shadow-black/40 backdrop-blur"
   820→            >
   821→              <p aria-live="polite" className="text-[12.5px] leading-relaxed text-zinc-200">
   822→                {bubble.text}
   823→              </p>
   824→              <button
   825→                type="button"
   826→                onClick={dismiss}
   827→                aria-label="Dismiss tip"
   828→                className="absolute right-1.5 top-1.5 rounded-md p-1 text-zinc-500 transition-colors hover:bg-zinc-800 hover:text-zinc-300"
   829→              >
   830→                <X className="h-3 w-3" />
   831→              </button>
   832→            </motion.div>
   833→          )}
   834→        </AnimatePresence>
   835→      </div>
   836→    </div>,
   837→    document.body,
   838→  );
   839→}
