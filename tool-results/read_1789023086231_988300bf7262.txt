    75→function loadOpenModules(): number[] | null {
    76→  try {
    77→    const raw = localStorage.getItem(MODULES_OPEN_KEY);
    78→    if (!raw) return null;
    79→    const arr = JSON.parse(raw);
    80→    return Array.isArray(arr) ? arr.filter((n) => typeof n === "number") : null;
    81→  } catch {
    82→    return null;
    83→  }
    84→}

   110→function useBadgeWatcher() {
   111→  const xpEvents = useProgress((s) => s.xpEvents);
   112→  const mcqAttempts = useProgress((s) => s.mcqAttempts);
   113→  const badges = useProgress((s) => s.badges);
   114→  const xpByDay = useProgress((s) => s.xpByDay);
   115→  const dailyGoal = useProgress((s) => s.dailyGoal);
   116→  const prevLevel = useRef(-1);
   117→  const ready = useRef(false);
   118→
   119→  const xp = useMemo(() => totalXp({ xpEvents }), [xpEvents]);
   120→  const level = levelFor(xp);
   121→
   122→  const ctx = useMemo(() => {
   123→    const problemsSolved = Object.keys(xpEvents).filter((k) => k.startsWith("solve:")).length;
   124→    const mcqsCorrect = Object.values(mcqAttempts).filter((a) => a.correct).length;
   125→    const lessonsDone = Object.keys(xpEvents).filter((k) => k.startsWith("lesson:")).length;
   126→    const perfectQuizzes = LESSONS.filter((l) =>
   127→      l.mcqs.every((m) => mcqAttempts[m.id]?.firstTry),
   128→    ).length;
   129→    return { lessonsDone, problemsSolved, mcqsCorrect, perfectQuizzes };
   130→  }, [xpEvents, mcqAttempts]);
   131→
   132→  const streak = useMemo(() => currentStreak({ xpByDay, dailyGoal }), [xpByDay, dailyGoal]);
   133→
   134→  useEffect(() => {
   135→    if (!ready.current) {
   136→      ready.current = true;
   137→      prevLevel.current = level.index;
   138→      return;
   139→    }
   140→    const store = useProgress.getState();
   141→    for (const b of BADGES) {
   142→      if (store.badges[b.id] === undefined && b.test(ctx, xp, streak)) {
   143→        if (store.unlockBadge(b.id)) {
   144→          celebrate({ kind: "badge", title: `Badge · ${b.title}`, sub: b.desc });
   145→        }
   146→      }
   147→    }
   148→    if (level.index > prevLevel.current) {
   149→      celebrate({ kind: "level", title: `Level up — ${level.title}`, sub: `Rank ${level.index + 1} of 6` });
   150→      prevLevel.current = level.index;
   151→    }
   152→  }, [ctx, xp, streak, level]);
   153→
   154→  return { xp, level, streak, ctx };
   155→}

   157→function RankBlock({ xp, level, streak }: { xp: number; level: ReturnType<typeof levelFor>; streak: number }) {
   158→  return (
   159→    <div className="sheen rounded-lg border border-zinc-800 bg-zinc-900/50 p-3.5">
   160→      <div className="flex items-center justify-between">

   162→        <span className="tnum font-mono text-xs text-emerald-300">{xp.toLocaleString()} XP</span>
   163→      </div>
   164→      <div className="bar-shimmer mt-2.5 h-1.5 overflow-hidden rounded-full bg-zinc-800">
   165→        <div
   166→          className="h-full rounded-full bg-emerald-400 transition-[width] duration-700 ease-out"
   167→          style={{ width: `${Math.round(level.pct * 100)}%` }}
   168→        />
   169→      </div>
   170→      <p className="mt-1.5 text-[11px] text-zinc-500">
   171→        {level.next === null ? "Top rank reached — Query Wizard" : `${(level.next - xp).toLocaleString()} XP to ${titleAfter(level.index)}`}
   172→      </p>
   173→      <div className="mt-2.5 flex items-center gap-1.5 border-t border-zinc-800 pt-2.5 text-[11px] text-zinc-400">
   174→        <Flame className={cn("h-3.5 w-3.5", streak > 0 ? "flame-live text-amber-400" : "text-zinc-600")} />
   175→        {streak > 0 ? `${streak}-day streak — keep it alive` : "Hit your daily goal to start a streak"}
   176→      </div>
   177→    </div>
   178→  );
   179→}

   181→function titleAfter(index: number): string {
   182→  const ladder = ["Query Apprentice", "Analyst I", "Analyst II", "Senior Analyst", "Query Wizard"];
   183→  return ladder[Math.min(index, ladder.length - 1)];
   184→}
