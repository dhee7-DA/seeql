   139→function useBadgeWatcher() {
   140→  const xpEvents = useProgress((s) => s.xpEvents);
   141→  const mcqAttempts = useProgress((s) => s.mcqAttempts);
   142→  const badges = useProgress((s) => s.badges);
   143→  const xpByDay = useProgress((s) => s.xpByDay);
   144→  const dailyGoal = useProgress((s) => s.dailyGoal);
   145→  const prevLevel = useRef(-1);
   146→  const ready = useRef(false);
   147→
   148→  const xp = useMemo(() => totalXp({ xpEvents }), [xpEvents]);
   149→  const level = levelFor(xp);
   150→
   151→  const ctx = useMemo(() => {
   152→    const problemsSolved = Object.keys(xpEvents).filter((k) => k.startsWith("solve:")).length;
   153→    const mcqsCorrect = Object.values(mcqAttempts).filter((a) => a.correct).length;
   154→    const lessonsDone = Object.keys(xpEvents).filter((k) => k.startsWith("lesson:")).length;
   155→    const perfectQuizzes = LESSONS.filter((l) =>
   156→      l.mcqs.every((m) => mcqAttempts[m.id]?.firstTry),
   157→    ).length;
   158→    return { lessonsDone, problemsSolved, mcqsCorrect, perfectQuizzes };
   159→  }, [xpEvents, mcqAttempts]);
   160→
   161→  const streak = useMemo(() => currentStreak({ xpByDay, dailyGoal }), [xpByDay, dailyGoal]);
   162→
   163→  useEffect(() => {
   164→    if (!ready.current) {
   165→      ready.current = true;
   166→      prevLevel.current = level.index;
   167→      return;
   168→    }
   169→    const store = useProgress.getState();
   170→    for (const b of BADGES) {
   171→      if (store.badges[b.id] === undefined && b.test(ctx, xp, streak)) {
   172→        if (store.unlockBadge(b.id)) {
   173→          celebrate({ kind: "badge", title: `Badge · ${b.title}`, sub: b.desc });
   174→        }
   175→      }
   176→    }
   177→    if (level.index > prevLevel.current) {
   178→      celebrate({ kind: "level", title: `Level up — ${level.title}`, sub: `Rank ${level.index + 1} of 6` });
   179→      prevLevel.current = level.index;
   180→    }
   181→  }, [ctx, xp, streak, level]);
   182→
   183→  return { xp, level, streak, ctx };
   184→}

   186→function RankBlock({ xp, level, streak }: { xp: number; level: ReturnType<typeof levelFor>; streak: number }) {
   187→  return (
   188→    <div className="sheen rounded-lg border border-zinc-800 bg-zinc-900/50 p-3.5">
   189→      <div className="flex items-center justify-between">
   190→        <span className="font-display text-sm font-semibold text-zinc-100">{level.title}</span>
   191→        <span className="tnum font-mono text-xs text-emerald-300">{xp.toLocaleString()} XP</span>
   192→      </div>
   193→      <div className="bar-shimmer mt-2.5 h-1.5 overflow-hidden rounded-full bg-zinc-800">
   194→        <div
   195→          className="h-full rounded-full bg-emerald-400 transition-[width] duration-700 ease-out"
   196→          style={{ width: `${Math.round(level.pct * 100)}%` }}
   197→        />
   198→      </div>
   199→      <p className="mt-1.5 text-[11px] text-zinc-500">
   200→        {level.next === null ? "Top rank reached — Query Wizard" : `${(level.next - xp).toLocaleString()} XP to ${titleAfter(level.index)}`}
   201→      </p>
   202→      <div className="mt-2.5 flex items-center gap-1.5 border-t border-zinc-800 pt-2.5 text-[11px] text-zinc-400">
   203→        <Flame className={cn("h-3.5 w-3.5", streak > 0 ? "flame-live text-amber-400" : "text-zinc-600")} />
   204→        {streak > 0 ? `${streak}-day streak — keep it alive` : "Hit your daily goal to start a streak"}
   205→      </div>
   206→    </div>
   207→  );
   208→}

   210→function titleAfter(index: number): string {
   211→  const ladder = ["Query Apprentice", "Analyst I", "Analyst II", "Senior Analyst", "Query Wizard"];
   212→  return ladder[Math.min(index, ladder.length - 1)];
