   102→function JsonView({ plan }: { plan: QueryPlan }) {
   103→  const [copied, setCopied] = useState(false);
   104→  const json = JSON.stringify({ ...plan, tree: undefined }, null, 2);
   105→
   106→  const copy = async () => {
   107→    try {
   108→      await navigator.clipboard.writeText(json);
   109→      setCopied(true);
   110→      setTimeout(() => setCopied(false), 1600);
   111→    } catch {
   112→      // clipboard unavailable — ignore
   113→    }
   114→  };
   115→
   116→  return (
   117→    <div className="space-y-3">
   118→      <div className="flex items-center justify-between">
   119→        <p className="text-xs text-zinc-400">
   120→          The structured plan the visualizer derived from your query.
   121→        </p>
   122→        <Button
   123→          variant="outline"
   124→          size="sm"
   125→          onClick={copy}
   126→          className="h-7 gap-1.5 border-zinc-700 text-xs text-zinc-300 hover:bg-zinc-800"
   127→        >
   128→          {copied ? <Check className="h-3.5 w-3.5 text-emerald-400" /> : <Copy className="h-3.5 w-3.5" />}
   129→          {copied ? "Copied" : "Copy JSON"}
   130→        </Button>
   131→      </div>
   132→      <pre className="thin-scrollbar max-h-[520px] overflow-auto rounded-xl border border-zinc-800 bg-zinc-900/70 p-4 font-mono text-xs leading-relaxed text-zinc-300">
   133→        {json}
   134→      </pre>
   135→    </div>
   136→  );
   137→}

   141→function PlaygroundView({
   142→  go,
   143→  sharedSql,
   144→  sharedDialect,
   145→}: {
   146→  go: (m: Mode) => void;
   147→  /** SQL coming from a #playground?q=… share link — boots the editor with it. */
   148→  sharedSql?: string | null;
   149→  sharedDialect?: DialectName;
   150→}) {
   151→  const [sql, setSql] = useState(sharedSql ?? DEFAULT_SAMPLE.sql);
   152→  const [dialect, setDialect] = useState<DialectName>(sharedDialect ?? DEFAULT_SAMPLE.dialect);
   153→  const [sampleId, setSampleId] = useState<string>(sharedSql ? "" : DEFAULT_SAMPLE.id);
   154→  const [copiedLink, setCopiedLink] = useState(false);
   155→  const [plan, setPlan] = useState<ParseResult | null>(null);
   156→  const [exec, setExec] = useState<ExecResponse | null>(null);
   157→  const [running, setRunning] = useState(false);
   158→  const [history, setHistory] = useState<string[]>([]);
   159→  const [resetting, setResetting] = useState(false);
   160→  const [tableCounts, setTableCounts] = useState<Record<string, number> | null>(null);
   161→
   162→  // live layer — the picture follows the text while it is being written
   163→  const [livePlan, setLivePlan] = useState<ParseResult | null>(null);
   164→  const [liveSql, setLiveSql] = useState("");
   165→  const [liveError, setLiveError] = useState<ParseError | null>(null);
   166→  const [ranSql, setRanSql] = useState("");
   167→  const [caret, setCaret] = useState(0);
   168→  const [hoverKind, setHoverKind] = useState<ClauseKind | null>(null);
   169→  const [pinnedKind, setPinnedKind] = useState<ClauseKind | null>(null);
   170→  const [dataVersion, setDataVersion] = useState(0); // bumped on DB reset
   171→
   172→  // live table row counts — powers the FROM dot-field in the query flow strip
   173→  useEffect(() => {
   174→    fetch("/api/stats")
   175→      .then((r) => r.json())
   176→      .then((j) => {
   177→        if (j?.ok) setTableCounts(j.counts as Record<string, number>);
   178→      })
   179→      .catch(() => {
   180→        /* flow strip simply shows no dot-field */
   181→      });
   182→  }, []);
   183→
   184→  // live parse — debounced so the flow strip, pipeline and stats redraw while
   185→  // the query is being typed, without firing on every keystroke
   186→  useEffect(() => {
   187→    const timer = window.setTimeout(() => {
   188→      const trimmed = sql.trim();
   189→      if (!trimmed) {
   190→        setLivePlan(null);
   191→        setLiveSql("");
   192→        setLiveError(null);
   193→        return;
   194→      }
   195→      // raw text goes out so error offsets align with the editor content
   196→      fetch("/api/parse", {
   197→        method: "POST",
   198→        headers: { "Content-Type": "application/json" },
   199→        body: JSON.stringify({ sql, dialect }),
   200→      })
   201→        .then((r) => r.json() as Promise<ParseResult>)
   202→        .then((res) => {
   203→          setLivePlan(res);
   204→          setLiveSql(sql);
   205→          setLiveError(res.ok ? null : res);
   206→        })
   207→        .catch(() => {
   208→          /* transient network hiccup — keep the previous live state */
   209→        });
   210→    }, 400);
   211→    return () => window.clearTimeout(timer);
   212→  }, [sql, dialect]);
   213→
   214→  const loadHistory = useCallback(() => {
   215→    try {
   216→      const raw = window.localStorage.getItem(HISTORY_KEY);
   217→      if (raw) setHistory(JSON.parse(raw) as string[]);
   218→    } catch {
   219→      // ignore malformed history
   220→    }
   221→  }, []);
   222→
   223→  const pushHistory = useCallback((text: string) => {
   224→    setHistory((prev) => {
   225→      const next = [text, ...prev.filter((h) => h !== text)].slice(0, HISTORY_MAX);
   226→      try {
   227→        window.localStorage.setItem(HISTORY_KEY, JSON.stringify(next));
   228→      } catch {
   229→        // storage unavailable
   230→      }
   231→      return next;
   232→    });
   233→  }, []);
   234→
   235→  /** Parse + execute in parallel; both panels update independently. */
   236→  const runAll = useCallback(
   237→    async (sqlText: string, dia: DialectName) => {
   238→      const trimmed = sqlText.trim();
   239→      if (!trimmed) return;
   240→      setRunning(true);
   241→      try {
   242→        const [parseRes, execRes] = await Promise.all([
   243→          // raw text → parser shifts error offsets back into the editor text
   244→          fetch("/api/parse", {
   245→            method: "POST",
   246→            headers: { "Content-Type": "application/json" },
   247→            body: JSON.stringify({ sql: sqlText, dialect: dia }),
   248→          }).then((r) => r.json() as Promise<ParseResult>),
   249→          fetch("/api/query", {
   250→            method: "POST",
   251→            headers: { "Content-Type": "application/json" },
   252→            body: JSON.stringify({ sql: trimmed }),
   253→          }).then((r) => r.json() as Promise<ExecResponse>),
   254→        ]);
   255→        setPlan(parseRes);
   256→        setRanSql(sqlText);
   257→        setExec(execRes);
   258→        setLivePlan(parseRes);
   259→        setLiveSql(sqlText);
   260→        setLiveError(parseRes.ok ? null : parseRes);
   261→        pushHistory(trimmed);
   262→      } catch {
   263→        setPlan((p) => p ?? { ok: false, error: "Could not reach the server. Please try again." });
   264→        setExec({ ok: false, error: "Could not reach the server. Please try again." });
   265→      } finally {
   266→        setRunning(false);
   267→      }
   268→    },
   269→    [pushHistory],
   270→  );
   271→
   272→  const parseOnly = useCallback(async (sqlText: string, dia: DialectName) => {
   273→    try {
   274→      const res = await fetch("/api/parse", {
   275→        method: "POST",
   276→        headers: { "Content-Type": "application/json" },
   277→        body: JSON.stringify({ sql: sqlText, dialect: dia }),
   278→      });
   279→      const parsed = (await res.json()) as ParseResult;
   280→      setLivePlan(parsed);
   281→      setLiveSql(sqlText);
   282→      setLiveError(parsed.ok ? null : parsed);
   283→    } catch {
   284→      setPlan({ ok: false, error: "Could not reach the parser. Please try again." });
   285→    }
   286→  }, []);
   287→
   288→  // run the initial query once on load — a #playground?q=… share link boots
   289→  // its own SQL, otherwise the default sample runs so the page shows live
   290→  // diagrams + data immediately
   291→  const bootSql = sharedSql ?? DEFAULT_SAMPLE.sql;
   292→  const bootDialect = sharedDialect ?? DEFAULT_SAMPLE.dialect;
   293→  useEffect(() => {
   294→    loadHistory();
   295→    runAll(bootSql, bootDialect);
   296→  }, [loadHistory, runAll, bootSql, bootDialect]);
   297→
   298→  /** Copy a link that reopens the playground with the current query loaded. */
   299→  const shareLink = async () => {
   300→    const params = new URLSearchParams({ q: sql });
   301→    if (dialect !== DEFAULT_SAMPLE.dialect) params.set("d", dialect);
   302→    const url = `${window.location.origin}${window.location.pathname}#playground?${params.toString()}`;
   303→    let ok = false;
   304→    try {
   305→      await navigator.clipboard.writeText(url);
   306→      ok = true;
   307→    } catch {
   308→      // clipboard API unavailable/denied — legacy fallback for restricted contexts
   309→      try {
   310→        const ta = document.createElement("textarea");
   311→        ta.value = url;
   312→        ta.style.position = "fixed";
   313→        ta.style.opacity = "0";
   314→        document.body.appendChild(ta);
   315→        ta.select();
   316→        ok = document.execCommand("copy");
   317→        document.body.removeChild(ta);
   318→      } catch {
   319→        ok = false;
   320→      }
   321→    }
   322→    if (ok) {
   323→      setCopiedLink(true);
   324→      setTimeout(() => setCopiedLink(false), 1600);
   325→    }
   326→  };
   327→
   328→  const onSampleChange = (id: string) => {
   329→    const sample = SAMPLE_QUERIES.find((s) => s.id === id);
   330→    if (!sample) return;
   331→    setSampleId(id);
   332→    setSql(sample.sql);
   333→    setDialect(sample.dialect);
   334→    runAll(sample.sql, sample.dialect);
   335→  };
   336→
   337→  const onDialectChange = (d: string) => {
   338→    const next = d as DialectName;
   339→    setDialect(next);
   340→    parseOnly(sql, next);
   341→  };
   342→
   343→  const onHistoryPick = (text: string) => {
   344→    setSampleId("");
   345→    setSql(text);
   346→    runAll(text, dialect);
   347→  };
   348→
   349→  const onSchemaPickTable = (table: string) => {
   350→    const text = `SELECT *\nFROM ${table}\nLIMIT 5;`;
   351→    setSampleId("");
   352→    setSql(text);
   353→    runAll(text, dialect);
   354→  };
   355→
   356→  const resetDatabase = async () => {
   357→    if (!window.confirm("Restore the sample database to its original data?")) return;
   358→    setResetting(true);
   359→    try {
   360→      await fetch("/api/reset", { method: "POST" });
   361→      // the data under the same SQL changed — bump so the row funnel and the
   362→      // FROM dot-field re-count against the fresh database
   363→      setDataVersion((v) => v + 1);
   364→      setTableCounts(null);
   365→      fetch("/api/stats")
   366→        .then((r) => r.json())
   367→        .then((j) => {
   368→          if (j?.ok) setTableCounts(j.counts as Record<string, number>);
   369→        })
   370→        .catch(() => {
   371→          /* dot-field simply skips counts */
   372→        });
   373→      // re-run current query so the user sees fresh results
   374→      await runAll(sql, dialect);
   375→    } catch {
   376→      // keep silent — the re-run will surface errors if any
   377→    } finally {
   378→      setResetting(false);
   379→    }
   380→  };
   381→
   382→  const good = plan !== null && plan.ok;
   383→  const execOk = exec !== null && exec.ok;
   384→  const planSteps = execOk ? (exec as ExecResult).plan?.length ?? 0 : 0;
   385→  const isWrite = execOk && (exec as ExecResult).changes !== null;
   386→
   387→  // ---- text ↔ diagram linking ------------------------------------------
   388→  // The live (debounced) plan drives the picture; the last run's plan is the
   389→  // fallback while the current text is unparseable.
   390→  const viewPlan = livePlan?.ok ? livePlan : plan?.ok ? plan : null;
   391→  const viewPlanSql = livePlan?.ok ? liveSql : plan?.ok ? ranSql : "";
   392→  const clauseInfo = useMemo(
   393→    () => (viewPlanSql ? findClauses(viewPlanSql) : null),
   394→    [viewPlanSql],
   395→  );
   396→  const spanFor = (kind: ClauseKind | null) =>
   397→    kind && clauseInfo?.supported ? spanOf(clauseInfo.spans, kind) : null;
   398→  const hoverSpan = spanFor(hoverKind);
   399→  const pinnedSpan = spanFor(pinnedKind);
   400→  const caretKind = clauseInfo ? clauseAtOffset(clauseInfo.spans, caret) : null;
   401→  const activeKinds: ClauseKind[] | null = pinnedKind
   402→    ? [pinnedKind]
   403→    : caretKind
   404→      ? [caretKind]
   405→      : null;
   406→  const errorRange =
   407→    liveError && liveSql === sql ? liveError.location ?? null : null;
   408→  // editor marks only make sense while the editor text matches the diagram's
   409→  // text snapshot — otherwise offsets have drifted and marks would misplace
   410→  const editorInSync = viewPlanSql === sql;
   411→  const togglePinned = useCallback((kind: ClauseKind) => {
   412→    setPinnedKind((p) => (p === kind ? null : kind));
   413→    setHoverKind(null);
   414→  }, []);
   415→
   416→  return (
   417→    <div className="flex min-h-screen flex-col bg-zinc-950 text-zinc-100">
   418→      {/* header */}
   419→      <header className="sticky top-0 z-40 border-b border-zinc-800/80 bg-zinc-950/85 backdrop-blur">
   420→        <div className="mx-auto flex w-full max-w-7xl flex-wrap items-center gap-x-4 gap-y-3 px-4 py-3">
   421→          <div className="flex items-center gap-3">
   422→            <button onClick={() => go("landing")} aria-label="SeeQL — back to home" className="transition-opacity hover:opacity-80">
   423→              <LogoFull />
   424→            </button>
   425→            <div className="hidden border-l border-zinc-800 pl-3 sm:block">
   426→              <p className="text-xs text-zinc-500">
   427→                Write a query — see it parsed, executed and explained
   428→              </p>
   429→            </div>
   430→          </div>
   431→
   432→          <div className="ml-auto flex flex-wrap items-center gap-2">
   433→            <ThemeSwitcher />
   434→            <button
   435→              onClick={() => go("learn")}
   436→              className="hidden items-center gap-1.5 rounded-md border border-zinc-800 px-3.5 py-2 text-sm text-zinc-300 transition-colors hover:bg-zinc-900 sm:flex"
   437→            >
   438→              <GraduationCap className="h-4 w-4 text-emerald-400" />
   439→              Learn
   440→            </button>
   441→            <Select value={sampleId} onValueChange={onSampleChange}>
   442→              <SelectTrigger
   443→                size="sm"
   444→                className="w-[190px] border-zinc-700 bg-zinc-900 text-xs sm:w-[220px]"
   445→                aria-label="Load a sample query"
   446→              >
   447→                <SelectValue placeholder="Sample queries" />
   448→              </SelectTrigger>
   449→              <SelectContent>
   450→                {SAMPLE_QUERIES.map((s) => (
   451→                  <SelectItem key={s.id} value={s.id} className="text-xs">
   452→                    <span className="font-medium">{s.name}</span>
   453→                    <span className="ml-2 text-zinc-500">{s.description}</span>
   454→                  </SelectItem>
   455→                ))}
   456→              </SelectContent>
   457→            </Select>
   458→
   459→            <Select value={dialect} onValueChange={onDialectChange}>
   460→              <SelectTrigger
   461→                size="sm"
   462→                className="w-[130px] border-zinc-700 bg-zinc-900 text-xs"
   463→                aria-label="SQL dialect"
   464→              >
   465→                <SelectValue />
   466→              </SelectTrigger>
   467→              <SelectContent>
   468→                {DIALECTS.map((d) => (
   469→                  <SelectItem key={d} value={d} className="font-mono text-xs">
   470→                    {d}
   471→                  </SelectItem>
   472→                ))}
   473→              </SelectContent>
   474→            </Select>
   475→
   476→            <Button
   477→              variant="ghost"
   478→              size="sm"
   479→              onClick={shareLink}
   480→              className="h-8 gap-1.5 px-2.5 text-xs text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200"
   481→              aria-label="Copy a link to this query"
   482→            >
   483→              {copiedLink ? (
   484→                <Check className="h-4 w-4 text-emerald-400" />
   485→              ) : (
   486→                <Link2 className="h-4 w-4" />
   487→              )}
   488→              {copiedLink ? "Copied" : "Share"}
   489→            </Button>
   490→
   491→            <Button
   492→              onClick={() => runAll(sql, dialect)}
   493→              disabled={running}
   494→              className="h-8 gap-1.5 bg-emerald-500 font-medium text-zinc-950 hover:bg-emerald-400"
   495→            >
   496→              {running ? (
   497→                <Loader2 className="h-4 w-4 animate-spin" />
   498→              ) : (
   499→                <Play className="h-4 w-4" />
   500→              )}
   501→              Run
   502→            </Button>
   503→          </div>
   504→        </div>
   505→      </header>
   506→
   507→      {/* main */}
   508→      <main className="mx-auto w-full max-w-7xl flex-1 px-4 py-5">
   509→        <div className="grid items-start gap-5 lg:grid-cols-[minmax(360px,2fr)_minmax(0,3fr)]">
   510→          {/* editor panel */}
   511→          <section aria-label="SQL editor" className="flex flex-col gap-2">
   512→            <div className="flex items-center justify-between px-1">
   513→              <h2 className="text-xs font-semibold tracking-widest text-zinc-400 uppercase">
   514→                Query
   515→              </h2>
   516→              <div className="flex items-center gap-1">
   517→                {history.length > 0 && (
   518→                  <Select onValueChange={onHistoryPick}>
   519→                    <SelectTrigger
   520→                      size="sm"
   521→                      className="h-7 gap-1 border-none bg-transparent px-2 text-xs text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200 focus:ring-0"
   522→                      aria-label="Query history"
   523→                    >
   524→                      <span className="flex items-center gap-1.5">
   525→                        <History className="h-3.5 w-3.5" />
   526→                        History
   527→                      </span>
   528→                    </SelectTrigger>
   529→                    <SelectContent>
   530→                      {history.map((h, i) => (
   531→                        <SelectItem key={i} value={h} className="max-w-[320px] font-mono text-xs">
   532→                          <span className="block truncate">{h.split("\n")[0]}</span>
   533→                        </SelectItem>
   534→                      ))}
   535→                    </SelectContent>
   536→                  </Select>
   537→                )}
   538→                <Button
   539→                  variant="ghost"
   540→                  size="sm"
   541→                  onClick={() => setSql("")}
   542→                  className="h-7 gap-1 px-2 text-xs text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200"
   543→                >
   544→                  <Eraser className="h-3.5 w-3.5" />
   545→                  Clear
   546→                </Button>

   887→/**
   888→ * App shell — single exposed route, three views:
   889→ *   /            → landing
   890→ *   /#playground → SQL playground (the original visualizer)
   891→ *   /#learn      → the learning app
   892→ * Hash syncs with browser history so back/forward work.
   893→ */
   894→function subscribeHash(cb: () => void) {
   895→  window.addEventListener("hashchange", cb);
   896→  return () => window.removeEventListener("hashchange", cb);
   897→}
   898→
   899→export default function Home() {
   900→  // The hash IS the router — subscribe to it as an external system.
   901→  // useSyncExternalStore renders the server snapshot during hydration
   902→  // (landing) and re-renders with the real hash right after — no
   903→  // hydration mismatch, no setState-in-effect.
   904→  const hash = useSyncExternalStore(
   905→    subscribeHash,
   906→    () => window.location.hash,
   907→    () => "",
   908→  );
   909→  // The route is the part before "?" — a shared query rides along as
   910→  // #playground?q=<encoded sql>&d=<dialect> and never breaks deep links.
   911→  const qMark = hash.indexOf("?");
   912→  const route = qMark === -1 ? hash : hash.slice(0, qMark);
   913→  const mode: Mode =
   914→    route === "#playground" ? "playground" : route === "#learn" ? "learn" : "landing";
   915→
   916→  let sharedSql: string | null = null;
   917→  let sharedDialect: DialectName | undefined;
   918→  if (mode === "playground" && qMark !== -1) {
   919→    const params = new URLSearchParams(hash.slice(qMark + 1));
   920→    const q = params.get("q");
   921→    if (q && q.trim()) sharedSql = q;
   922→    const d = params.get("d");
   923→    if (d && (DIALECTS as readonly string[]).includes(d)) sharedDialect = d as DialectName;
   924→  }
   925→
   926→  const go = useCallback((m: Mode) => {

   930→
   931→  if (mode === "landing") {
   932→    return <Landing onStart={() => go("learn")} onPlayground={() => go("playground")} />;
   933→  }
   934→  if (mode === "learn") {
   935→    return (

   946→                className="hidden rounded-md border border-zinc-800 px-3.5 py-2 text-sm text-zinc-300 transition-colors hover:bg-zinc-900 sm:block"
