   117→function useBadgeWatcher() {
   118→  const xpEvents = useProgress((s) => s.xpEvents);
   119→  const mcqAttempts = useProgress((s) => s.mcqAttempts);
   120→  const badges = useProgress((s) => s.badges);
   121→  const xpByDay = useProgress((s) => s.xpByDay);
   122→  const dailyGoal = useProgress((s) => s.dailyGoal);
   123→  const prevLevel = useRef(-1);
   124→  const ready = useRef(false);
   125→
   126→  const xp = useMemo(() => totalXp({ xpEvents }), [xpEvents]);
   127→  const level = levelFor(xp);
   128→
   129→  const ctx = useMemo(() => {
   130→    const problemsSolved = Object.keys(xpEvents).filter((k) => k.startsWith("solve:")).length;
   131→    const mcqsCorrect = Object.values(mcqAttempts).filter((a) => a.correct).length;
   132→    const lessonsDone = Object.keys(xpEvents).filter((k) => k.startsWith("lesson:")).length;
   133→    const perfectQuizzes = LESSONS.filter((l) =>
   134→      l.mcqs.every((m) => mcqAttempts[m.id]?.firstTry),
   135→    ).length;
   136→    return { lessonsDone, problemsSolved, mcqsCorrect, perfectQuizzes };
   137→  }, [xpEvents, mcqAttempts]);
   138→
   139→  const streak = useMemo(() => currentStreak({ xpByDay, dailyGoal }), [xpByDay, dailyGoal]);
   140→
   141→  useEffect(() => {
   142→    if (!ready.current) {
   143→      ready.current = true;
   144→      prevLevel.current = level.index;
   145→      return;
   146→    }
   147→    const store = useProgress.getState();
   148→    for (const b of BADGES) {
   149→      if (store.badges[b.id] === undefined && b.test(ctx, xp, streak)) {
   150→        if (store.unlockBadge(b.id)) {
   151→          celebrate({ kind: "badge", title: `Badge · ${b.title}`, sub: b.desc });
   152→        }
   153→      }
   154→    }
   155→    if (level.index > prevLevel.current) {
   156→      celebrate({ kind: "level", title: `Level up — ${level.title}`, sub: `Rank ${level.index + 1} of 6` });
   157→      prevLevel.current = level.index;
   158→    }
   159→  }, [ctx, xp, streak, level]);
   160→
   161→  return { xp, level, streak, ctx };
   162→}
