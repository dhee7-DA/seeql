    44→function loadOpenModules(): number[] | null {
    45→  try {
    46→    const raw = localStorage.getItem(MODULES_OPEN_KEY);
    47→    if (!raw) return null;
    48→    const arr = JSON.parse(raw);
    49→    return Array.isArray(arr) ? arr.filter((n) => typeof n === "number") : null;
    50→  } catch {
    51→    return null;
    52→  }
    53→}

    79→function useBadgeWatcher() {
    80→  const xpEvents = useProgress((s) => s.xpEvents);
    81→  const mcqAttempts = useProgress((s) => s.mcqAttempts);
    82→  const badges = useProgress((s) => s.badges);
    83→  const xpByDay = useProgress((s) => s.xpByDay);
    84→  const dailyGoal = useProgress((s) => s.dailyGoal);
    85→  const prevLevel = useRef(-1);
    86→  const ready = useRef(false);
    87→
    88→  const xp = useMemo(() => totalXp({ xpEvents }), [xpEvents]);
    89→  const level = levelFor(xp);
    90→
    91→  const ctx = useMemo(() => {
    92→    const problemsSolved = Object.keys(xpEvents).filter((k) => k.startsWith("solve:")).length;
    93→    const mcqsCorrect = Object.values(mcqAttempts).filter((a) => a.correct).length;
    94→    const lessonsDone = Object.keys(xpEvents).filter((k) => k.startsWith("lesson:")).length;
    95→    const perfectQuizzes = LESSONS.filter((l) =>
    96→      l.mcqs.every((m) => mcqAttempts[m.id]?.firstTry),
    97→    ).length;
    98→    return { lessonsDone, problemsSolved, mcqsCorrect, perfectQuizzes };
    99→  }, [xpEvents, mcqAttempts]);
   100→
   101→  const streak = useMemo(() => currentStreak({ xpByDay, dailyGoal }), [xpByDay, dailyGoal]);
   102→
   103→  useEffect(() => {
   104→    if (!ready.current) {
   105→      ready.current = true;
   106→      prevLevel.current = level.index;
   107→      return;
   108→    }
   109→    const store = useProgress.getState();
   110→    for (const b of BADGES) {
   111→      if (store.badges[b.id] === undefined && b.test(ctx, xp, streak)) {
   112→        if (store.unlockBadge(b.id)) {
   113→          celebrate({ kind: "badge", title: `Badge · ${b.title}`, sub: b.desc });
   114→        }
   115→      }
   116→    }
   117→    if (level.index > prevLevel.current) {
   118→      celebrate({ kind: "level", title: `Level up — ${level.title}`, sub: `Rank ${level.index + 1} of 6` });
   119→      prevLevel.current = level.index;
   120→    }
   121→  }, [ctx, xp, streak, level]);
   122→
   123→  return { xp, level, streak, ctx };
   124→}

   131→        <span className="tnum font-mono text-xs text-emerald-300">{xp.toLocaleString()} XP</span>

   208→              const done = xpEvents[`lesson:${l.id}`] !== undefined;
   209→              return (
   210→                <button
   211→                  key={l.id}
   212→                  onClick={() => onOpenLesson(l.id)}
   213→                  className="group/lesson flex items-center gap-2.5 rounded-md px-2 py-1.5 text-left transition-colors hover:bg-zinc-900"

   233→            <button
   234→              onClick={() => onOpenLesson(nextInModule.id)}

   238→            </button>

   241→      </Collapse>
   242→    </div>
   243→  );
   244→}
