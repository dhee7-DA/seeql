    42→const MOD = COURSE_MODULES[0];
    43→
    44→function lessonDone(
    45→  progress: ReturnType<typeof useProgress.getState>,
    46→  lessonId: string,
    47→  mcqIds: string[],
    48→  problemIds: string[],
    49→): boolean {
    50→  if (manuallyCompleted(progress, lessonId)) return true;
    51→  const allMcq = mcqIds.length > 0 && mcqIds.every((id) => mcqPassed(progress, id));
    52→  const allProblems =
    53→    problemIds.length === 0 || problemIds.every((id) => problemSolved(progress, id));
    54→  return allMcq && allProblems;
    55→}

   487→/* ---------------- overview (module home) ---------------- */
   488→
   489→function Overview({
   490→  doneMap,
   491→  onOpen,
   492→}: {
   493→  doneMap: Record<string, boolean>;
   494→  onOpen: (ref: PracticeRef) => void;
   495→  onTryInPlayground: (sql: string) => void;
   496→}) {
   497→  const lessonCount = MOD.lessons.length;
   498→  const mcqCount = MOD.lessons.reduce((n, l) => n + l.mcqs.length, 0);
   499→  const diagramCount = MOD.lessons.reduce(
   500→    (n, l) => n + l.theory.filter((b) => b.kind === "diagram").length,
   501→    0,
   502→  );
   503→  const problemCount =
   504→    MOD.lessons.reduce((n, l) => n + l.problems.length, 0) +
   505→    MOD.caseStudies.reduce((n, c) => n + c.parts.length, 0);
   506→
   507→  return (
   508→    <div className="space-y-5">
   509→      {/* hero */}
   510→      <section className="overflow-hidden rounded-2xl border border-zinc-800 bg-gradient-to-br from-zinc-900 via-zinc-900/80 to-emerald-950/30 p-6">
   511→        <div className="flex items-center gap-2 text-[11px] font-semibold tracking-widest text-emerald-400 uppercase">
   512→          <GraduationCap className="h-4 w-4" />
   513→          Module 1 · {MOD.level}
   514→        </div>
   515→        <h1 className="mt-2.5 text-2xl font-bold tracking-tight text-zinc-50 sm:text-3xl">
   516→          {MOD.title}
   517→        </h1>
   518→        <p className="mt-1 text-sm text-zinc-400">{MOD.tagline}</p>
   519→        <p className="mt-3 max-w-2xl text-sm leading-relaxed text-zinc-300">
   520→          {MOD.description}
   521→        </p>
   522→        <div className="mt-4 flex flex-wrap items-center gap-2">
   523→          <span className="rounded-full border border-zinc-700 bg-zinc-900/70 px-3 py-1 text-xs text-zinc-300">
   524→            {lessonCount} lessons
   525→          </span>
   526→          <span className="rounded-full border border-zinc-700 bg-zinc-900/70 px-3 py-1 text-xs text-zinc-300">
