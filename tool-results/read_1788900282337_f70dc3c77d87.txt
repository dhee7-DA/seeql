    82→    82→function useBadgeWatcher() {
    83→    83→  const xpEvents = useProgress((s) => s.xpEvents);
    84→    84→  const mcqAttempts = useProgress((s) => s.mcqAttempts);
    85→    85→  const badges = useProgress((s) => s.badges);
    86→    86→  const xpByDay = useProgress((s) => s.xpByDay);
    87→    87→  const dailyGoal = useProgress((s) => s.dailyGoal);
    88→    88→  const prevLevel = useRef(-1);
    89→    89→  const ready = useRef(false);
    90→    90→
    91→    91→  const xp = useMemo(() => totalXp({ xpEvents }), [xpEvents]);
    92→    92→  const level = levelFor(xp);
    93→    93→
    94→    94→  const ctx = useMemo(() => {
    95→    95→    const problemsSolved = Object.keys(xpEvents).filter((k) => k.startsWith("solve:")).length;
    96→    96→    const mcqsCorrect = Object.values(mcqAttempts).filter((a) => a.correct).length;
    97→    97→    const lessonsDone = Object.keys(xpEvents).filter((k) => k.startsWith("lesson:")).length;
    98→    98→    const perfectQuizzes = LESSONS.filter((l) =>
    99→    99→      l.mcqs.every((m) => mcqAttempts[m.id]?.firstTry),
   100→   100→    ).length;
   101→   101→    return { lessonsDone, problemsSolved, mcqsCorrect, perfectQuizzes };
   102→   102→  }, [xpEvents, mcqAttempts]);
   103→   103→
   104→   104→  const streak = useMemo(() => currentStreak({ xpByDay, dailyGoal }), [xpByDay, dailyGoal]);
   105→   105→
   106→   106→  useEffect(() => {
   107→   107→    if (!ready.current) {
   108→   108→      ready.current = true;
   109→   109→      prevLevel.current = level.index;
   110→   110→      return;
   111→   111→    }
   112→   112→    const store = useProgress.getState();
   113→   113→    for (const b of BADGES) {
   114→   114→      if (store.badges[b.id] === undefined && b.test(ctx, xp, streak)) {
   115→   115→        if (store.unlockBadge(b.id)) {
   116→   116→          celebrate({ kind: "badge", title: `Badge · ${b.title}`, sub: b.desc });
   117→   117→        }
   118→   118→      }
   119→   119→    }
   120→   120→    if (level.index > prevLevel.current) {
   121→   121→      celebrate({ kind: "level", title: `Level up — ${level.title}`, sub: `Rank ${level.index + 1} of 6` });
   122→   122→      prevLevel.current = level.index;
   123→   123→    }
   124→   124→  }, [ctx, xp, streak, level]);
   125→   125→
   126→   126→  return { xp, level, streak, ctx };
   127→   127→}
