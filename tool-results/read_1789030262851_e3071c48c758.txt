   140→async function runInline(sql: string, sandbox: boolean): Promise<SingleResult> {
   141→  const r = await fetch("/api/query", {
   142→    method: "POST",
   143→    headers: { "Content-Type": "application/json" },
   144→    body: JSON.stringify({ sql, sandbox }),
   145→  });
   146→  return (await r.json()) as SingleResult;
   147→}
   148→
   149→async function runBatch(sqls: string[]): Promise<{ ok: boolean; results?: BatchItemResult[]; error?: string }> {
   150→  const r = await fetch("/api/query", {
   151→    method: "POST",
   152→    headers: { "Content-Type": "application/json" },
   153→    body: JSON.stringify({ sqls }),
   154→  });
   155→  return (await r.json()) as { ok: boolean; results?: BatchItemResult[]; error?: string };
   156→}

   158→function CodeBlock({ sql, caption, runnable = true }: { sql: string; caption?: string; runnable?: boolean }) {
   159→  const [result, setResult] = useState<SingleResult | null>(null);
   160→  const [batch, setBatch] = useState<BatchItemResult[] | null>(null);
   161→  const [batchError, setBatchError] = useState<string | null>(null);
   162→  const [busy, setBusy] = useState(false);
   163→  // Multi-statement blocks run as one sandboxed sequence — the statements see
   164→  // each other's effects and nothing persists.
   165→  const statements = useMemo(() => splitStatements(sql), [sql]);
   166→  const isMulti = statements.length > 1;
   167→  const isRead = looksLikeRead(sql);
   168→  const canRun = runnable && sql.trim().length > 0;
   169→
   170→  async function run() {
   171→    if (busy) return;
   172→    setBusy(true);
   173→    setBatchError(null);
   174→    try {
   175→      if (isMulti) {
   176→        setResult(null);
   177→        setBatch(null);
   178→        const r = await runBatch(statements);
   179→        if (r.ok && r.results) setBatch(r.results);
   180→        else setBatchError(r.error ?? "The sequence could not be run.");
   181→      } else {
   182→        setBatch(null);
   183→        // Writes and DDL run sandboxed: the statement really executes,
   184→        // then the whole thing is rolled back — the store stays untouched.
   185→        setResult(await runInline(sql, !isRead));
   186→      }
   187→    } finally {
   188→      setBusy(false);
   189→    }
   190→  }
   191→
   192→  const affected = result?.ok && (result.rows ?? []).length === 0 && result.changes != null;
   193→  const label = isMulti
   194→    ? `Run all · ${statements.length} steps`
   195→    : isRead ? "Run here" : "Run sandboxed";
   196→
   197→  return (
   198→    <div className="my-4 overflow-hidden rounded-lg border border-zinc-800 bg-zinc-950">
   199→      {(caption || canRun) && (
   200→        <div className="flex items-center justify-between border-b border-zinc-800 px-3.5 py-2">
   201→          <span className="text-[11px] font-medium uppercase tracking-wider text-zinc-500">{caption ?? "SQL"}</span>
   202→          {canRun && (
   203→            <button
   204→              onClick={run}
   205→              disabled={busy}
   206→              className="rounded border border-emerald-400/40 px-2 py-0.5 font-mono text-[11px] text-emerald-300 hover:bg-emerald-400/10 disabled:opacity-50"
   207→            >
   208→              {busy ? "running…" : label}
   209→            </button>
   210→          )}
   211→        </div>
   212→      )}
   213→      <pre className="overflow-x-auto px-3.5 py-3 font-mono text-xs leading-relaxed text-zinc-300 thin-scrollbar"><SqlNodes text={sql} /></pre>
   214→      {result && (
   215→        <div className="border-t border-zinc-800 p-3">
   216→          {result.ok ? (
   217→            affected ? (
   218→              <p className="font-mono text-xs text-zinc-400">
   219→                <span className="text-emerald-300">{result.changes}</span> row(s) affected — sandboxed run, rolled back.
   220→              </p>
   221→            ) : (
   222→              <ResultGrid data={{ columns: result.columns ?? [], rows: (result.rows ?? []) as never[] }} maxHeight="max-h-56" />
   223→            )
   224→          ) : (
   225→            <p className="font-mono text-xs text-rose-300">{result.error}</p>
   226→          )}
   227→        </div>
   228→      )}
   229→      {batch && (
   230→        <div className="divide-y divide-zinc-800/70 border-t border-zinc-800">
   231→          {batch.map((item, i) => (
   232→            <div key={i} className="p-3">
   233→              <div className="mb-2 flex items-center gap-2">
   234→                <span className="flex h-4.5 w-4.5 items-center justify-center rounded border border-zinc-700 bg-zinc-900 font-mono text-[10px] text-zinc-400">
   235→                  {i + 1}
   236→                </span>
   237→                <span className="font-mono text-[10px] uppercase tracking-wider text-zinc-500">
   238→                  {item.statementType ?? "SQL"}
   239→                </span>
   240→                {item.ok ? (
   241→                  <span className="font-mono text-[10px] text-emerald-300/90">
   242→                    {item.rows && item.rows.length > 0
   243→                      ? `${item.rows.length} row${item.rows.length === 1 ? "" : "s"}`
   244→                      : item.changes != null ? `${item.changes} affected` : "ok"}
   245→                  </span>
   246→                ) : (
   247→                  <span className="font-mono text-[10px] text-rose-300">error</span>
   248→                )}
   249→              </div>
   250→              {item.ok ? (
   251→                item.rows && item.rows.length > 0 ? (
   252→                  <ResultGrid data={{ columns: item.columns ?? [], rows: item.rows as never[] }} maxHeight="max-h-44" />
   253→                ) : (
   254→                  <p className="font-mono text-xs text-zinc-400">
   255→                    <span className="text-emerald-300">{item.changes}</span> row(s) affected — sandboxed run, rolled back.
   256→                  </p>
   257→                )
   258→              ) : (
   259→                <p className="font-mono text-xs text-rose-300">{item.error}</p>
   260→              )}
   261→            </div>
   262→          ))}
   263→        </div>
   264→      )}
   265→      {batchError && (
   266→        <div className="border-t border-zinc-800 p-3">
   267→          <p className="font-mono text-xs text-rose-300">{batchError}</p>
   268→        </div>
   269→      )}
   270→    </div>
   271→  );
   272→}

   439→export function LessonView({ lesson, onBonusChecked }: { lesson: Lesson; onBonusChecked?: () => void }) {
   440→  const [tab, setTab] = useState<Tab>("learn");
   441→  const store = useProgress();
   442→  const bonusRef = useRef(false);
   443→
   444→  const quizStats = useMemo(() => {
   445→    const first = lesson.mcqs.every((m) => store.mcqAttempts[m.id]?.firstTry);
   446→    const allChecked = lesson.mcqs.every((m) => store.mcqAttempts[m.id] !== undefined);
   447→    return { allChecked, first, allCorrect: lesson.mcqs.every((m) => store.mcqAttempts[m.id]?.correct) };
   448→  }, [lesson, store.mcqAttempts]);
   449→
   450→  // Lessons whose module owns a drill bank (today: Module 1) swap the legacy
   451→  // 12-question quiz for their keyword drill sets — the real MCQ layer.
   452→  const drill = useMemo(() => {
   453→    const sets = drillSetsForLesson(lesson.id);
   454→    if (!sets.length) return null;
   455→    const questions = sets.flatMap((s) => s.questions);
   456→    const ids = new Set(questions.map((q) => q.id));
   457→    const tried = Object.keys(store.mcqAttempts).filter((id) => ids.has(id)).length;
   458→    const right = questions.filter((q) => store.mcqAttempts[q.id]?.correct).length;
   459→    const total = ids.size;
   460→    // a lesson round = one 10-question drill across the lesson's sets
   461→    return { sets, total, tried, right, done: tried >= Math.min(10, total) };
   462→  }, [lesson.id, store.mcqAttempts]);
   463→  const drillKeywords = useMemo(() => drill?.sets.map((s) => s.keyword), [drill]);
   464→  const modNum = moduleOfLesson(lesson.id).num;
   465→
   466→  // same for the case deck: lessons whose module owns one get a Cases tab
   467→  const caseBank = useMemo(() => casesForLesson(lesson.num), [lesson.num]);
   468→
   469→  const solvedCount = lesson.problems.filter((p) => store.xpEvents[`solve:${p.id}`] !== undefined).length;
   470→  const quizDone = drill ? drill.done : quizStats.allCorrect;
   471→  const complete = quizDone && solvedCount === lesson.problems.length;
   472→

   481→          return triedNow >= Math.min(10, ids.size);
   482→        })()
   483→      : lesson.mcqs.every((m) => s.mcqAttempts[m.id]?.correct);
   484→    const allSolved = lesson.problems.every((p) => s.xpEvents[`solve:${p.id}`] !== undefined);
   485→    if (quizDoneNow && allSolved && s.awardXp(`lesson:${lesson.id}`, 50)) {
   486→      celebrate({ kind: "level", title: `Lesson ${lesson.num} complete · +50 XP`, sub: lesson.title });
   487→      s.markLessonDone(lesson.id);
   488→      bonusRef.current = true;
   489→      onBonusChecked?.();
   490→    }
   491→  }
   492→

   502→      ? [{ id: "cases" as Tab, label: "Cases", icon: FileStack, note: `${caseBank.deck.length} cases` }]

   506→      label: "Practice",
   507→      icon: TerminalSquare,
   508→      note: `${solvedCount}/${lesson.problems.length} solved`,
   509→    },
   510→  ];
   511→
   512→  return (

   514→      <div className="flex items-center gap-3">

   526→        {tabs.map((t) => (
   527→          <button
   528→            key={t.id}
   529→            onClick={() => setTab(t.id)}
   530→            className={cn(
   531→              "flex flex-1 items-center justify-center gap-2 rounded-md px-3 py-2 text-xs font-medium transition-colors sm:flex-none sm:px-4",
   532→              tab === t.id
   533→                ? "bg-zinc-800 text-zinc-50 shadow-[inset_0_1px_0_0_rgb(255_255_255/0.06)]"
   534→                : "text-zinc-500 hover:text-zinc-300",
   535→            )}
   536→          >
   537→            <t.icon className={cn("h-3.5 w-3.5", tab === t.id && "text-emerald-300")} />
   538→            {t.label}
   539→            <span className="tnum hidden text-[10px] text-zinc-500 sm:inline">· {t.note}</span>
   540→          </button>
   541→        ))}
   542→      </div>
   543→
   544→      {tab === "learn" && (
   545→        <div className="mt-5 max-w-3xl">
   546→          {lesson.blocks.map((b, i) => {
   547→            switch (b.kind) {
   548→              case "p":
   549→                return <Rich key={i} text={b.text} className="text-[15px] leading-7 text-zinc-300" />;
   550→              case "h":
   551→                return (
   552→                  <div key={i} className="mb-3 mt-7 flex items-center gap-2.5">
   553→                    <span className="h-3.5 w-1 rounded-full bg-emerald-400" />

   644→          {lesson.takeaway && (

   681→              onClick={() => setTab("quiz")}
   682→              className="btn-glow rounded-md bg-emerald-400 px-3.5 py-1.5 text-xs font-medium text-[#052e1c] transition-colors hover:bg-emerald-300"

   695→              : `This lesson's real MCQ bank: ${drill.total} questions across ${drill.sets.length === 1 ? "one keyword set" : `${drill.sets.length} keyword sets`}, drilled from six angles. Rounds of 10, unseen first — +5 XP per answer, +3 bonus on first tries.`}

   735→            {solvedCount}/{lesson.problems.length} solved so far.
