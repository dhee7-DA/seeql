   112→function useBadgeWatcher() {
   113→  const xpEvents = useProgress((s) => s.xpEvents);
   114→  const mcqAttempts = useProgress((s) => s.mcqAttempts);
   115→  const badges = useProgress((s) => s.badges);
   116→  const xpByDay = useProgress((s) => s.xpByDay);
   117→  const dailyGoal = useProgress((s) => s.dailyGoal);
   118→  const prevLevel = useRef(-1);
   119→  const ready = useRef(false);
   120→
   121→  const xp = useMemo(() => totalXp({ xpEvents }), [xpEvents]);
   122→  const level = levelFor(xp);
   123→
   124→  const ctx = useMemo(() => {
   125→    const problemsSolved = Object.keys(xpEvents).filter((k) => k.startsWith("solve:")).length;
   126→    const mcqsCorrect = Object.values(mcqAttempts).filter((a) => a.correct).length;
   127→    const lessonsDone = Object.keys(xpEvents).filter((k) => k.startsWith("lesson:")).length;
   128→    const perfectQuizzes = LESSONS.filter((l) =>
   129→      l.mcqs.every((m) => mcqAttempts[m.id]?.firstTry),
   130→    ).length;
   131→    return { lessonsDone, problemsSolved, mcqsCorrect, perfectQuizzes };
   132→  }, [xpEvents, mcqAttempts]);
   133→
   134→  const streak = useMemo(() => currentStreak({ xpByDay, dailyGoal }), [xpByDay, dailyGoal]);
   135→
   136→  useEffect(() => {
   137→    if (!ready.current) {
   138→      ready.current = true;
   139→      prevLevel.current = level.index;
   140→      return;
   141→    }
   142→    const store = useProgress.getState();
   143→    for (const b of BADGES) {
   144→      if (store.badges[b.id] === undefined && b.test(ctx, xp, streak)) {
   145→        if (store.unlockBadge(b.id)) {
   146→          celebrate({ kind: "badge", title: `Badge · ${b.title}`, sub: b.desc });
   147→        }
   148→      }
   149→    }
   150→    if (level.index > prevLevel.current) {
   151→      celebrate({ kind: "level", title: `Level up — ${level.title}`, sub: `Rank ${level.index + 1} of 6` });
   152→      prevLevel.current = level.index;
   153→    }
   154→  }, [ctx, xp, streak, level]);
   155→
   156→  return { xp, level, streak, ctx };
   157→}

   303→/** Course overview — the default view: hero ("where was I"), the course map,

   306→  xp,
   307→  level,
   308→  streak,
   309→  ctx,
   310→  openLesson,
   311→}: {
   312→  xp: number;
   313→  level: ReturnType<typeof levelFor>;
   314→  streak: number;
   315→  ctx: ReturnType<typeof useBadgeWatcher>["ctx"];
   316→  openLesson: (id: string) => void;
   317→}) {
   318→  const xpEvents = useProgress((s) => s.xpEvents);
   319→  const badgesMap = useProgress((s) => s.badges);
   320→  const [openOverview, setOpenOverview] = useState<Set<number>>(() => new Set());

   323→    LESSONS.find((l) => xpEvents[`lesson:${l.id}`] === undefined) ?? LESSONS[LESSONS.length - 1];

   325→  const totalProblems = LESSONS.reduce((a, l) => a + l.problems.length, 0);

   328→    setOpenOverview((s) => {
   329→      const n = new Set(s);
   330→      if (n.has(num)) n.delete(num);
   331→      else n.add(num);
   332→      return n;
   333→    });
   334→
   335→  return (
   336→    <div className="bg-ambient max-w-4xl">
   337→      {/* Hero — "where was I" + the one tap that matters, before any scrolling */}
   338→      <Panel pad={5}>
   339→        <p className="font-mono text-xs text-zinc-500">{MODULES.length} MODULES · BEGINNER → EXPERT</p>
   340→        <div className="mt-2 flex flex-wrap items-end justify-between gap-x-6 gap-y-3">
   341→          <h1 className="font-display text-3xl font-semibold tracking-[-0.02em] text-zinc-50">The SeeQL course</h1>
   342→          <div className="flex flex-wrap items-center gap-2">
   343→            <Chip k="Rank" v={level.title} />
   344→            <Chip k="Total XP" v={xp.toLocaleString()} />
   345→            <Chip k="Streak" v={streak > 0 ? `${streak}d` : "—"} />
   346→            <Chip k="Badges" v={`${Object.keys(badgesMap).length}/${BADGES.length}`} />
   347→          </div>
   348→        </div>
   349→        <p className="mt-3 max-w-2xl text-[15px] leading-7 text-zinc-400">
   350→          {MODULES.length} modules, one real storefront database. Read a concept, prove it in a quiz, then solve
   351→          graded live queries — {totalMcqs} quiz questions and {totalProblems} graded queries in total.
   352→        </p>
   353→
   354→        <div className="mt-5 flex flex-wrap items-center gap-2.5">
   355→          <Btn variant="primary" onClick={() => openLesson(nextLesson.id)}>
   356→            {ctx.lessonsDone > 0 ? `Continue — lesson ${nextLesson.num} · ${nextLesson.title} →` : "Start learning →"}
   357→          </Btn>
   358→          <span className="tnum text-[11px] text-zinc-500">
   359→            {ctx.problemsSolved} queries solved · {ctx.mcqsCorrect} quiz answers

   422→/** The active view's body — one switch, fed by the shared helpers and the
   423→    gamification values (the overview needs them for its hero). */
   424→function ViewBody({
   425→  view,
   426→  openLesson,
   427→  openPerf,
   428→  xp,
   429→  level,
   430→  streak,
   431→  ctx,
   432→}: {
   433→  view: View;
   434→  openLesson: (id: string) => void;
   435→  openPerf: (sql: string) => void;
   436→  xp: number;
   437→  level: ReturnType<typeof levelFor>;
   438→  streak: number;
   439→  ctx: ReturnType<typeof useBadgeWatcher>["ctx"];
   440→}) {
   441→  switch (view.kind) {
   442→    case "lesson": {
   443→      const lesson = LESSONS.find((l) => l.id === view.id) ?? null;
   444→      return lesson ? <LessonView key={lesson.id} lesson={lesson} /> : null;
   445→    }
   446→    case "seeq":
   447→      return <SeeqView />;
   448→    case "explorer":
   449→      return <DataExplorer />;
   450→    case "library":
   451→      return <DiagramLibrary onOpenLesson={openLesson} />;
   452→    case "drills":
   453→      return <DrillYard />;
   454→    case "mixed":
   455→      return <MixedQuiz />;
   456→    case "cases":
   457→      return <CaseFiles />;
   458→    case "predict":
   459→      return <PredictView />;
   460→    case "clinic":
   461→      return <ErrorClinic />;
   462→    case "golf":
   463→      return <QueryGolf onOpenPerf={openPerf} />;
   464→    case "perf":
   465→      return <PerfLab initialSql={view.sql} initialTab={view.tab} onOpenLesson={openLesson} />;
   466→    case "interview":
   467→      return <InterviewTrack />;
   468→    case "studio":
   469→      return <SchemaStudio />;
   470→    case "explain":
   471→      return <ExplainBack />;
   472→    case "mastery":
   473→      return <MasteryMap onOpenLesson={openLesson} />;
   474→    case "sheet":
   475→      return <CheatSheet />;
   476→    case "mentor":
   477→      return <MentorChat />;
   478→    default:
   479→      return <Overview xp={xp} level={level} streak={streak} ctx={ctx} openLesson={openLesson} />;
   480→  }

   483→export function LearnApp() {
   484→  const [view, setView] = useState<View>({ kind: "overview" });
   485→
   486→  // The top header's SeeQ button (page shell) navigates here by event — the
   487→  // view state is private to this component. Listener callbacks are event
   488→  // handlers, so the setState inside is compliant.
   489→  useEffect(() => {
   490→    const onNav = (e: Event) => {
   491→      const kind = (e as CustomEvent<{ kind?: View["kind"] }>).detail?.kind;
   492→      if (!kind) return;
   493→      setView({ kind } as View);
   494→      window.scrollTo({ top: 0 });
   495→    };
   496→    window.addEventListener("seeql:nav", onNav);
   497→    return () => window.removeEventListener("seeql:nav", onNav);
   498→  }, []);
   499→  const { xp, level, streak, ctx } = useBadgeWatcher();
   500→  const xpEvents = useProgress((s) => s.xpEvents);
   501→  const resetAll = useProgress((s) => s.resetAll);
   502→  const [confirmReset, setConfirmReset] = useState(false);
   503→
   504→  // Collapsible modules — persisted; the sidebar defaults to the "continue"
   505→  // module. (The overview course map keeps its own reveal state.)
   506→  const nextLesson = LESSONS.find((l) => xpEvents[`lesson:${l.id}`] === undefined) ?? LESSONS[LESSONS.length - 1];
   507→  const continueModuleNum = moduleOf(nextLesson.id)?.num ?? 1;
   508→  const [openModules, setOpenModules] = useState<Set<number>>(() => {
   509→    const saved = loadOpenModules();
   510→    return new Set(saved ?? [continueModuleNum]);
   511→  });
   512→
   513→  useEffect(() => {
   514→    try {
   515→      localStorage.setItem(MODULES_OPEN_KEY, JSON.stringify([...openModules]));
   516→    } catch {
   517→      /* private mode — ignore */
   518→    }
   519→  }, [openModules]);
   520→
   521→  const toggleModule = (num: number) =>
   522→    setOpenModules((s) => {
   523→      const n = new Set(s);
   524→      if (n.has(num)) n.delete(num);
   525→      else n.add(num);
   526→      return n;
   527→    });
   528→
   529→  const openLesson = (id: string) => {
   530→    const mod = moduleOf(id);
   531→    if (mod) {
   532→      // Auto-reveal the module that owns the lesson being opened.
   533→      setOpenModules((s) => (s.has(mod.num) ? s : new Set(s).add(mod.num)));
   534→    }
   535→    setView({ kind: "lesson", id });
   536→  };
   537→
   538→  const openPerf = (sql: string) => setView({ kind: "perf", sql, tab: "workbench" });
   539→  const doneCount = LESSONS.filter((l) => xpEvents[`lesson:${l.id}`] !== undefined).length;
   540→  const totalProblems = LESSONS.reduce((a, l) => a + l.problems.length, 0);
   541→  const totalMcqs = LESSONS.reduce((a, l) => a + l.mcqs.length, 0);
   542→  const allOpen = openModules.size === MODULES.length;
   543→  const reduced = useReducedMotion() ?? false;
   544→
   545→  return (
   546→    <div className="mx-auto flex w-full max-w-7xl flex-1 flex-col gap-6 px-4 py-6 lg:flex-row">
   547→      <CelebrationHost />
   548→      {/* SeeqBuddy now mounts at the app root (page.tsx) so the buddy is on
   549→          every view — landing, playground, learn — and survives switches. */}
   550→
   551→      {/* sidebar — below the content on mobile, sticky rail on desktop */}
   552→      {/* Definite height (not max-height): max-h-full on the inner panel is a
   553→          percentage, which only resolves against a definite parent height —
   554→          with max-height alone the panel overflows and buries the Toolbox. */}
   555→      <aside className="order-2 w-full shrink-0 lg:order-1 lg:sticky lg:top-[76px] lg:h-[calc(100vh-96px)] lg:w-[300px] lg:self-start">
   556→        <Panel pad={0} className="flex max-h-full flex-col overflow-hidden">
   557→          <div className="shrink-0 border-b border-zinc-800 p-4">
   558→            <div className="flex items-center justify-between">
   559→              <div className="flex items-center gap-2">
   560→                <GraduationCap className="h-4 w-4 text-emerald-400" />
   561→                <h2 className="font-display text-sm font-semibold text-zinc-100">SeeQL Curriculum</h2>
   562→              </div>
   563→              <div className="flex items-center gap-0.5">
   564→                <Btn
   565→                  variant="ghost"
   566→                  onClick={() => setOpenModules(new Set(MODULES.map((m) => m.num)))}
   567→                  disabled={allOpen}
   568→                  title="Expand all modules"
   569→                  aria-label="Expand all modules"
   570→                >
   571→                  <ChevronsUpDown className="h-3.5 w-3.5" />
   572→                </Btn>
   573→                <Btn
   574→                  variant="ghost"
   575→                  onClick={() => setOpenModules(new Set())}
   576→                  disabled={!allOpen}
   577→                  title="Collapse all modules"
   578→                  aria-label="Collapse all modules"
   579→                >
   580→                  <ChevronsDownUp className="h-3.5 w-3.5" />
   581→                </Btn>
   582→              </div>
   583→            </div>
   584→            <p className="mt-1 text-[11px] text-zinc-500">
   585→              {MODULES.length} modules · {LESSONS.length} lessons · {totalProblems} graded queries · {totalMcqs} quiz questions
   586→            </p>
   587→            <ProgressBar value={doneCount / LESSONS.length} className="mt-3" />
   588→            <p className="mt-1.5 text-[11px] text-zinc-500">{doneCount}/{LESSONS.length} lessons complete</p>
   589→          </div>
   590→
   591→          <div className="min-h-0 flex-1 overflow-y-auto thin-scrollbar">
   592→            <div className="shrink-0 p-3">
   593→              <RankBlock xp={xp} level={level} streak={streak} />
   594→            </div>
   595→            {MODULES.map((m) => {
   596→              const mDone = m.lessons.filter((l) => xpEvents[`lesson:${l.id}`] !== undefined).length;
   597→              const open = openModules.has(m.num);
   598→              const isContinue = m.num === continueModuleNum;
   599→              const containsActive = view.kind === "lesson" && moduleOf(view.id)?.num === m.num;
   600→              return (
   601→                <nav key={m.id} className="border-b border-zinc-800/60 px-3 pb-1.5 pt-2 last:border-b-0">
   602→                  {/* module header — click to expand/collapse */}
   603→                  <button
   604→                    onClick={() => toggleModule(m.num)}
   605→                    aria-expanded={open}
   606→                    className="group flex w-full items-center gap-2.5 rounded-md px-1.5 py-2 text-left transition-colors hover:bg-zinc-900"
   607→                  >
   608→                    <span
   609→                      className={cn(
   610→                        "flex h-6 w-6 shrink-0 items-center justify-center rounded-md border font-mono text-[10px] font-medium transition-colors",
   611→                        mDone === m.lessons.length
   612→                          ? "border-emerald-400/50 bg-emerald-400/10 text-emerald-300"
   613→                          : containsActive
   614→                            ? "border-emerald-400/40 text-emerald-300"
   615→                            : isContinue
   616→                              ? "border-emerald-400/30 text-emerald-300/90"
   617→                              : "border-zinc-800 text-zinc-500",
   618→                      )}
   619→                    >
   620→                      {mDone === m.lessons.length ? <Check className="h-3 w-3" /> : m.num}
   621→                    </span>
   622→                    <span className="min-w-0 flex-1">
   623→                      <span
   624→                        className={cn(
   625→                          "block truncate text-xs font-medium",
   626→                          containsActive || mDone === m.lessons.length ? "text-zinc-100" : "text-zinc-300 group-hover:text-zinc-100",
   627→                        )}
   628→                      >
   629→                        {m.title}
   630→                      </span>
   631→                      <span className="mt-1.5 block h-0.5 overflow-hidden rounded-full bg-zinc-800">
   632→                        <span
   633→                          className="block h-full rounded-full bg-emerald-400/80 transition-[width] duration-500 ease-out"
   634→                          style={{ width: `${(mDone / m.lessons.length) * 100}%` }}
   635→                        />
   636→                      </span>
   637→                    </span>
   638→                    <span className="tnum shrink-0 font-mono text-[10px] text-zinc-600">
   639→                      {mDone}/{m.lessons.length}
   640→                    </span>
   641→                    <ChevronDown
   642→                      className={cn(
   643→                        "h-3.5 w-3.5 shrink-0 text-zinc-600 transition-transform duration-200",
   644→                        open && "rotate-180 text-zinc-400",
   645→                      )}
   646→                    />
   647→                  </button>
   648→
   649→                  {/* lessons — revealed on expand, connected by a tree rail */}
   650→                  <Collapse open={open}>
   651→                    <div className="ml-[16px] mt-0.5 flex flex-col gap-0.5 border-l border-zinc-800/70 pl-2 pb-2">
   652→                      {m.lessons.map((l) => {
   653→                        const done = xpEvents[`lesson:${l.id}`] !== undefined;
   654→                        const active = view.kind === "lesson" && view.id === l.id;
   655→                        return (
   656→                          <button
   657→                            key={l.id}
   658→                            onClick={() => openLesson(l.id)}
   659→                            className={cn(
   660→                              "relative flex items-start gap-2.5 rounded-md px-2.5 py-1.5 text-left transition-colors",
   661→                              active ? "bg-emerald-400/[0.09]" : "hover:bg-zinc-900",
   662→                            )}
   663→                          >
   664→                            {active && <span className="absolute inset-y-1 left-0 w-0.5 rounded-full bg-emerald-400" />}
   665→                            <span
   666→                              className={cn(
   667→                                "mt-0.5 flex h-5 w-5 shrink-0 items-center justify-center rounded-full text-[10px] font-medium transition-colors",
   668→                                done
   669→                                  ? "bg-emerald-400 text-[#052e1c]"
   670→                                  : active
   671→                                    ? "border border-emerald-400/60 text-emerald-200"
   672→                                    : "border border-zinc-700 text-zinc-500",
   673→                              )}
   674→                            >
   675→                              {done ? <Check className="h-2.5 w-2.5" /> : l.num}
   676→                            </span>
   677→                            <span className="min-w-0">
   678→                              <span className={cn("block truncate text-xs font-medium", active ? "text-zinc-50" : done ? "text-zinc-500" : "text-zinc-300")}>
   679→                                {l.title}
   680→                              </span>
   681→                              <span className="block text-[10px] text-zinc-600">{l.minutes} min · {l.problems.length} queries</span>
   682→                            </span>
   683→                          </button>
   684→                        );
   685→                      })}
   686→                    </div>
   687→                  </Collapse>
   688→                </nav>
   689→              );
   690→            })}
   691→          </div>
   692→
   693→          <div className="shrink-0 border-t border-zinc-800/60 p-3">
   694→            <SectionLabel className="mb-1.5">Toolbox</SectionLabel>
   695→            <div className="flex flex-col gap-0.5">
   696→              {CORE_AND_TOOLBOX.map((t) => (
   697→                <ToolRow
   698→                  key={t.kind}
   699→                  icon={t.icon}
   700→                  label={t.label}
   701→                  note={t.note}
   702→                  active={view.kind === t.kind}
   703→                  onClick={() => setView({ kind: t.kind } as View)}
   704→                />
   705→              ))}
   706→            </div>
   707→            <SectionLabel className="mb-1 mt-2">Practice lab</SectionLabel>
   708→            <div className="flex flex-col gap-0.5">
   709→              {LAB_TOOLS.map((t) => (
   710→                <ToolRow
   711→                  key={t.kind}
   712→                  icon={t.icon}
   713→                  label={t.label}
   714→                  note={t.note}
   715→                  active={view.kind === t.kind}
   716→                  onClick={() => setView({ kind: t.kind } as View)}
   717→                />
   718→              ))}
   719→            </div>
   720→          </div>
   721→
   722→          <div className="shrink-0 border-t border-zinc-800 p-3">
   723→            {confirmReset ? (
   724→              <div className="flex items-center justify-between gap-2 text-[11px]">
   725→                <span className="text-zinc-400">Erase all progress?</span>
   726→                <span className="flex gap-1.5">
   727→                  <Btn variant="outline" danger onClick={() => { resetAll(); setConfirmReset(false); }}>Erase</Btn>
   728→                  <Btn variant="outline" onClick={() => setConfirmReset(false)}>Keep</Btn>
   729→                </span>
   730→              </div>
   731→            ) : (
   732→              <Btn
   733→                variant="ghost"
   734→                onClick={() => setConfirmReset(true)}
   735→                className="p-0 text-[11px] text-zinc-600 hover:bg-transparent hover:text-zinc-400"
   736→              >
   737→                <RotateCcw className="h-3 w-3" /> Reset progress
   738→              </Btn>
   739→            )}
   740→          </div>
   741→        </Panel>
   742→      </aside>
   743→
   744→      {/* content — first on mobile so the course starts at the top of the page.
   745→          Each switch gets the same 0.2s fade-and-rise entrance (reduced-motion
   746→          aware), so moving between tools feels intentional, not like a page
   747→          hard-swapping underneath you. */}
   748→      <main className="order-1 min-w-0 flex-1 pb-16 lg:order-2">
   749→        <ToolRail view={view} setView={setView} />
   750→        <motion.div
   751→          key={view.kind === "lesson" ? `lesson:${view.id}` : view.kind}
   752→          initial={reduced ? { opacity: 0 } : { opacity: 0, y: 8 }}
   753→          animate={{ opacity: 1, y: 0 }}
   754→          transition={{ duration: 0.22, ease: "easeOut" }}
   755→        >
   756→          <ViewBody view={view} openLesson={openLesson} openPerf={openPerf} xp={xp} level={level} streak={streak} ctx={ctx} />
   757→        </motion.div>
   758→      </main>
   759→    </div>
   760→  );
   761→}
