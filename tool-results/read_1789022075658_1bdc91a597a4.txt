    47→function loadOpenModules(): number[] | null {
    48→  try {
    49→    const raw = localStorage.getItem(MODULES_OPEN_KEY);
    50→    if (!raw) return null;
    51→    const arr = JSON.parse(raw);
    52→    return Array.isArray(arr) ? arr.filter((n) => typeof n === "number") : null;
    53→  } catch {
    54→    return null;
    55→  }
    56→}

    82→function useBadgeWatcher() {
    83→  const xpEvents = useProgress((s) => s.xpEvents);
    84→  const mcqAttempts = useProgress((s) => s.mcqAttempts);
    85→  const badges = useProgress((s) => s.badges);
    86→  const xpByDay = useProgress((s) => s.xpByDay);
    87→  const dailyGoal = useProgress((s) => s.dailyGoal);
    88→  const prevLevel = useRef(-1);
    89→  const ready = useRef(false);
    90→
    91→  const xp = useMemo(() => totalXp({ xpEvents }), [xpEvents]);
    92→  const level = levelFor(xp);
    93→
    94→  const ctx = useMemo(() => {
    95→    const problemsSolved = Object.keys(xpEvents).filter((k) => k.startsWith("solve:")).length;
    96→    const mcqsCorrect = Object.values(mcqAttempts).filter((a) => a.correct).length;
    97→    const lessonsDone = Object.keys(xpEvents).filter((k) => k.startsWith("lesson:")).length;
    98→    const perfectQuizzes = LESSONS.filter((l) =>
    99→      l.mcqs.every((m) => mcqAttempts[m.id]?.firstTry),
   100→    ).length;
   101→    return { lessonsDone, problemsSolved, mcqsCorrect, perfectQuizzes };
   102→  }, [xpEvents, mcqAttempts]);
   103→
   104→  const streak = useMemo(() => currentStreak({ xpByDay, dailyGoal }), [xpByDay, dailyGoal]);
   105→
   106→  useEffect(() => {
   107→    if (!ready.current) {
   108→      ready.current = true;
   109→      prevLevel.current = level.index;
   110→      return;
   111→    }
   112→    const store = useProgress.getState();
   113→    for (const b of BADGES) {
   114→      if (store.badges[b.id] === undefined && b.test(ctx, xp, streak)) {
   115→        if (store.unlockBadge(b.id)) {
   116→          celebrate({ kind: "badge", title: `Badge · ${b.title}`, sub: b.desc });
   117→        }
   118→      }
   119→    }
   120→    if (level.index > prevLevel.current) {
   121→      celebrate({ kind: "level", title: `Level up — ${level.title}`, sub: `Rank ${level.index + 1} of 6` });
   122→      prevLevel.current = level.index;
   123→    }
   124→  }, [ctx, xp, streak, level]);
   125→
   126→  return { xp, level, streak, ctx };
   127→}

   148→      </div>

   211→              const done = xpEvents[`lesson:${l.id}`] !== undefined;
   212→              return (
   213→                <button
   214→                  key={l.id}
   215→                  onClick={() => onOpenLesson(l.id)}
   216→                  className="group/lesson flex items-center gap-2.5 rounded-md px-2 py-1.5 text-left transition-colors hover:bg-zinc-900"

   236→            <button
   237→              onClick={() => onOpenLesson(nextInModule.id)}
   238→              className="mt-2 flex w-full items-center justify-center gap-1.5 rounded-md border border-emerald-400/30 bg-emerald-400/[0.06] px-3 py-1.5 text-[11px] font-medium text-emerald-300 transition-colors hover:bg-emerald-400/[0.12]"

   241→            </button>

   244→      </Collapse>
   245→    </div>
   246→  );
   247→}
