    38→function loadOpenModules(): number[] | null {
    39→  try {
    40→    const raw = localStorage.getItem(MODULES_OPEN_KEY);
    41→    if (!raw) return null;
    42→    const arr = JSON.parse(raw);
    43→    return Array.isArray(arr) ? arr.filter((n) => typeof n === "number") : null;
    44→  } catch {
    45→    return null;
    46→  }
    47→}

    73→function useBadgeWatcher() {
    74→  const xpEvents = useProgress((s) => s.xpEvents);
    75→  const mcqAttempts = useProgress((s) => s.mcqAttempts);
    76→  const badges = useProgress((s) => s.badges);
    77→  const xpByDay = useProgress((s) => s.xpByDay);
    78→  const dailyGoal = useProgress((s) => s.dailyGoal);
    79→  const prevLevel = useRef(-1);
    80→  const ready = useRef(false);
    81→
    82→  const xp = useMemo(() => totalXp({ xpEvents }), [xpEvents]);
    83→  const level = levelFor(xp);
    84→
    85→  const ctx = useMemo(() => {
    86→    const problemsSolved = Object.keys(xpEvents).filter((k) => k.startsWith("solve:")).length;
    87→    const mcqsCorrect = Object.values(mcqAttempts).filter((a) => a.correct).length;
    88→    const lessonsDone = Object.keys(xpEvents).filter((k) => k.startsWith("lesson:")).length;
    89→    const perfectQuizzes = LESSONS.filter((l) =>
    90→      l.mcqs.every((m) => mcqAttempts[m.id]?.firstTry),
    91→    ).length;
    92→    return { lessonsDone, problemsSolved, mcqsCorrect, perfectQuizzes };
    93→  }, [xpEvents, mcqAttempts]);
    94→
    95→  const streak = useMemo(() => currentStreak({ xpByDay, dailyGoal }), [xpByDay, dailyGoal]);
    96→
    97→  useEffect(() => {
    98→    if (!ready.current) {
    99→      ready.current = true;
   100→      prevLevel.current = level.index;
   101→      return;
   102→    }
   103→    const store = useProgress.getState();
   104→    for (const b of BADGES) {
   105→      if (store.badges[b.id] === undefined && b.test(ctx, xp, streak)) {
   106→        if (store.unlockBadge(b.id)) {
   107→          celebrate({ kind: "badge", title: `Badge · ${b.title}`, sub: b.desc });
   108→        }
   109→      }
   110→    }
   111→    if (level.index > prevLevel.current) {
   112→      celebrate({ kind: "level", title: `Level up — ${level.title}`, sub: `Rank ${level.index + 1} of 6` });
   113→      prevLevel.current = level.index;
   114→    }
   115→  }, [ctx, xp, streak, level]);
   116→
   117→  return { xp, level, streak, ctx };
   118→}

   125→        <span className="tnum font-mono text-xs text-emerald-300">{xp.toLocaleString()} XP</span>
   126→      </div>

   205→                  key={l.id}
   206→                  onClick={() => onOpenLesson(l.id)}

   241→  const ladder = ["Query Apprentice", "Analyst I", "Analyst II", "Senior Analyst", "Query Wizard"];
