   111→   111→function useBadgeWatcher() {
   112→   112→  const xpEvents = useProgress((s) => s.xpEvents);
   113→   113→  const mcqAttempts = useProgress((s) => s.mcqAttempts);
   114→   114→  const badges = useProgress((s) => s.badges);
   115→   115→  const xpByDay = useProgress((s) => s.xpByDay);
   116→   116→  const dailyGoal = useProgress((s) => s.dailyGoal);
   117→   117→  const prevLevel = useRef(-1);
   118→   118→  const ready = useRef(false);
   119→   119→
   120→   120→  const xp = useMemo(() => totalXp({ xpEvents }), [xpEvents]);
   121→   121→  const level = levelFor(xp);
   122→   122→
   123→   123→  const ctx = useMemo(() => {
   124→   124→    const problemsSolved = Object.keys(xpEvents).filter((k) => k.startsWith("solve:")).length;
   125→   125→    const mcqsCorrect = Object.values(mcqAttempts).filter((a) => a.correct).length;
   126→   126→    const lessonsDone = Object.keys(xpEvents).filter((k) => k.startsWith("lesson:")).length;
   127→   127→    const perfectQuizzes = LESSONS.filter((l) =>
   128→   128→      l.mcqs.every((m) => mcqAttempts[m.id]?.firstTry),
   129→   129→    ).length;
   130→   130→    return { lessonsDone, problemsSolved, mcqsCorrect, perfectQuizzes };
   131→   131→  }, [xpEvents, mcqAttempts]);
   132→   132→
   133→   133→  const streak = useMemo(() => currentStreak({ xpByDay, dailyGoal }), [xpByDay, dailyGoal]);
   134→   134→
   135→   135→  useEffect(() => {
   136→   136→    if (!ready.current) {
   137→   137→      ready.current = true;
   138→   138→      prevLevel.current = level.index;
   139→   139→      return;
   140→   140→    }
   141→   141→    const store = useProgress.getState();
   142→   142→    for (const b of BADGES) {
   143→   143→      if (store.badges[b.id] === undefined && b.test(ctx, xp, streak)) {
   144→   144→        if (store.unlockBadge(b.id)) {
   145→   145→          celebrate({ kind: "badge", title: `Badge · ${b.title}`, sub: b.desc });
   146→   146→        }
   147→   147→      }
   148→   148→    }
   149→   149→    if (level.index > prevLevel.current) {
   150→   150→      celebrate({ kind: "level", title: `Level up — ${level.title}`, sub: `Rank ${level.index + 1} of 6` });
   151→   151→      prevLevel.current = level.index;
   152→   152→    }
   153→   153→  }, [ctx, xp, streak, level]);
   154→   154→
   155→   155→  return { xp, level, streak, ctx };
   156→   156→}
