    40→function ArrowHead({ x, y, ang = 0, cls = "stroke-zinc-600", h = 6.5, delay = 0 }: { x: number; y: number; ang?: number; cls?: string; h?: number; delay?: number }) {
    41→  // ang = direction of travel at the tip (radians, 0 = pointing right)
    42→  const a1 = ang + Math.PI - 0.46;
    43→  const a2 = ang + Math.PI + 0.46;
    44→  return (
    45→    <motion.g {...fade(delay)} className={cls} strokeWidth={1.5} strokeLinecap="round" fill="none">
    46→      <line x1={x} y1={y} x2={x + h * Math.cos(a1)} y2={y + h * Math.sin(a1)} />
    47→      <line x1={x} y1={y} x2={x + h * Math.cos(a2)} y2={y + h * Math.sin(a2)} />
    48→    </motion.g>
    49→  );
    50→}

    52→function Arrow({ x1, y1, x2, y2, cls = "stroke-zinc-600" }: { x1: number; y1: number; x2: number; y2: number; cls?: string }) {
    53→  const ang = Math.atan2(y2 - y1, x2 - x1);
    54→  const h = 6.5;
    55→  const a1 = ang + Math.PI - 0.46;
    56→  const a2 = ang + Math.PI + 0.46;
    57→  return (
    58→    <g className={cls} strokeWidth={1.5} strokeLinecap="round" fill="none">
    59→      <line x1={x1} y1={y1} x2={x2} y2={y2} />
    60→      <line x1={x2} y1={y2} x2={x2 + h * Math.cos(a1)} y2={y2 + h * Math.sin(a1)} />
    61→      <line x1={x2} y1={y2} x2={x2 + h * Math.cos(a2)} y2={y2 + h * Math.sin(a2)} />
    62→    </g>
    63→  );
    64→}

    76→/** Smooth S-curve link with horizontal tangents — the workhorse for
    77→ *  row-to-row foreign-key connections. Points rightward (x2 > x1). */
    78→function Link({
    79→  x1,
    80→  y1,
    81→  x2,
    82→  y2,
    83→  cls = "stroke-zinc-600",
    84→  w = 1.5,
    85→  dash,
    86→  arrow = true,
    87→  dot,
    88→  bend = 0.42,
    89→  delay = 0,
    90→}: {
    91→  x1: number;
    92→  y1: number;
    93→  x2: number;
    94→  y2: number;
    95→  cls?: string;
    96→  w?: number;
    97→  dash?: string;
    98→  arrow?: boolean;
    99→  dot?: string;
   100→  bend?: number;
   101→  delay?: number;
   102→}) {
   103→  const dx = Math.max(8, Math.min(Math.abs(x2 - x1) * bend, Math.abs(x2 - x1) / 2));
   104→  const d = `M ${x1} ${y1} C ${x1 + dx} ${y1}, ${x2 - dx} ${y2}, ${x2} ${y2}`;
   105→  return (
   106→    <g>
   107→      <motion.path {...drawIn(delay)} d={d} className={cls} strokeWidth={w} strokeDasharray={dash} fill="none" strokeLinecap="round" />
   108→      {dot && <motion.circle {...fade(delay + 0.1)} cx={x1} cy={y1} r={3} className={dot} />}
   109→      {arrow && <ArrowHead x={x2} y={y2} ang={0} cls={cls} delay={delay + 0.45} />}
   110→    </g>
   111→  );
   112→}

   114→/** Rounded orthogonal polyline through the given points — for routes that
   115→ *  must travel around other elements (down, across, up). */
   116→function Pipe({
   117→  pts,
   118→  cls = "stroke-zinc-600",
   119→  w = 1.5,
   120→  dash,
   121→  arrowAng,
   122→  arrowCls,
   123→  r = 12,
   124→  delay = 0,
   125→}: {
   126→  pts: Array<[number, number]>;
   127→  cls?: string;
   128→  w?: number;
   129→  dash?: string;
   130→  /** direction of travel at the final point, if an arrowhead is wanted */
   131→  arrowAng?: number;
   132→  arrowCls?: string;
   133→  r?: number;
   134→  delay?: number;
   135→}) {
   136→  let d = `M ${pts[0][0]} ${pts[0][1]}`;
   137→  for (let i = 1; i < pts.length - 1; i++) {
   138→    const [px, py] = pts[i - 1];
   139→    const [cx, cy] = pts[i];
   140→    const [nx, ny] = pts[i + 1];
   141→    const inLen = Math.hypot(cx - px, cy - py) || 1;
   142→    const outLen = Math.hypot(nx - cx, ny - cy) || 1;
   143→    const r1 = Math.min(r, inLen / 2);
   144→    const r2 = Math.min(r, outLen / 2);
   145→    const ax = cx - ((cx - px) / inLen) * r1;
   146→    const ay = cy - ((cy - py) / inLen) * r1;
   147→    const bx = cx + ((nx - cx) / outLen) * r2;
   148→    const by = cy + ((ny - cy) / outLen) * r2;
   149→    d += ` L ${ax} ${ay} Q ${cx} ${cy}, ${bx} ${by}`;
   150→  }
   151→  const last = pts[pts.length - 1];
   152→  d += ` L ${last[0]} ${last[1]}`;
   153→  return (
   154→    <g>
   155→      <motion.path {...drawIn(delay)} d={d} className={cls} strokeWidth={w} strokeDasharray={dash} fill="none" strokeLinecap="round" />
   156→      {arrowAng !== undefined && <ArrowHead x={last[0]} y={last[1]} ang={arrowAng} cls={arrowCls ?? cls} delay={delay + 0.45} />}
   157→    </g>
   158→  );
   159→}

   184→/** Deterministic pseudo-random for dot layouts (stable across renders). */
   185→function rng(seed: number) {
   186→  let s = seed >>> 0;
   187→  return () => {
   188→    s = (1664525 * s + 1013904223) >>> 0;
   189→    return s / 4294967296;
   190→  };
   191→}

   193→function DotField({ n, cols, x0, y0, gap = 15, r = 4, passFrom }: { n: number; cols: number; x0: number; y0: number; gap?: number; r?: number; passFrom?: number }) {
   194→  const rand = rng(7);
   195→  const dots: React.JSX.Element[] = [];
   196→  for (let i = 0; i < n; i++) {
   197→    const cx = x0 + (i % cols) * gap + rand() * 4 - 2;
   198→    const cy = y0 + Math.floor(i / cols) * gap + rand() * 4 - 2;
   199→    const pass = passFrom === undefined || i < passFrom;
   200→    dots.push(
   201→      <circle key={i} cx={cx} cy={cy} r={r} className={pass ? "fill-zinc-500" : "fill-zinc-700"} opacity={pass ? 0.9 : 0.55} />,
   202→    );
   203→  }
   204→  return <g>{dots}</g>;
   205→}
