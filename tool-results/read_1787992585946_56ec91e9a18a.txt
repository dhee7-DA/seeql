    97→    97→function JsonView({ plan }: { plan: QueryPlan }) {

    99→    99→  const json = JSON.stringify({ ...plan, tree: undefined }, null, 2);
   100→   100→
   101→   101→  const copy = async () => {
   102→   102→    try {
   103→   103→      await navigator.clipboard.writeText(json);
   104→   104→      setCopied(true);
   105→   105→      setTimeout(() => setCopied(false), 1600);
