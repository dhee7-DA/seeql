    25→/** Normalize identifier-ish values: "x" | {value:"x"} | null → string|null */
    26→function nameOf(val: unknown): string | null {
    27→  if (val === null || val === undefined) return null;
    28→  if (typeof val === "string") return val;
    29→  if (typeof val === "number") return String(val);
    30→  if (typeof val === "object") {
    31→    const rec = val as Record<string, unknown>;
    32→    if ("value" in rec && typeof rec.value !== "object")
    33→      return rec.value === null ? null : String(rec.value);
    34→    if ("name" in rec) return nameOf(rec.name);
    35→  }
    36→  return null;
    37→}
    38→
    39→function funcName(node: Record<string, unknown>): string {
    40→  const raw = node.name;
    41→  const direct = nameOf(raw);
    42→  if (direct) return direct;
    43→  if (Array.isArray(raw)) {
    44→    const parts = (raw as Array<Record<string, unknown>>)
    45→      .map((p) => nameOf(p) ?? "")
    46→      .filter(Boolean);
    47→    if (parts.length) return parts.join(".");
    48→  }
    49→  if (raw && typeof raw === "object") {
    50→    const inner = (raw as Record<string, unknown>).name;
    51→    if (Array.isArray(inner)) {
    52→      const parts = (inner as Array<Record<string, unknown>>)
    53→        .map((p) => nameOf(p) ?? "")
    54→        .filter(Boolean);
    55→      if (parts.length) return parts.join(".");
    56→    }
    57→  }
    58→  return "FUNC";
    59→}
    60→
    61→function argsToArray(args: unknown): Record<string, unknown>[] {
    62→  if (!args) return [];
    63→  if (Array.isArray(args)) return args as Record<string, unknown>[];
    64→  const a = args as Record<string, unknown>;
    65→  if (a.expr) return [a.expr as Record<string, unknown>];
    66→  if (a.value !== undefined) return [].concat(a.value as never) as Record<string, unknown>[];
    67→  return [];
    68→}

    70→/** Render an OVER (...) clause of a window function */
    71→function overToString(over: unknown): string {
    72→  if (!over || typeof over !== "object") return "";
    73→  const o = over as Record<string, unknown>;
    74→  if (o.type === "window") {
    75→    const specWrapper = o.as_window_specification as Record<string, unknown> | undefined;
    76→    const spec = (specWrapper?.window_specification ?? specWrapper) as
    77→      | Record<string, unknown>
    78→      | undefined;
    79→    if (!spec) return " OVER (...)";
    80→    const parts: string[] = [];
    81→    const pb = spec.partitionby;
    82→    if (Array.isArray(pb) && pb.length) {
    83→      parts.push(
    84→        `PARTITION BY ${pb
    85→          .map((p: Record<string, unknown>) => exprToString(p?.expr ?? p))
    86→          .join(", ")}`,
    87→      );
    88→    }
    89→    const ob = spec.orderby;
    90→    if (Array.isArray(ob) && ob.length) {
    91→      parts.push(
    92→        `ORDER BY ${ob
    93→          .map((x: Record<string, unknown>) =>
    94→            [exprToString(x.expr), x.type ? String(x.type) : ""].filter(Boolean).join(" "),
    95→          )
    96→          .join(", ")}`,
    97→      );
    98→    }
    99→    const frame = spec.window_frame_clause;
   100→    if (frame) {
   101→      const frameText =
   102→        typeof frame === "string" ? frame.toUpperCase() : exprToString(frame);
   103→      if (frameText && frameText !== "?") parts.push(frameText);
   104→    }
   105→    return ` OVER (${parts.join(" ")})`;
   106→  }
   107→  if (nameOf(o.type)) return ` OVER ${nameOf(o.type)}`;
   108→  return " OVER (...)";
   109→}

   115→export function exprToString(e: unknown): string {
   116→  if (e === null || e === undefined) return "";
   117→  if (typeof e === "string") return e;
   118→  if (typeof e === "number") return String(e);
   119→  if (typeof e !== "object") return String(e);
   120→  const node = e as Record<string, unknown>;
   121→
   122→  // Sub-selects inside expressions arrive wrapped as { tableList, columnList, ast }
   123→  if (!node.type && node.ast && typeof node.ast === "object") {
   124→    return exprToString(node.ast);
   125→  }
   126→
   127→  switch (node.type) {
   128→    case "expr":
   129→      return exprToString(node.expr);
   130→    case "default":
   131→      return String(node.value ?? "");
   132→    case "column_ref": {
   133→      const table = nameOf(node.table);
   134→      // "t.*" arrives as a column_ref whose column is the string "*"
   135→      if (node.column === "*") return table ? `${table}.*` : "*";
   136→      const col =
   137→        nameOf(node.column) ??
   138→        (node.column && typeof node.column === "object"
   139→          ? exprToString(node.column)
   140→          : "?");
   141→      return table ? `${table}.${col}` : col;
   142→    }
   143→    case "star": {
   144→      const table = nameOf(node.table);
   145→      return table ? `${table}.*` : "*";
   146→    }
   147→    case "number":
   148→    case "integer":
   149→    case "double":
   150→    case "float":
   151→      return String(node.value);
   152→    case "string":
   153→    case "single_quote_string":
   154→    case "double_quote_string":
   155→    case "backticks_quote_string":
   156→      return `'${String(node.value ?? "")}'`;
   157→    case "bool":
   158→      return String(node.value);
   159→    case "null":
   160→      return "NULL";
   161→    case "param":
   162→      return "?";
   163→    case "expr_list":
   164→    case "value_list": {
   165→      const inner = argsToArray(node.value)
   166→        .map(exprToString)
   167→        .join(", ");
   168→      return `(${inner})`;
   169→    }
   170→    case "binary_expr": {
   171→      const left = exprToString(node.left);
   172→      const right = exprToString(node.right);
   173→      return `${left} ${String(node.operator ?? "")} ${right}`;
   174→    }
   175→    case "unary_expr":
   176→      return `${String(node.operator ?? "")}${String(node.operator) === "-" ? "" : " "}${exprToString(node.expr)}`;
   177→    case "aggr_func": {
   178→      const args = node.args as Record<string, unknown> | undefined;
   179→      const distinct = args?.distinct ? "DISTINCT " : "";
   180→      let inner = "*";
   181→      if (args?.expr) inner = exprToString(args.expr);
   182→      else if (args?.args)
   183→        inner = argsToArray(args.args)
   184→          .map(exprToString)
   185→          .join(", ");
   186→      return `${funcName(node)}(${distinct}${inner})${overToString(node.over)}`;
   187→    }
   188→    case "window_func": {
   189→      const inner = argsToArray(node.args)
   190→        .map(exprToString)
   191→        .join(", ");
   192→      return `${funcName(node)}(${inner})${overToString(node.over)}`;
   193→    }
   194→    case "function":
   195→    case "scalar_function": {
   196→      const inner = argsToArray(node.args)
   197→        .map(exprToString)
   198→        .join(", ");
   199→      return `${funcName(node)}(${inner})${overToString(node.over)}`;
   200→    }
   201→    case "case": {
   202→      const operand = node.operand ? exprToString(node.operand) : "";
   203→      const whens = Array.isArray(node.whens)
   204→        ? (node.whens as Array<Record<string, unknown>>)
   205→            .map((w) => {
   206→              const cond = w.cond ?? w.when;
   207→              const result = w.result ?? w.then;
   208→              return `WHEN ${exprToString(cond)} THEN ${exprToString(result)}`;
   209→            })
   210→            .join(" ")
   211→        : "";
   212→      const els = node.else ? ` ELSE ${exprToString(node.else)}` : "";
   213→      return `CASE ${operand ? `${operand} ` : ""}${whens}${els} END`;
   214→    }
   215→    case "cast": {
   216→      const target = node.target as Record<string, unknown> | undefined;
   217→      const dataType = nameOf(target?.dataType) ?? "";
   218→      const expr = exprToString(node.expression ?? node.expr);
   219→      return `CAST(${expr} AS ${dataType})`;
   220→    }
   221→    case "exprlist":
   222→      return argsToArray(node.value)
   223→        .map(exprToString)
   224→        .join(", ");
   225→    case "select": {
   226→      const cols = Array.isArray(node.columns)
   227→        ? (node.columns as Array<Record<string, unknown>>)
   228→        : [];
   229→      const colTexts = cols
   230→        .slice(0, 2)
   231→        .map((c) => exprToString(c?.expr ?? c))
   232→        .filter(Boolean);
   233→      const fromArr = Array.isArray(node.from)
   234→        ? (node.from as Array<Record<string, unknown>>)
   235→            .map((f) => nameOf(f.table))
   236→            .filter((n): n is string => !!n)
   237→        : [];
   238→      let text = `SELECT ${colTexts.join(", ") || "…"}`;
   239→      if (colTexts.length > 1 || cols.length > 2) text += ", …";
   240→      if (fromArr.length) text += ` FROM ${fromArr.join(", ")}`;
   241→      return `(${text})`;
   242→    }
   243→    case "interval": {
   244→      const unit = nameOf(node.unit) ?? "";
   245→      return `INTERVAL ${exprToString(node.value)} ${unit}`.trim();
   246→    }
   247→    default: {
   248→      if (node.value !== undefined && typeof node.value !== "object")
   249→        return String(node.value);
   250→      if (node.expr) return exprToString(node.expr);
   251→      if (node.column) return exprToString(node.column);
   252→      const viaName = nameOf(node.name);
   253→      if (viaName) {
   254→        const inner = argsToArray(node.args)
   255→          .map(exprToString)
   256→          .join(", ");
   257→        return `${viaName}(${inner})`;
   258→      }
   259→      return "?";
   260→    }
   261→  }
   262→}

   264→/** Recursively collect referenced table/alias names inside an expression */
   265→function collectTableRefs(node: unknown, acc: Set<string>): void {
   266→  if (!node || typeof node !== "object") return;
   267→  if (Array.isArray(node)) {
   268→    node.forEach((n) => collectTableRefs(n, acc));
   269→    return;
   270→  }
   271→  const rec = node as Record<string, unknown>;
   272→  if (rec.type === "column_ref" || ("column" in rec && "table" in rec)) {
   273→    const t = nameOf(rec.table);
   274→    if (t) acc.add(t);
   275→  }
   276→  for (const key of Object.keys(rec)) {
   277→    if (key === "type" || key === "location" || key === "range") continue;
   278→    const val = rec[key];
   279→    if (val && typeof val === "object") collectTableRefs(val, acc);
   280→  }
   281→}
   282→
   283→/** Recursively find every SELECT node and collect its FROM-clause tables */
   284→function collectTablesDeep(node: unknown, acc: Set<string>, depth = 0): void {
   285→  if (!node || typeof node !== "object" || depth > 8) return;
   286→  if (Array.isArray(node)) {
   287→    node.forEach((n) => collectTablesDeep(n, acc, depth));
   288→    return;
   289→  }
   290→  const rec = node as Record<string, unknown>;
   291→  if (Array.isArray(rec.from)) {
   292→    for (const f of rec.from as Array<Record<string, unknown>>) {
   293→      const t = nameOf(f.table);
   294→      if (t) acc.add(f.as ? `${t} (${nameOf(f.as) ?? f.as})` : t);
   295→      else if (f.expr) {
   296→        acc.add(`(subquery)${f.as ? ` AS ${nameOf(f.as) ?? f.as}` : ""}`);
   297→        collectTablesDeep(f.expr, acc, depth + 1);
   298→      }
   299→    }
   300→  }
   301→  for (const key of Object.keys(rec)) {
   302→    if (key === "type" || key === "location" || key === "range") continue;
   303→    const val = rec[key];
   304→    if (val && typeof val === "object") collectTablesDeep(val, acc, depth + 1);
   305→  }
   306→}

   308→/** Check if any nested expression contains an aggregation or a subquery */
   309→function scanFlags(
   310→  node: unknown,
   311→  flags: { aggr: boolean; subquery: boolean; window: boolean },
   312→): void {
   313→  if (!node || typeof node !== "object") return;
   314→  if (Array.isArray(node)) {
   315→    node.forEach((n) => scanFlags(n, flags));
   316→    return;
   317→  }
   318→  const rec = node as Record<string, unknown>;
   319→  if (rec.type === "aggr_func") flags.aggr = true;
   320→  if (rec.type === "window_func") flags.window = true;
   321→  if (rec.type === "select") flags.subquery = true;
   322→  for (const key of Object.keys(rec)) {
   323→    if (key === "type" || key === "location" || key === "range") continue;
   324→    const val = rec[key];
   325→    if (val && typeof val === "object") scanFlags(val, flags);
   326→  }
   327→}

   333→function flattenConditions(node: unknown, acc: string[]): void {
   334→  if (!node) return;
   335→  const rec = node as Record<string, unknown>;
   336→  if (rec.type === "binary_expr" && rec.operator === "AND") {
   337→    flattenConditions(rec.left, acc);
   338→    flattenConditions(rec.right, acc);
   339→  } else {
   340→    acc.push(exprToString(node));
   341→  }
   342→}

   348→interface WalkCtx {
   349→  tables: Map<string, TableInfo>;
   350→  joins: JoinInfo[];
   351→  columns: ColumnInfo[];
   352→  where: string[];
   353→  groupBy: string[];
   354→  having: string[];
   355→  orderBy: OrderItem[];
   356→  limit: string | null;
   357→  offset: string | null;
   358→  distinct: boolean;
   359→  setOps: string[];
   360→  ctes: CteInfo[];
   361→  cteNames: Set<string>;
   362→  flags: { aggr: boolean; subquery: boolean; window: boolean };
   363→  depth: number;
   364→}
   365→
   366→function upsertTable(ctx: WalkCtx, info: TableInfo): void {
   367→  const existing = ctx.tables.get(info.id);
   368→  if (existing) {
   369→    existing.joinType = existing.joinType ?? info.joinType;
   370→    existing.onText = existing.onText ?? info.onText;
   371→    existing.joinsTo = Array.from(new Set([...existing.joinsTo, ...info.joinsTo]));
   372→    existing.columns = Array.from(new Set([...existing.columns, ...info.columns]));
   373→    return;
   374→  }
   375→  ctx.tables.set(info.id, info);
   376→}

   378→function walkSelect(select: Record<string, unknown>, ctx: WalkCtx): void {
   379→  // ---- CTEs -------------------------------------------------------
   380→  const withs = select.with as Array<Record<string, unknown>> | undefined;
   381→  if (Array.isArray(withs) && ctx.depth === 0) {
   382→    for (const w of withs) {
   383→      const name = nameOf(w.name) ?? "cte";
   384→      const stmt = w.stmt as Record<string, unknown> | undefined;
   385→      const tablesUsed = new Set<string>();
   386→      if (stmt) collectTablesDeep(stmt, tablesUsed);
   387→      const declCols = Array.isArray(w.columns)
   388→        ? (w.columns as Array<Record<string, unknown>>)
   389→            .map((c) => nameOf(c?.column ?? c))
   390→            .filter((c): c is string => !!c)
   391→        : null;
   392→      ctx.ctes.push({
   393→        name,
   394→        declaredColumns: declCols,
   395→        tablesUsed: Array.from(tablesUsed).slice(0, 8),
   396→        summary: summarizeSelect(stmt),
   397→      });
   398→      ctx.cteNames.add(name);
   399→    }
   400→  }
   401→
   402→  // ---- FROM / JOIN ------------------------------------------------
   403→  const from = (select.from as Array<Record<string, unknown>> | undefined) ?? [];
   404→  const seenIds: string[] = [];
   405→
   406→  from.forEach((f, index) => {
   407→    if ((f.type as string | undefined) === "dual") return;
   408→
   409→    let id: string;
   410→    let name: string;
   411→    let kind: TableInfo["kind"];
   412→
   413→    if (f.expr && (f.expr as Record<string, unknown>).type === "select") {
   414→      kind = "subquery";
   415→      name = "(derived table)";
   416→      id = String(f.as ?? `sub${ctx.tables.size}`);
   417→      ctx.flags.subquery = true;
   418→      scanFlags(f.expr, ctx.flags);
   419→      walkSelect(f.expr as Record<string, unknown>, {
   420→        ...ctx,
   421→        depth: ctx.depth + 1,
   422→      });
   423→    } else {
   424→      const db = nameOf(f.db);
   425→      name = `${db ? `${db}.` : ""}${nameOf(f.table) ?? "?"}`;
   426→      kind = ctx.cteNames.has(name) ? "cte" : "table";
   427→      id = String(nameOf(f.as) ?? name);
   428→    }
   429→
   430→    const joinRaw = index === 0 ? null : nameOf(f.join) ?? null;
   431→    const joinType = joinRaw ? joinRaw.replace(/_/g, " ").toUpperCase() : null;
   432→
   433→    const onRefs = new Set<string>();
   434→    let onText: string | null = null;
   435→    if (f.on) {
   436→      onText = exprToString(f.on);
   437→      collectTableRefs(f.on, onRefs);
   438→    } else if (Array.isArray(f.using)) {
   439→      const cols = (f.using as Array<Record<string, unknown>>)
   440→        .map((u) => nameOf(u?.column ?? u) ?? "")
   441→        .filter(Boolean);
   442→      onText = `USING (${cols.join(", ")})`;
   443→    }
   444→
   445→    upsertTable(ctx, {
   446→      id,
   447→      name,
   448→      alias: nameOf(f.as),
   449→      kind,
   450→      joinType,
   451→      onText,
   452→      joinsTo: [],
   453→      columns: [],
   454→      depth: ctx.depth,
   455→    });
   456→    if (!seenIds.includes(id)) seenIds.push(id);
   457→
   458→    if (joinType) {
   459→      const resolved = new Set<string>();
   460→      for (const ref of onRefs) {
   461→        if (ctx.tables.has(ref) && ref !== id) resolved.add(ref);
   462→      }
   463→      if (resolved.size === 0 && seenIds.length > 1) {
   464→        const prev = seenIds[seenIds.length - 2];
   465→        if (prev !== id) resolved.add(prev);
   466→      }
   467→      const left = Array.from(resolved)[0] ?? seenIds[seenIds.length - 2];
   468→      if (left && left !== id) {
   469→        ctx.joins.push({
   470→          left,
   471→          right: id,
   472→          type: joinType,
   473→          on: onText ?? "(condition not available)",
   474→        });
   475→        const leftInfo = ctx.tables.get(left);
   476→        if (leftInfo)
   477→          leftInfo.joinsTo = Array.from(new Set([...leftInfo.joinsTo, id]));
   478→        const rightInfo = ctx.tables.get(id);
   479→        if (rightInfo)
   480→          rightInfo.joinsTo = Array.from(new Set([...rightInfo.joinsTo, left]));
   481→      }
   482→    }
   483→  });
   484→
   485→  // ---- SELECT list ------------------------------------------------
   486→  const rawColumns = select.columns;
   487→  const colList = Array.isArray(rawColumns)
   488→    ? (rawColumns as Array<Record<string, unknown>>)
   489→    : rawColumns === "*"
   490→      ? [{ expr: { type: "star" } }]
   491→      : [];
   492→
   493→  for (const c of colList) {
   494→    const expr = (c.expr ?? c) as Record<string, unknown> | undefined;
   495→    if (!expr || typeof expr !== "object") continue;
   496→    scanFlags(expr, ctx.flags);
   497→    const alias = nameOf(c.as);
   498→    const isStar =
   499→      expr.type === "star" ||
   500→      (expr.type === "column_ref" && expr.column === "*");
   501→    const text = exprToString(expr);
   502→
   503→    let tableRef: string | null = null;
   504→    if (expr.type === "column_ref") tableRef = nameOf(expr.table);
   505→    else if (expr.type === "star") tableRef = nameOf(expr.table);
   506→    else {
   507→      const refs = new Set<string>();
   508→      collectTableRefs(expr, refs);
   509→      const known = Array.from(refs).find((r) => ctx.tables.has(r));
   510→      if (known) tableRef = known;
   511→    }
   512→    // Unqualified column (e.g. `SELECT id`): if exactly one table is in
   513→    // scope at this depth, the column can only belong to it.
   514→    if (!tableRef && expr.type === "column_ref") {
   515→      const scopeTables = Array.from(ctx.tables.values()).filter(
   516→        (t) => t.depth === ctx.depth,
   517→      );
   518→      if (scopeTables.length === 1) tableRef = scopeTables[0].id;
   519→    }
   520→
   521→    const innerAggr = { aggr: false, subquery: false, window: false };
   522→    if (expr.type !== "aggr_func") scanFlags(expr, innerAggr);
   523→
   524→    ctx.columns.push({
   525→      text: alias ? `${text} AS ${alias}` : text,
   526→      alias,
   527→      table: tableRef,
   528→      aggregation:
   529→        expr.type === "aggr_func"
   530→          ? funcName(expr)
   531→          : innerAggr.aggr
   532→            ? "AGG"
   533→            : null,
   534→      isStar,
   535→    });
   536→
   537→    if (isStar && !tableRef) {
   538→      for (const t of ctx.tables.values()) {
   539→        if (t.depth === ctx.depth) t.columns.push("*");
   540→      }
   541→    } else if (tableRef) {
   542→      const t = ctx.tables.get(tableRef);
   543→      const label =
   544→        alias ??
   545→        (tableRef && text.startsWith(`${tableRef}.`)
   546→          ? text.slice(tableRef.length + 1)
   547→          : text);
   548→      if (t && !t.columns.includes(label)) t.columns.push(label);
   549→    }
   550→  }
   551→
   552→  // ---- WHERE ------------------------------------------------------
   553→  if (select.where) {
   554→    flattenConditions(select.where, ctx.where);
   555→    scanFlags(select.where, ctx.flags);
   556→  }
   557→
   558→  // ---- GROUP BY / HAVING ------------------------------------------
   559→  const gb = select.groupby;
   560→  const gbList: unknown[] = Array.isArray(gb)
   561→    ? gb
   562→    : gb && typeof gb === "object" && Array.isArray((gb as Record<string, unknown>).columns)
   563→      ? ((gb as Record<string, unknown>).columns as unknown[])
   564→      : [];
   565→  for (const g of gbList) ctx.groupBy.push(exprToString(g));
   566→
   567→  if (select.having) {
   568→    flattenConditions(select.having, ctx.having);
   569→    scanFlags(select.having, ctx.flags);
   570→  }
   571→
   572→  // ---- ORDER BY ---------------------------------------------------
   573→  if (Array.isArray(select.orderby)) {
   574→    for (const o of select.orderby as Array<Record<string, unknown>>) {
   575→      ctx.orderBy.push({
   576→        text: exprToString(o.expr),
   577→        direction: (nameOf(o.type) as OrderItem["direction"]) ?? null,
   578→      });
   579→    }
   580→  }
   581→
   582→  // ---- LIMIT / OFFSET --------------------------------------------
   583→  const lim = select.limit;
   584→  const limArr = Array.isArray(lim)
   585→    ? (lim as Array<Record<string, unknown>>)
   586→    : lim && typeof lim === "object"
   587→      ? [lim as Record<string, unknown>]
   588→      : [];
   589→  for (const l of limArr) {
   590→    const vals = Array.isArray(l.value)
   591→      ? (l.value as Array<Record<string, unknown>>)
   592→      : [];
   593→    const nums = vals.map((v) => nameOf(v?.value) ?? exprToString(v));
   594→    const sep = String(l.seperator ?? l.separator ?? "").toLowerCase();
   595→    if (sep === "offset") {
   596→      if (nums[0] && ctx.limit === null) ctx.limit = nums[0];
   597→      if (nums[1]) ctx.offset = nums[1];
   598→    } else if (sep === "," || sep === "comma") {
   599→      // MySQL: LIMIT offset, count
   600→      if (nums[0]) ctx.offset = nums[0];
   601→      if (nums[1]) ctx.limit = nums[1];
   602→    } else if (nums[0] && ctx.limit === null) {
   603→      ctx.limit = nums[0];
   604→      if (nums[1]) ctx.offset = nums[1];
   605→    }
   606→  }
   607→
   608→  // ---- DISTINCT ---------------------------------------------------
   609→  const dist = select.distinct;
   610→  if (dist) {
   611→    if (typeof dist === "string") ctx.distinct = true;
   612→    else if (typeof dist === "object" && nameOf((dist as Record<string, unknown>).type))
   613→      ctx.distinct = true;
   614→  }
   615→
   616→  // ---- SET OPERATIONS (UNION / INTERSECT / EXCEPT) ----------------
   617→  let next = select._next as Record<string, unknown> | undefined;
   618→  let guard = 0;
   619→  while (next && guard < 8) {
   620→    guard += 1;
   621→    const op = String(next.union ?? next.setop ?? "UNION").toUpperCase();
   622→    ctx.setOps.push(op);
   623→    if (next.type === "select") {
   624→      walkSelect(next, ctx);
   625→      break;
   626→    }
   627→    next = next._next as Record<string, unknown> | undefined;
   628→  }
   629→}
   630→
   631→function summarizeSelect(select: Record<string, unknown> | undefined): string {
   632→  if (!select) return "(empty)";
   633→  const parts: string[] = [];
   634→  const cols = Array.isArray(select.columns) ? select.columns.length : 1;
   635→  parts.push(`SELECT ${cols > 0 ? `${cols} column${cols > 1 ? "s" : ""}` : "*"}`);
   636→  const from = (select.from as Array<Record<string, unknown>> | undefined) ?? [];
   637→  const fromNames = from
   638→    .map((f) => nameOf(f.table) ?? (f.expr ? "(subquery)" : "?"))
   639→    .filter((n) => n !== "?");
   640→  if (fromNames.length) {
   641→    parts.push(
   642→      `FROM ${fromNames[0]}${fromNames.length > 1 ? ` + ${fromNames.length - 1} more` : ""}`,
   643→    );
   644→  }
   645→  if (select.where) parts.push("filtered");
   646→  if (select.groupby) parts.push("grouped");
   647→  return parts.join(" · ");
   648→}

   654→export function buildQueryPlan(
   655→  sql: string,
   656→  dialect: Dialect,
   657→): QueryPlan | { ok: false; error: string; hint?: string } {
   658→  const trimmed = sql.trim();
   659→  if (!trimmed) {
   660→    return { ok: false, error: "Query is empty. Write or pick a SQL query first." };
   661→  }
   662→
   663→  const parser = new Parser();
   664→  let ast: unknown;
   665→  try {
   666→    ast = parser.astify(trimmed, { database: dialect });
   667→  } catch (err) {
   668→    const message = err instanceof Error ? err.message : String(err);
   669→    const hint =
   670→      "Check for missing commas, keywords or unbalanced parentheses. Some functions are dialect-specific — try switching the dialect.";
   671→    return { ok: false, error: message, hint };
   672→  }
   673→
   674→  if (Array.isArray(ast)) ast = ast[0];
   675→  const root = ast as Record<string, unknown> | null;
   676→  if (!root) {
   677→    return { ok: false, error: "Could not parse the statement." };
   678→  }
   679→
   680→  const statementType = String(root.type ?? "unknown").toUpperCase();
   681→
   682→  const ctx: WalkCtx = {
   683→    tables: new Map(),
   684→    joins: [],
   685→    columns: [],
   686→    where: [],
   687→    groupBy: [],
   688→    having: [],
   689→    orderBy: [],
   690→    limit: null,
   691→    offset: null,
   692→    distinct: false,
   693→    setOps: [],
   694→    ctes: [],
   695→    cteNames: new Set(),
   696→    flags: { aggr: false, subquery: false, window: false },
   697→    depth: 0,
   698→  };
   699→
   700→  if (statementType === "SELECT") {
   701→    walkSelect(root, ctx);
   702→  } else {
   703→    // Non-SELECT: extract whatever we can (target + referenced tables).
   704→    const tablesAcc = new Set<string>();
   705→    collectTablesDeep(root, tablesAcc);
   706→    const targetArr = root.table as Array<Record<string, unknown>> | undefined;
   707→    const targetName =
   708→      (targetArr && targetArr.length > 0
   709→        ? nameOf(targetArr[0].table) ?? nameOf(targetArr[0].expr)
   710→        : null) ??
   711→      Array.from(tablesAcc)[0] ??
   712→      "(unknown)";
   713→    const ensure = (id: string) => {
   714→      if (!ctx.tables.has(id))
   715→        ctx.tables.set(id, {
   716→          id,
   717→          name: id,
   718→          alias: null,
   719→          kind: "table",
   720→          joinType: null,
   721→          onText: null,
   722→          joinsTo: [],
   723→          columns: [],
   724→          depth: 0,
   725→        });
   726→    };
   727→    ensure(targetName);
   728→    for (const t of tablesAcc) ensure(t);
   729→  }
   730→
   731→  const tables = Array.from(ctx.tables.values());
   732→  const stats = computeStats(tables, ctx);
   733→
   734→  return {
   735→    ok: true,
   736→    statementType,
   737→    dialect,
   738→    distinct: ctx.distinct,
   739→    ctes: ctx.ctes,
   740→    tables,
   741→    joins: ctx.joins,
   742→    columns: ctx.columns,
   743→    where: ctx.where,
   744→    groupBy: ctx.groupBy,
   745→    having: ctx.having,
   746→    orderBy: ctx.orderBy,
   747→    limit: ctx.limit,
   748→    offset: ctx.offset,
   749→    setOps: ctx.setOps,
   750→    hasSubquery: ctx.flags.subquery,
   751→    stats,
   752→  };
   753→}
   754→
   755→function computeStats(tables: TableInfo[], ctx: WalkCtx): PlanStats {
   756→  const joinCount = ctx.joins.length;
   757→  const filters = ctx.where.length + ctx.having.length;
   758→  const aggregations = ctx.columns.filter((c) => c.aggregation).length;
   759→  const ctes = ctx.ctes.length;
   760→
   761→  const complexity =
   762→    tables.length * 2 +
   763→    joinCount * 2 +
   764→    filters +
   765→    aggregations +
   766→    ctes * 4 +
   767→    (ctx.flags.subquery ? 3 : 0) +
   768→    (ctx.flags.window ? 2 : 0) +
   769→    ctx.groupBy.length +
   770→    ctx.orderBy.length +
   771→    ctx.setOps.length * 3;
   772→
   773→  const label: PlanStats["label"] =
   774→    complexity <= 7 ? "Simple" : complexity <= 16 ? "Moderate" : "Complex";
   775→
   776→  return {
   777→    tables: tables.length,
   778→    joins: joinCount,
   779→    filters,
   780→    aggregations,
   781→    ctes,
   782→    subqueries: ctx.flags.subquery ? 1 : 0,
   783→    complexity,
   784→    label,
   785→  };
   786→}
