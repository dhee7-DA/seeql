   100→   100→function JsonView({ plan }: { plan: QueryPlan }) {
   101→   101→  const [copied, setCopied] = useState(false);
   102→   102→  const json = JSON.stringify({ ...plan, tree: undefined }, null, 2);
   103→   103→
   104→   104→  const copy = async () => {
   105→   105→    try {
   106→   106→      await navigator.clipboard.writeText(json);
   107→   107→      setCopied(true);
   108→   108→      setTimeout(() => setCopied(false), 1600);
   109→   109→    } catch {
   110→   110→      // clipboard unavailable — ignore
   111→   111→    }
   112→   112→  };
   113→   113→
   114→   114→  return (
   115→   115→    <div className="space-y-3">
   116→   116→      <div className="flex items-center justify-between">
   117→   117→        <p className="text-xs text-zinc-400">
   118→   118→          The structured plan the visualizer derived from your query.
   119→   119→        </p>
   120→   120→        <Button
   121→   121→          variant="outline"
   122→   122→          size="sm"
   123→   123→          onClick={copy}
   124→   124→          className="h-7 gap-1.5 border-zinc-700 text-xs text-zinc-300 hover:bg-zinc-800"
   125→   125→        >
   126→   126→          {copied ? <Check className="h-3.5 w-3.5 text-emerald-400" /> : <Copy className="h-3.5 w-3.5" />}

   132→   132→      </pre>
